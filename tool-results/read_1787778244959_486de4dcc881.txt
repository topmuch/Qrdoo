    19→function FadeIn({ children, delay = 0, className = '' }: { children: React.ReactNode; delay?: number; className?: string }) {
    20→  return (
    21→    <motion.div initial={{ opacity: 0, y: 30 }} whileInView={{ opacity: 1, y: 0 }} viewport={{ once: true, margin: '-50px' }} transition={{ duration: 0.6, delay }} className={className}>
    22→      {children}
    23→    </motion.div>
    24→  );
    25→}

    27→function Navbar({ onGoToDashboard }: { onGoToDashboard: () => void }) {
    28→  const [mobileOpen, setMobileOpen] = useState(false);
    29→  const [scrolled, setScrolled] = useState(false);
    30→
    31→  useEffect(() => {
    32→    const onScroll = () => setScrolled(window.scrollY > 20);
    33→    window.addEventListener('scroll', onScroll, { passive: true });
    34→    return () => window.removeEventListener('scroll', onScroll);
    35→  }, []);
    36→
    37→  const links = [
    38→    { label: 'Accueil', href: '#hero' },
    39→    { label: 'Fonctionnalites', href: '#avantages' },
    40→    { label: 'Nouveautes V3', href: '#v3' },
    41→    { label: 'Tarifs', href: '#pricing' },
    42→    { label: 'Contact', href: '#cta-final' },
    43→  ];
    44→
    45→  const handleNavClick = (e: React.MouseEvent<HTMLAnchorElement>, href: string) => {
    46→    e.preventDefault();
    47→    setMobileOpen(false);
    48→    const target = document.querySelector(href);
    49→    if (target) target.scrollIntoView({ behavior: 'smooth' });
    50→  };
    51→
    52→  const navBg = scrolled ? 'bg-white/90 backdrop-blur-md border-b border-gray-200 shadow-sm' : 'bg-transparent';
    53→  const txtColor = scrolled ? 'text-gray-900' : 'text-gray-600';
    54→
    55→  return (
    56→    <nav className={`fixed top-0 left-0 right-0 z-50 transition-all duration-300 ${navBg}`}>
    57→      <div className="max-w-7xl mx-auto px-4 md:px-8 flex items-center justify-between h-16">
    58→        <a href="#hero" onClick={(e) => handleNavClick(e, '#hero')} className="flex items-center gap-2 group">
    59→          <div className="w-8 h-8 rounded-lg bg-violet-600 flex items-center justify-center">
    60→            <QrCode className="w-5 h-5 text-white group-hover:scale-110 transition-transform" />
    61→          </div>
    62→          <span className={`font-bold text-lg tracking-tight ${scrolled ? 'text-gray-900' : 'text-gray-800'}`}>QR Domotik</span>
    63→        </a>
    64→        <div className="hidden md:flex items-center gap-8">
    65→          {links.map((link) => (
    66→            <a key={link.href} href={link.href} onClick={(e) => handleNavClick(e, link.href)} className={`text-sm ${txtColor} hover:text-gray-900 transition-colors`}>
    67→              {link.label}
    68→            </a>
    69→          ))}
    70→          <button onClick={onGoToDashboard} className="ml-2 px-5 py-2.5 rounded-xl bg-violet-600 text-white text-sm font-medium hover:bg-violet-700 transition-colors shadow-sm shadow-violet-600/25">
    71→            Se connecter
    72→          </button>
    73→        </div>
    74→        <button onClick={() => setMobileOpen(!mobileOpen)} className="md:hidden p-2 text-gray-600 hover:text-gray-900 transition-colors" aria-label="Menu">
    75→          {mobileOpen ? <X className="w-6 h-6" /> : <Menu className="w-6 h-6" />}
    76→        </button>
    77→      </div>
    78→      <div className={`md:hidden overflow-hidden transition-all duration-300 ${mobileOpen ? 'max-h-80 opacity-100' : 'max-h-0 opacity-0'}`}>
    79→        <div className="px-4 pb-4 space-y-1 bg-white/95 backdrop-blur-md border-t border-gray-100">
    80→          {links.map((link) => (
    81→            <a key={link.href} href={link.href} onClick={(e) => handleNavClick(e, link.href)} className="block py-3 px-3 text-gray-600 hover:text-gray-900 hover:bg-gray-50 rounded-lg transition-colors">
    82→              {link.label}
    83→            </a>
    84→          ))}
    85→          <button onClick={() => { setMobileOpen(false); onGoToDashboard(); }} className="w-full mt-2 px-5 py-3 rounded-xl bg-violet-600 text-white text-sm font-medium hover:bg-violet-700 transition-colors">
    86→            Se connecter
    87→          </button>
    88→        </div>
    89→      </div>
    90→    </nav>
    91→  );
    92→}

   100→    </motion.section>
   101→  );
   102→}
