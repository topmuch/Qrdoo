   119→  const [promos, setPromos] = useState<PromoData[]>([]);
   120→  const [flashSales, setFlashSales] = useState<FlashSaleData[]>([]);

   122→  const [transactions, setTransactions] = useState<TransactionData[]>([]);

   125→  // Filters

   128→  const [pMerchant, setPMerchant] = useState('all');

   133→  const [tStatus, setTStatus] = useState('all');
   134→
   135→  // Dialogs
   136→  const [merchantDialog, setMerchantDialog] = useState(false);
   137→  const [promoDialog, setPromoDialog] = useState(false);
   138→  const [flashDialog, setFlashDialog] = useState(false);
   139→  const [couponDialog, setCouponDialog] = useState(false);
   140→  const [editItem, setEditItem] = useState<Record<string, string> | null>(null);
   141→
   142→  // Forms

   144→  const [pf, setPf] = useState({ merchantId: '', title: '', description: '', originalPrice: '', promoPrice: '', validUntil: '', category: '' });
   145→  const [ff, setFf] = useState({ merchantId: '', promoId: '', title: '', flashPrice: '', startsAt: '', endsAt: '', maxRedemptions: '', costEuros: '0.5' });
   146→  const [cf, setCf] = useState({ merchantId: '', discountType: 'percentage', discountValue: '', maxUses: '1', validUntil: '' });
   147→
   148→  // Submitting
   149→  const [submitting, setSubmitting] = useState(false);
   150→
   151→  // Countdown for flash sales
   152→  const [now, setNow] = useState(Date.now());
   153→  const countdownRef = useRef<ReturnType<typeof setInterval> | null>(null);
   154→  useEffect(() => {
   155→    countdownRef.current = setInterval(() => setNow(Date.now()), 1000);
   156→    return () => { if (countdownRef.current) clearInterval(countdownRef.current); };
   157→  }, []);
   158→
   159→  // ── Fetch ────────────────────────────────────────────────────
   160→  const fetchAll = useCallback(async () => {
   161→    try {
   162→      setLoading(true);
   163→      const [m, p, f, c, t] = await Promise.all([
   164→        fetch('/api/client/merchants').then(r => r.json()).then(d => Array.isArray(d) ? d : d.data ?? []),
   165→        fetch('/api/client/promos').then(r => r.json()).then(d => Array.isArray(d) ? d : d.data ?? []),
   166→        fetch('/api/client/flash-sales?status=all').then(r => r.json()).then(d => Array.isArray(d) ? d : d.data ?? []),
   167→        fetch('/api/client/coupons?status=all').then(r => r.json()).then(d => Array.isArray(d) ? d : d.data ?? []),
   168→        fetch('/api/client/transactions?limit=100').then(r => r.json()).then(d => d.transactions ?? d.data ?? []),

   174→  useEffect(() => { fetchAll(); }, [fetchAll]);

   178→    setSubmitting(true);
   179→    try {
   180→      const res = await fetch(url, { method, headers: { 'Content-Type': 'application/json' }, body: body ? JSON.stringify(body) : undefined });
   181→      if (!res.ok) { const d = await res.json().catch(() => ({})); throw new Error(d.error || 'Erreur'); }
   182→      return true;
   183→    } finally { setSubmitting(false); }
   184→  };
   185→
   186→  const resetMerchantForm = () => setMf({ name: '', category: '', address: '', phone: '', description: '', website: '' });
   187→  const resetPromoForm = () => setPf({ merchantId: '', title: '', description: '', originalPrice: '', promoPrice: '', validUntil: '', category: '' });
   188→  const resetFlashForm = () => setFf({ merchantId: '', promoId: '', title: '', flashPrice: '', startsAt: '', endsAt: '', maxRedemptions: '', costEuros: '0.5' });

   194→    if (editItem) {
   195→      await apiCall(`/api/client/merchants/${editItem.id}`, 'PATCH', body);

   197→    } else {
   198→      await apiCall('/api/client/merchants', 'POST', body);
   199→      toast.success('Commerçant créé');
   200→    }
   201→    setMerchantDialog(false); setEditItem(null); resetMerchantForm(); fetchAll();
   202→  };
   203→
   204→  const handleDeleteMerchant = async (id: string) => {
   205→    await apiCall(`/api/client/merchants/${id}`, 'DELETE');
   206→    toast.success('Commerçant supprimé'); fetchAll();
   207→  };
   208→

   210→    await apiCall(`/api/client/merchants/${m.id}`, 'PATCH', { [field]: !m[field] });
   211→    toast.success(`${field === 'isActive' ? 'Statut' : 'Vérification'} mis à jour`); fetchAll();
   212→  };
   213→

   217→    if (editItem) {
   218→      await apiCall(`/api/client/promos/${editItem.id}`, 'PATCH', body);
   219→      toast.success('Promotion mise à jour');
   220→    } else {
   221→      await apiCall('/api/client/promos', 'POST', body);
   222→      toast.success('Promotion créée');
   223→    }
   224→    setPromoDialog(false); setEditItem(null); resetPromoForm(); fetchAll();
   225→  };
   226→
   227→  const handleDeletePromo = async (id: string) => {
   228→    await apiCall(`/api/client/promos/${id}`, 'DELETE'); toast.success('Promotion supprimée'); fetchAll();
   229→  };
   230→
   231→  const handleSaveFlash = async () => {
   232→    if (!ff.merchantId || !ff.flashPrice || !ff.startsAt || !ff.endsAt) return;
   233→    const body = { merchantId: ff.merchantId, promoId: ff.promoId || undefined, title: ff.title || undefined, flashPrice: parseFloat(ff.flashPrice), startsAt: ff.startsAt, endsAt: ff.endsAt, maxRedemptions: ff.maxRedemptions ? parseInt(ff.maxRedemptions) : undefined, costEuros: parseFloat(ff.costEuros) || 0.5 };
   234→    await apiCall('/api/client/flash-sales', 'POST', body);
   235→    toast.success('Vente flash créée'); setFlashDialog(false); resetFlashForm(); fetchAll();
   236→  };
   237→
   238→  const handleFlashAction = async (id: string, action: 'active' | 'cancelled' | 'delete') => {
   239→    if (action === 'delete') {
   240→      await apiCall(`/api/client/flash-sales/${id}`, 'DELETE'); toast.success('Vente flash supprimée');
   241→    } else {
   242→      await apiCall(`/api/client/flash-sales/${id}`, 'PATCH', { status: action });
   243→      toast.success(action === 'active' ? 'Vente flash activée' : 'Vente flash annulée');
   244→    }
   245→    fetchAll();
   246→  };
   247→
   248→  const handleSaveCoupon = async () => {
   249→    if (!cf.merchantId || !cf.discountValue) return;
   250→    const body = { merchantId: cf.merchantId, discountType: cf.discountType, discountValue: parseFloat(cf.discountValue), maxUses: parseInt(cf.maxUses) || 1, validUntil: cf.validUntil || undefined };
   251→    await apiCall('/api/client/coupons', 'POST', body);
   252→    toast.success('Coupon créé'); setCouponDialog(false); resetCouponForm(); fetchAll();
   253→  };
   254→
   255→  const handleScanCoupon = async (id: string) => {

   317→            <TabsTrigger value="flash" className="gap-1.5"><Zap className="h-3.5 w-3.5" /> Ventes Flash</TabsTrigger>
