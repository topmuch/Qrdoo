   116→   116→/*  SVG → PNG conversion utility (browser-only, zero native deps)       */

   118→   118→
   119→   119→/** Convert an SVG Blob to a PNG data-URL using the browser Canvas API */
   120→   120→function svgBlobToPngDataUrl(svgBlob: Blob, size: number): Promise<string> {
   121→   121→  return new Promise((resolve, reject) => {
   122→   122→    const url = URL.createObjectURL(svgBlob);
   123→   123→    const img = new Image();
   124→   124→    img.onload = () => {
   125→   125→      const canvas = document.createElement('canvas');
   126→   126→      canvas.width = size;
   127→   127→      canvas.height = size;
   128→   128→      const ctx = canvas.getContext('2d');
   129→   129→      if (!ctx) { URL.revokeObjectURL(url); reject(new Error('No 2d context')); return; }
   130→   130→      ctx.drawImage(img, 0, 0, size, size);
   131→   131→      URL.revokeObjectURL(url);
   132→   132→      resolve(canvas.toDataURL('image/png'));
   133→   133→    };
   134→   134→    img.onerror = () => { URL.revokeObjectURL(url); reject(new Error('SVG image load failed')); };
   135→   135→    img.src = url;
   136→   136→  });
   137→   137→}
   138→   138→
   139→   139→/** Convert an inline SVG element to a PNG data-URL */
   140→   140→function svgElementToPngDataUrl(svgEl: SVGSVGElement, size: number): Promise<string> {
   141→   141→  return new Promise((resolve, reject) => {
   142→   142→    const svgData = new XMLSerializer().serializeToString(svgEl);
   143→   143→    const svgBlob = new Blob([svgData], { type: 'image/svg+xml;charset=utf-8' });
   144→   144→    svgBlobToPngDataUrl(svgBlob, size).then(resolve).catch(reject);
   145→   145→  });
   146→   146→}

   148→   148→/**
   149→   149→ * Fallback QR PNG generator using qrcode.react (pure JS, no native deps).
   150→   150→ * Renders QRCodeSVG into a hidden container, extracts the SVG, converts to PNG.
   151→   151→ */
   152→   152→async function generateFallbackQrPng(
   153→   153→  data: string,
   154→   154→  size: number,
   155→   155→  fgColor: string,
   156→   156→  bgColor: string,
   157→   157→  level: string,
   158→   158→): Promise<string> {
   159→   159→  const container = document.createElement('div');
   160→   160→  container.style.cssText = 'position:fixed;left:-9999px;top:-9999px;';
   161→   161→  document.body.appendChild(container);
   162→   162→
   163→   163→  const root = createRoot(container);
   164→   164→  root.render(
   165→   165→    createElement(QRCodeSVG, {
   166→   166→      value: data,
   167→   167→      size,
   168→   168→      bgColor,
   169→   169→      fgColor,
   170→   170→      level: level as 'L' | 'M' | 'Q' | 'H',
   171→   171→    }),
   172→   172→  );
   173→   173→
   174→   174→  // Wait for React to commit the render
   175→   175→  await new Promise((r) => setTimeout(r, 60));
   176→   176→
   177→   177→  const svgEl = container.querySelector('svg');
   178→   178→  if (!svgEl) {
   179→   179→    root.unmount();
   180→   180→    document.body.removeChild(container);
   181→   181→    throw new Error('Fallback: SVG element not rendered');
   182→   182→  }
   183→   183→
   184→   184→  const pngUrl = await svgElementToPngDataUrl(svgEl, size);
   185→   185→  root.unmount();
   186→   186→  document.body.removeChild(container);
   187→   187→  return pngUrl;
   188→   188→}
