    95→// ── Component ──
    96→export function SetupPageContent({ params }: { params: Promise<{ token: string }> }) {
    97→  const { token } = use(params);
    98→  const { data: session } = useSession();
    99→  const router = useRouter();
   100→
   101→  // Token state
   102→  const [tokenStatus, setTokenStatus] = useState<TokenStatus>('loading');
   103→  const [plaqueInfo, setPlaqueInfo] = useState<PlaqueInfo | null>(null);
   104→  const [claimedInfo, setClaimedInfo] = useState<{ hubSlug?: string; homeName?: string }>({});
   105→
   106→  // Step navigation
   107→  const [step, setStep] = useState<SetupStep>('welcome');
   108→  const [direction, setDirection] = useState(1);
   109→
   110→  // Account form
   111→  const [fullName, setFullName] = useState('');
   112→  const [email, setEmail] = useState('');
   113→  const [password, setPassword] = useState('');
   114→  const [showPassword, setShowPassword] = useState(false);
   115→
   116→  // Plan
   117→  const [selectedPlan, setSelectedPlan] = useState<string>('famille');
   118→
   119→  // PIN
   120→  const [pin, setPin] = useState('');
   121→  const [pinConfirm, setPinConfirm] = useState('');
   122→
   123→  // Home
   124→  const [homeName, setHomeName] = useState('');
   125→
   126→  // Submission
   127→  const [submitting, setSubmitting] = useState(false);
   128→
   129→  // ── Check token on mount ──
   130→  const checkToken = useCallback(async () => {
   131→    try {
   132→      const res = await fetch(`/api/setup/${encodeURIComponent(token)}`);
   133→      const data = await res.json();
   134→      if (data.status === 'available') {
   135→        setTokenStatus('available');
   136→        setPlaqueInfo(data.plaque);
   137→      } else if (data.status === 'claimed') {
   138→        setTokenStatus('claimed');
   139→        setClaimedInfo({ hubSlug: data.hubSlug, homeName: data.homeName });
   140→      } else {
   141→        setTokenStatus('not_found');
   142→      }
   143→    } catch {
   144→      setTokenStatus('error');
   145→    }
   146→  }, [token]);
   147→
   148→  useEffect(() => { checkToken(); }, [checkToken]);
   149→
   150→  // Pre-fill if logged in
   151→  useEffect(() => {
   152→    if (session?.user) {
   153→      const u = session.user as Record<string, any>;
   154→      setEmail(u?.email || '');
   155→      setFullName(u?.name || '');
   156→    }
   157→  }, [session]);
   158→
   159→  // ── Step navigation helpers ──
   160→  const goNext = () => {
   161→    setDirection(1);
   162→    const order: SetupStep[] = ['welcome', 'account', 'plan', 'pin', 'home', 'success'];
   163→    const idx = order.indexOf(step);
   164→    if (idx < order.length - 1) setStep(order[idx + 1]);
   165→  };
   166→
   167→  const goBack = () => {
   168→    setDirection(-1);
   169→    const order: SetupStep[] = ['welcome', 'account', 'plan', 'pin', 'home', 'success'];
   170→    const idx = order.indexOf(step);
   171→    if (idx > 0) setStep(order[idx - 1]);
   172→  };
   173→
   174→  // ── Validate & navigate ──
   175→  const handleWelcomeNext = () => goNext();
   176→
   177→  const handleAccountNext = () => {
   178→    if (!fullName.trim()) { toast.error('Entrez votre nom'); return; }
   179→    if (!email.trim() || !email.includes('@')) { toast.error('Email invalide'); return; }
   180→    if (!session?.user && (!password || password.length < 6)) { toast.error('Mot de passe requis (6+ caractères)'); return; }
   181→    goNext();
   182→  };
   183→
   184→  const handlePlanNext = () => {
   185→    if (!selectedPlan) { toast.error('Choisissez un plan'); return; }
   186→    goNext();
   187→  };
   188→
   189→  const handlePinNext = () => {
   190→    if (!/^\d{4}$/.test(pin)) { toast.error('Le PIN doit être 4 chiffres'); return; }
   191→    if (pin !== pinConfirm) { toast.error('Les PINs ne correspondent pas'); return; }
   192→    goNext();
   193→  };
   194→
   195→  const handleHomeNext = () => {
   196→    if (!homeName.trim()) { toast.error('Nommez votre logement'); return; }
   197→    goNext();
   198→  };
   199→
   200→  // ── Submit ──
   201→  const handleSubmit = async () => {
   202→    setSubmitting(true);
   203→    try {
   204→      const body: Record<string, string> = {
   205→        email: email.trim(),
   206→        fullName: fullName.trim(),
   207→        pin,
   208→        homeName: homeName.trim(),
   209→        plan: selectedPlan,
   210→      };
   211→
   212→      // If user is not logged in, include password
   213→      if (!session?.user) {
   214→        body.password = password;
   215→      } else {
   216→        body.existingUserId = (session.user as Record<string, any>)?.id;
   217→      }
   218→
   219→      const res = await fetch(`/api/setup/${encodeURIComponent(token)}`, {
   220→        method: 'POST',
   221→        headers: { 'Content-Type': 'application/json' },
   222→        body: JSON.stringify(body),
   223→      });
   224→      const data = await res.json();
   225→
   226→      if (!res.ok) {
   227→        throw new Error(data.error || 'Erreur lors de la configuration');
   228→      }
   229→
   230→      // Show success step
   231→      setDirection(1);
   232→      setStep('success');
   233→
   234→      // If new user, auto-login after a delay
   235→      if (data.isNewUser) {
   236→        setTimeout(async () => {
   237→          const loginRes = await fetch('/api/auth/callback/credentials', {
   238→            method: 'POST',
   239→            headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
   240→            body: new URLSearchParams({
   241→              email: email.trim(),
   242→              password: password,
   243→              callbackUrl: '/',
   244→            }).toString(),
   245→          });
   246→          // If login succeeds, redirect to dashboard; otherwise just show success
   247→          if (loginRes.ok) {
   248→            window.location.href = '/';
   249→          }
   250→        }, 3000);
   251→      } else {
   252→        // Already logged in → redirect to hub after delay
   253→        setTimeout(() => {
   254→          window.location.href = data.hubSlug ? `/hub/${data.hubSlug}` : '/';
   255→        }, 3000);
   256→      }
   257→    } catch (e: any) {
   258→      toast.error(e.message || 'Erreur');
   259→    } finally {
   260→      setSubmitting(false);
   261→    }
   262→  };
   263→
   264→  // ── Step indicator ──
   265→  const stepOrder: SetupStep[] = ['welcome', 'account', 'plan', 'pin', 'home', 'success'];
   266→  const currentIdx = stepOrder.indexOf(step);
   267→
   268→  // ── Loading state ──
   269→  if (tokenStatus === 'loading') {
   270→    return (
   271→      <div className="min-h-screen flex items-center justify-center bg-gradient-to-br from-violet-950 via-slate-900 to-slate-950">
   272→        <motion.div
   273→          initial={{ opacity: 0, scale: 0.8 }}
   274→          animate={{ opacity: 1, scale: 1 }}
   275→          className="flex flex-col items-center gap-4"
   276→        >
   277→          <div className="relative">
   278→            <div className="h-16 w-16 rounded-2xl bg-gradient-to-br from-violet-500 to-purple-600 flex items-center justify-center shadow-lg shadow-violet-500/30">
   279→              <QrCode className="h-8 w-8 text-white" />
   280→            </div>
   281→            <div className="absolute -bottom-1 -right-1 h-5 w-5 rounded-full bg-emerald-500 border-2 border-slate-900 flex items-center justify-center">
   282→              <div className="h-2 w-2 rounded-full bg-white animate-pulse" />
   283→            </div>
   284→          </div>
   285→          <div className="text-center">
   286→            <p className="text-violet-200 font-medium">Vérification de votre plaque...</p>
   287→            <p className="text-xs text-slate-400 mt-1 font-mono">{token}</p>
   288→          </div>
   289→        </motion.div>
   290→      </div>
   291→    );
   292→  }
   293→
   294→  // ── Error states ──
   295→  if (tokenStatus === 'not_found' || tokenStatus === 'error') {
   296→    const Icon = tokenStatus === 'error' ? AlertCircle : XCircle;
   297→    const msg = tokenStatus === 'not_found'
   298→      ? { title: 'Plaque non trouvée', desc: "Ce code d'activation n'existe pas. Vérifiez votre plaque et réessayez." }
   299→      : { title: 'Erreur', desc: 'Impossible de vérifier la plaque. Réessayez.' };
   300→    return (
   301→      <div className="min-h-screen flex items-center justify-center bg-gradient-to-br from-violet-950 via-slate-900 to-slate-950 p-4">
   302→        <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} className="w-full max-w-md">
   303→          <div className="rounded-2xl bg-slate-800/80 backdrop-blur-xl border border-slate-700 p-8 text-center">
   304→            <div className="mx-auto mb-4 flex h-16 w-16 items-center justify-center rounded-full bg-red-500/20">
   305→              <Icon className="h-8 w-8 text-red-400" />
   306→            </div>
   307→            <h1 className="text-xl font-bold text-white mb-2">{msg.title}</h1>
   308→            <p className="text-sm text-slate-400 mb-1 font-mono">{token}</p>
   309→            <p className="text-sm text-slate-400 mb-6">{msg.desc}</p>
   310→            <button onClick={() => { setTokenStatus('loading'); checkToken(); }} className="w-full h-11 rounded-xl bg-slate-700 text-slate-200 text-sm font-medium hover:bg-slate-600 transition-colors">
   311→              Réessayer
   312→            </button>
   313→          </div>
   314→        </motion.div>
   315→      </div>
   316→    );
   317→  }
   318→
   319→  // ── Already claimed ──
   320→  if (tokenStatus === 'claimed') {
   321→    return (
   322→      <div className="min-h-screen flex items-center justify-center bg-gradient-to-br from-violet-950 via-slate-900 to-slate-950 p-4">
   323→        <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} className="w-full max-w-md">
   324→          <div className="rounded-2xl bg-slate-800/80 backdrop-blur-xl border border-slate-700 p-8 text-center">
   325→            <div className="mx-auto mb-4 flex h-16 w-16 items-center justify-center rounded-full bg-emerald-500/20">
   326→              <CheckCircle2 className="h-8 w-8 text-emerald-400" />
   327→            </div>
   328→            <h1 className="text-xl font-bold text-white mb-2">Plaque déjà configurée</h1>
   329→            {claimedInfo.homeName && (
   330→              <p className="text-sm text-slate-400 mb-1">Logement : <span className="text-violet-300 font-semibold">{claimedInfo.homeName}</span></p>
   331→            )}
   332→            <p className="text-sm text-slate-400 mb-6">Cette plaque est déjà liée à un compte.</p>
   333→            {claimedInfo.hubSlug && (
   334→              <button
   335→                onClick={() => router.push(`/hub/${claimedInfo.hubSlug}`)}
   336→                className="w-full h-11 rounded-xl bg-gradient-to-r from-violet-600 to-purple-600 text-white text-sm font-semibold hover:from-violet-700 hover:to-purple-700 transition-all shadow-lg shadow-violet-600/25"
   337→              >
   338→                Accéder au Hub
   339→              </button>
   340→            )}
   341→            <button onClick={() => router.push('/')} className="w-full h-11 mt-3 rounded-xl bg-slate-700 text-slate-200 text-sm font-medium hover:bg-slate-600 transition-colors">
   342→              Aller à l'accueil
   343→            </button>
   344→          </div>
   345→        </motion.div>
   346→      </div>
   347→    );
   348→  }
   349→
   350→  // ── Main multi-step flow ──
   351→  return (
   352→    <div className="min-h-screen bg-gradient-to-br from-violet-950 via-slate-900 to-slate-950 flex flex-col">
   353→      {/* Header bar with step indicator */}
   354→      <div className="w-full px-4 pt-6 pb-2">
   355→        <div className="max-w-md mx-auto">
   356→          {/* Logo + progress */}
   357→          <div className="flex items-center justify-between mb-4">
   358→            <div className="flex items-center gap-2">
   359→              <div className="h-8 w-8 rounded-lg bg-gradient-to-br from-violet-500 to-purple-600 flex items-center justify-center">
   360→                <QrCode className="h-4 w-4 text-white" />
   361→              </div>
   362→              <span className="text-sm font-bold text-white tracking-tight">QR Domotik</span>
   363→            </div>
   364→            <span className="text-xs text-slate-400">
   365→              Étape {Math.min(currentIdx + 1, 5)} / 5
   366→            </span>
   367→          </div>
   368→          {/* Progress bar */}
   369→          <div className="h-1.5 rounded-full bg-slate-800 overflow-hidden">
   370→            <motion.div
   371→              className="h-full rounded-full bg-gradient-to-r from-violet-500 to-purple-500"
   372→              initial={{ width: 0 }}
   373→              animate={{ width: `${(Math.min(currentIdx, 4) / 5) * 100}%` }}
   374→              transition={{ duration: 0.4, ease: 'easeOut' }}
   375→            />
   376→          </div>
   377→        </div>
   378→      </div>
   379→
   380→      {/* Step content */}
   381→      <div className="flex-1 flex items-center justify-center px-4 py-6">
   382→        <div className="w-full max-w-md relative overflow-hidden">
   383→          <AnimatePresence mode="wait" custom={direction}>
   384→            {/* ─── STEP 1: Welcome ─── */}
   385→            {step === 'welcome' && (
   386→              <motion.div
   387→                key="welcome"
   388→                custom={direction}
   389→                variants={slideVariants}
   390→                initial="enter"
   391→                animate="center"
   392→                exit="exit"
   393→                transition={{ duration: 0.35, ease: 'easeInOut' }}
   394→                className="space-y-6"
   395→              >
   396→                <div className="text-center space-y-4">
   397→                  <motion.div
   398→                    initial={{ scale: 0 }}
   399→                    animate={{ scale: 1 }}
   400→                    transition={{ delay: 0.15, type: 'spring', stiffness: 200 }}
   401→                    className="mx-auto flex h-24 w-24 items-center justify-center rounded-3xl bg-gradient-to-br from-violet-500 to-purple-600 shadow-2xl shadow-violet-500/30"
   402→                  >
   403→                    <QrCode className="h-12 w-12 text-white" />
   404→                  </motion.div>
   405→                  <motion.h1 initial={fadeUp.initial} animate={fadeUp.animate} transition={{ delay: 0.2 }} className="text-2xl font-bold text-white">
   406→                    Bienvenue !
   407→                  </motion.h1>
   408→                  <motion.p initial={fadeUp.initial} animate={fadeUp.animate} transition={{ delay: 0.3 }} className="text-slate-400 text-sm leading-relaxed">
   409→                    Votre plaque QR est prête à être configurée.<br />
   410→                    En quelques étapes, créez votre espace domotique intelligent.
   411→                  </motion.p>
   412→                  {plaqueInfo && (
   413→                    <motion.div initial={fadeUp.initial} animate={fadeUp.animate} transition={{ delay: 0.35 }} className="inline-flex items-center gap-2 rounded-full bg-slate-800/80 border border-slate-700 px-4 py-2">
   414→                      <span className="text-xs text-slate-400">Plaque :</span>
   415→                      <span className="text-xs font-mono text-violet-300 font-semibold">{plaqueInfo.activationCode}</span>
   416→                    </motion.div>
   417→                  )}
   418→                </div>
   419→
   420→                <motion.div initial={fadeUp.initial} animate={fadeUp.animate} transition={{ delay: 0.4 }} className="space-y-3">
   421→                  <div className="grid grid-cols-3 gap-3">
   422→                    {[
   423→                      { icon: Wifi, label: 'Wi-Fi', color: 'text-emerald-400' },
   424→                      { icon: Shield, label: 'Sécurité', color: 'text-amber-400' },
   425→                      { icon: Users, label: 'Famille', color: 'text-violet-400' },
   426→                    ].map((item, i) => (

   441→                <motion.button

   445→                  onClick={handleWelcomeNext}

   454→            {/* ─── STEP 2: Account ─── */}

   456→              <motion.div
   457→                key="account"

   459→                variants={slideVariants}

   465→              >
   466→                <div className="text-center">
