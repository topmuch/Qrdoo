   124→  return (
   125→    <span className="inline-flex items-center gap-0.5">
   126→      {Array.from({ length: 5 }).map((_, i) => (

   128→          key={i}

   159→// ── Component ──────────────────────────────────────────────────
   160→export function AdminArtisans() {
   161→  // Data state
   162→  const [professionals, setProfessionals] = useState<ProfessionalData[]>([]);
   163→  const [loading, setLoading] = useState(true);
   164→  const [error, setError] = useState<string | null>(null);
   165→
   166→  // Filters
   167→  const [search, setSearch] = useState('');
   168→  const [categoryFilter, setCategoryFilter] = useState<string>('all');
   169→  const [verifiedFilter, setVerifiedFilter] = useState<boolean | null>(null);
   170→
   171→  // Create / Edit dialog
   172→  const [formOpen, setFormOpen] = useState(false);
   173→  const [editingId, setEditingId] = useState<string | null>(null);
   174→  const [form, setForm] = useState<FormFields>(EMPTY_FORM);
   175→  const [formError, setFormError] = useState<string | null>(null);
   176→  const [formLoading, setFormLoading] = useState(false);
   177→
   178→  // Detail dialog
   179→  const [detailOpen, setDetailOpen] = useState(false);
   180→  const [detailPro, setDetailPro] = useState<ProfessionalData | null>(null);
   181→
   182→  // Delete dialog
   183→  const [deleteOpen, setDeleteOpen] = useState(false);
   184→  const [deleteId, setDeleteId] = useState<string | null>(null);
   185→  const [deleteLoading, setDeleteLoading] = useState(false);
   186→
   187→  // Toggle loading states
   188→  const [togglingId, setTogglingId] = useState<string | null>(null);
   189→  const [togglingField, setTogglingField] = useState<'isActive' | 'isVerified' | null>(null);
   190→
   191→  // ── Fetch ─────────────────────────────────────────────────────
   192→  const fetchProfessionals = useCallback(async () => {
   193→    try {
   194→      setLoading(true);
   195→      setError(null);
   196→      const params = new URLSearchParams();
   197→      if (search.trim()) params.set('search', search.trim());
   198→      if (categoryFilter !== 'all') params.set('category', categoryFilter);
   199→      if (verifiedFilter !== null) params.set('isVerified', String(verifiedFilter));
   200→
   201→      const res = await fetch(`/api/client/professionals?${params.toString()}`);
   202→      if (!res.ok) throw new Error('Erreur lors du chargement des professionnels');
   203→      const data = await res.json();
   204→      setProfessionals(Array.isArray(data) ? data : data.professionals ?? data.data ?? []);
   205→    } catch (err) {
   206→      setError(err instanceof Error ? err.message : 'Erreur inconnue');
   207→    } finally {
   208→      setLoading(false);
   209→    }
   210→  }, [search, categoryFilter, verifiedFilter]);
   211→
   212→  useEffect(() => {
   213→    fetchProfessionals();
   214→  }, [fetchProfessionals]);
   215→
   216→  // ── Stats ─────────────────────────────────────────────────────
   217→  const totalPros = professionals.length;
   218→  const verifiedCount = professionals.filter((p) => p.isVerified).length;
   219→  const activeCount = professionals.filter((p) => p.isActive).length;
   220→
   221→  // ── Form helpers ──────────────────────────────────────────────
   222→  const openCreateForm = () => {
   223→    setEditingId(null);
   224→    setForm(EMPTY_FORM);
   225→    setFormError(null);
   226→    setFormOpen(true);
   227→  };
   228→
   229→  const openEditForm = (pro: ProfessionalData) => {
   230→    setEditingId(pro.id);
   231→    setForm({
   232→      businessName: pro.businessName,
   233→      category: pro.category,
   234→      subcategory: pro.subcategory ?? '',
   235→      description: pro.description ?? '',
   236→      location: pro.location ?? '',
   237→      serviceRadiusKm: pro.serviceRadiusKm,
   238→      hourlyRate: pro.hourlyRate != null ? String(pro.hourlyRate) : '',
   239→      isUrgentAvailable: pro.isUrgentAvailable,
   240→      isVerified: pro.isVerified,
   241→    });
   242→    setFormError(null);
   243→    setFormOpen(true);
   244→  };
   245→
   246→  const handleSubmit = async () => {
   247→    if (!form.businessName.trim() || !form.category) {
   248→      setFormError('Le nom de l\'entreprise et la catégorie sont requis');
   249→      return;
   250→    }
   251→
   252→    try {
   253→      setFormLoading(true);
   254→      setFormError(null);
   255→
   256→      const payload = {
   257→        businessName: form.businessName.trim(),
   258→        category: form.category,
   259→        subcategory: form.subcategory.trim() || null,
   260→        description: form.description.trim() || null,
   261→        location: form.location.trim() || null,
   262→        serviceRadiusKm: form.serviceRadiusKm,
   263→        hourlyRate: form.hourlyRate ? parseFloat(form.hourlyRate) : null,
   264→        isUrgentAvailable: form.isUrgentAvailable,
   265→        isVerified: form.isVerified,
   266→      };
   267→
   268→      let res: Response;
   269→      if (editingId) {
   270→        res = await fetch(`/api/client/professionals/${editingId}`, {
   271→          method: 'PATCH',
   272→          headers: { 'Content-Type': 'application/json' },
   273→          body: JSON.stringify(payload),
   274→        });
   275→      } else {
   276→        res = await fetch('/api/client/professionals', {
   277→          method: 'POST',
   278→          headers: { 'Content-Type': 'application/json' },
   279→          body: JSON.stringify(payload),
   280→        });
   281→      }
   282→
   283→      const data = await res.json();
   284→      if (!res.ok) {
   285→        setFormError(data.error || 'Erreur lors de l\'enregistrement');
   286→        return;
   287→      }
   288→
   289→      setFormOpen(false);
   290→      fetchProfessionals();
   291→    } catch {
   292→      setFormError('Erreur réseau, veuillez réessayer');
   293→    } finally {
   294→      setFormLoading(false);
   295→    }
   296→  };
   297→
   298→  // ── Toggle handler ────────────────────────────────────────────
   299→  const handleToggle = async (id: string, field: 'isActive' | 'isVerified', currentValue: boolean) => {
   300→    setTogglingId(id);
   301→    setTogglingField(field);
   302→    try {
   303→      const res = await fetch(`/api/client/professionals/${id}`, {
   304→        method: 'PATCH',
   305→        headers: { 'Content-Type': 'application/json' },
   306→        body: JSON.stringify({ [field]: !currentValue }),
   307→      });
   308→      if (!res.ok) throw new Error();
   309→      fetchProfessionals();
   310→    } catch {
   311→      // silent fail — will revert visually on next fetch
   312→    } finally {
   313→      setTogglingId(null);
   314→      setTogglingField(null);
   315→    }
   316→  };
   317→
   318→  // ── Delete handler ────────────────────────────────────────────
   319→  const handleDelete = async () => {
   320→    if (!deleteId) return;
   321→    try {
   322→      setDeleteLoading(true);
   323→      const res = await fetch(`/api/client/professionals/${deleteId}`, {
   324→        method: 'PATCH',
   325→        headers: { 'Content-Type': 'application/json' },
   326→        body: JSON.stringify({ isActive: false }),
   327→      });
   328→      if (!res.ok) throw new Error();
   329→      setDeleteOpen(false);
   330→      setDeleteId(null);
   331→      fetchProfessionals();
   332→    } catch {
   333→      // silent
   334→    } finally {
   335→      setDeleteLoading(false);
   336→    }
   337→  };
   338→
   339→  // ── Render ────────────────────────────────────────────────────
   340→  return (
   341→    <Card>
   342→      <CardHeader>
   343→        <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   344→          <div>
   345→            <CardTitle>Professionnels</CardTitle>
   346→            <CardDescription>
   347→              Gestion complète des artisans et professionnels de la plateforme
   348→            </CardDescription>
   349→          </div>
   350→          <Button size="sm" onClick={openCreateForm}>
   351→            <Plus className="mr-2 h-4 w-4" />
   352→            Ajouter un professionnel
   353→          </Button>
   354→        </div>
   355→      </CardHeader>
   356→
   357→      <CardContent className="space-y-4">
   358→        {/* ── Stats Bar ─────────────────────────────────────────── */}
   359→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-3">
   360→          <div className="flex items-center gap-3 rounded-lg border p-4">
   361→            <div className="flex h-10 w-10 items-center justify-center rounded-full bg-slate-100 dark:bg-slate-800">
   362→              <Users className="h-5 w-5 text-slate-600 dark:text-slate-300" />
   363→            </div>
   364→            <div>
   365→              <p className="text-2xl font-bold">{totalPros}</p>
   366→              <p className="text-xs text-muted-foreground">Total professionnels</p>
   367→            </div>
   368→          </div>
   369→          <div className="flex items-center gap-3 rounded-lg border p-4">
   370→            <div className="flex h-10 w-10 items-center justify-center rounded-full bg-emerald-100 dark:bg-emerald-900/30">
   371→              <ShieldCheck className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   372→            </div>
   373→            <div>
   374→              <p className="text-2xl font-bold">{verifiedCount}</p>
   375→              <p className="text-xs text-muted-foreground">Vérifiés</p>
   376→            </div>
   377→          </div>
   378→          <div className="flex items-center gap-3 rounded-lg border p-4">
   379→            <div className="flex h-10 w-10 items-center justify-center rounded-full bg-blue-100 dark:bg-blue-900/30">
   380→              <Activity className="h-5 w-5 text-blue-600 dark:text-blue-400" />
   381→            </div>
   382→            <div>
   383→              <p className="text-2xl font-bold">{activeCount}</p>
   384→              <p className="text-xs text-muted-foreground">Actifs</p>
   385→            </div>
   386→          </div>
   387→        </div>
   388→
   389→        {/* ── Filters ───────────────────────────────────────────── */}
   390→        <div className="flex flex-col gap-3 sm:flex-row sm:items-center">
   391→          <div className="relative flex-1 max-w-sm">
   392→            <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   393→            <Input
   394→              placeholder="Rechercher par nom ou lieu..."
   395→              value={search}
   396→              onChange={(e) => setSearch(e.target.value)}
   397→              className="pl-9"
   398→            />
   399→          </div>
   400→          <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   401→            <SelectTrigger className="w-full sm:w-48">
   402→              <SelectValue placeholder="Catégorie" />
   403→            </SelectTrigger>
   404→            <SelectContent>
   405→              <SelectItem value="all">Toutes les catégories</SelectItem>
   406→              {CATEGORIES.map((cat) => (
   407→                <SelectItem key={cat} value={cat}>
   408→                  {cat}
   409→                </SelectItem>
   410→              ))}
   411→            </SelectContent>
   412→          </Select>
   413→          <div className="flex items-center gap-2">
   414→            <Label htmlFor="verified-filter" className="text-sm whitespace-nowrap">
   415→              Vérifiés uniquement
   416→            </Label>
   417→            <Switch
   418→              id="verified-filter"
   419→              checked={verifiedFilter === true}
   420→              onCheckedChange={(checked) => setVerifiedFilter(checked ? true : null)}
   421→            />
   422→          </div>
   423→        </div>
   424→
   425→        {/* ── Loading ───────────────────────────────────────────── */}
   426→        {loading && <TableSkeleton />}
   427→
   428→        {/* ── Error ─────────────────────────────────────────────── */}
   429→        {error && !loading && (
   430→          <div className="flex flex-col items-center justify-center py-12 text-center">
   431→            <p className="text-sm text-destructive">{error}</p>
   432→            <Button variant="outline" size="sm" className="mt-3" onClick={fetchProfessionals}>
   433→              Réessayer
   434→            </Button>
   435→          </div>
   436→        )}
   437→
   438→        {/* ── Empty ─────────────────────────────────────────────── */}
   439→        {!loading && !error && professionals.length === 0 && (
   440→          <div className="flex flex-col items-center justify-center py-12 text-center">
   441→            <Users className="mb-4 h-12 w-12 text-muted-foreground" />
   442→            <h3 className="text-lg font-medium">Aucun professionnel</h3>
   443→            <p className="mt-1 text-sm text-muted-foreground">
   444→              Aucun professionnel ne correspond à vos filtres.
   445→            </p>
   446→          </div>
   447→        )}
   448→
   449→        {/* ── Table ─────────────────────────────────────────────── */}
   450→        {!loading && !error && professionals.length > 0 && (
   451→          <div className="overflow-x-auto">
   452→            <Table>
   453→              <TableHeader>
   454→                <TableRow>
   455→                  <TableHead>Entreprise</TableHead>
   456→                  <TableHead>Catégorie</TableHead>
   457→                  <TableHead className="hidden md:table-cell">Lieu</TableHead>
   458→                  <TableHead className="text-right">Taux horaire</TableHead>
   459→                  <TableHead className="hidden sm:table-cell">Note</TableHead>
   460→                  <TableHead className="text-center">Vérifié</TableHead>
   461→                  <TableHead className="text-center">Actif</TableHead>
   462→                  <TableHead className="hidden lg:table-cell text-center">Travaux</TableHead>
   463→                  <TableHead className="text-right">Actions</TableHead>
   464→                </TableRow>
   465→              </TableHeader>
   466→              <TableBody>

   468→                  <TableRow key={pro.id}>
   469→                    <TableCell className="font-medium">{pro.businessName}</TableCell>
   470→                    <TableCell>
   471→                      <Badge variant="outline" className="text-xs">
   472→                        {pro.category}
   473→                      </Badge>
   474→                    </TableCell>
   475→                    <TableCell className="hidden md:table-cell text-muted-foreground">
   476→                      {pro.location || '—'}
   477→                    </TableCell>
   478→                    <TableCell className="text-right">
   479→                      {pro.hourlyRate != null ? `${pro.hourlyRate}€` : '—'}
   480→                    </TableCell>
   481→                    <TableCell className="hidden sm:table-cell">
   482→                      <Stars rating={pro.ratingAvg} />
   483→                    </TableCell>
   484→                    <TableCell className="text-center">

   490→                    </TableCell>
   491→                    <TableCell className="text-center">
   492→                      <Switch
   493→                        checked={pro.isActive}
   494→                        disabled={togglingId === pro.id && togglingField === 'isActive'}
   495→                        onCheckedChange={() => handleToggle(pro.id, 'isActive', pro.isActive)}
   496→                    />
   497→                    </TableCell>
   498→                    <TableCell className="hidden lg:table-cell text-center">
   499→                      {pro.totalJobsCompleted}
   500→                    </TableCell>
   501→                    <TableCell className="text-right">
   502→                      <div className="flex items-center justify-end gap-1">
   503→                        <Button
   504→                          variant="ghost"
   505→                          size="icon"
   506→                          className="h-8 w-8"
   507→                          onClick={() => {
   508→                            setDetailPro(pro);
   509→                            setDetailOpen(true);
   510→                          }}
   511→                          title="Voir les détails"
   512→                        >
   513→                          <Eye className="h-4 w-4" />
   514→                        </Button>
   515→                        <Button
   516→                          variant="ghost"
   517→                          size="icon"
   518→                          className="h-8 w-8"
   519→                          onClick={() => openEditForm(pro)}
   520→                          title="Modifier"
   521→                        >
   522→                          <Pencil className="h-4 w-4" />
   523→                        </Button>
   524→                        <Button
   525→                          variant="ghost"
   526→                          size="icon"
   527→                          className="h-8 w-8 text-destructive hover:text-destructive"
   528→                          onClick={() => {
   529→                            setDeleteId(pro.id);
   530→                            setDeleteOpen(true);
   531→                          }}
   532→                          title="Désactiver"
   533→                        >
   534→                          <Trash2 className="h-4 w-4" />
   535→                        </Button>
   536→                      </div>
   537→                    </TableCell>
   538→                  </TableRow>
   539→                ))}
   540→              </TableBody>
   541→            </Table>
   542→          </div>
   543→        )}
   544→      </CardContent>
   545→
   546→      {/* ── Create / Edit Dialog ────────────────────────────────── */}
   547→      <Dialog open={formOpen} onOpenChange={(open) => {
   548→        setFormOpen(open);
   549→        if (!open) {
   550→          setFormError(null);
   551→          setEditingId(null);
   552→        }
   553→      }}>
   554→        <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-lg">
   555→          <DialogHeader>
   556→            <DialogTitle>
   557→              {editingId ? 'Modifier le professionnel' : 'Nouveau professionnel'}
   558→            </DialogTitle>
   559→            <DialogDescription>
   560→              {editingId
   561→                ? 'Modifiez les informations du professionnel'
   562→                : 'Ajoutez un nouveau professionnel à la plateforme'}
   563→            </DialogDescription>
   564→          </DialogHeader>
   565→          <div className="grid gap-4 py-2">
   566→            {formError && (
   567→              <p className="text-sm text-destructive">{formError}</p>
   568→            )}
   569→            <div className="grid gap-2">
   570→              <Label htmlFor="form-businessName">Nom de l\'entreprise *</Label>
   571→              <Input
   572→                id="form-businessName"
   573→                placeholder="Martin Plomberie"
   574→                value={form.businessName}
   575→                onChange={(e) => setForm((f) => ({ ...f, businessName: e.target.value }))}
   576→              />
   577→            </div>
   578→            <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   579→              <div className="grid gap-2">
   580→                <Label htmlFor="form-category">Catégorie *</Label>
