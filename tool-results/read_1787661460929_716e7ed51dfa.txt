   409→  // ---- Batch validation ----
   410→  async function handleBatchValidate() {
   411→    const raw = batchCodes.trim();
   412→    if (!raw) return;
   413→    const codes = raw
   414→      .split(/[\n,]/)
   415→      .map((c) => c.trim().toUpperCase().replace(/[^A-Z0-9]/g, ''))
   416→      .filter((c) => c.length > 0);
   417→
   418→    setBatchValidating(true);
   419→    setBatchValidCodes([]);
   420→    setBatchInvalidCodes([]);
   421→
   422→    const valid: string[] = [];
   423→    const invalid: string[] = [];
   424→
   425→    // Check each code sequentially
   426→    for (const code of codes) {
   427→      try {
   428→        const res = await fetch(`/api/client/check-code?code=${encodeURIComponent(code)}`);
   429→        const data: CheckCodeResponse = await res.json();
   430→        if (data.valid) {
   431→          valid.push(code);
   432→        } else {
   433→          invalid.push(code);
   434→        }
   435→      } catch {
   436→        invalid.push(code);
   437→      }
   438→    }
   439→
   440→    setBatchValidCodes(valid);
   441→    setBatchInvalidCodes(invalid);
   442→    setBatchValidated(true);
   443→    setBatchValidating(false);
   444→  }
