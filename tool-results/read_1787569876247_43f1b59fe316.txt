   710→-- Vérifie si l'utilisateur courant est superadmin
   711→CREATE OR REPLACE FUNCTION is_superadmin()
   712→RETURNS BOOLEAN AS $$
   713→BEGIN
   714→    RETURN EXISTS (
   715→        SELECT 1 FROM public.users 
   716→        WHERE id = auth.uid() AND role = 'superadmin'
   717→    );
   718→END;
   719→$$ LANGUAGE plpgsql SECURITY DEFINER STABLE;
   720→
   721→-- Retourne les IDs des homes dont l'utilisateur est membre
   722→CREATE OR REPLACE FUNCTION user_home_ids()
   723→RETURNS UUID[] AS $$
   724→BEGIN
   725→    RETURN ARRAY(
   726→        SELECT home_id FROM public.home_members 
   727→        WHERE user_id = auth.uid()
   728→    );
   729→END;
   730→$$ LANGUAGE plpgsql SECURITY DEFINER STABLE;
   731→
   732→-- Vérifie si l'utilisateur est membre d'un home donné
   733→CREATE OR REPLACE FUNCTION is_home_member(p_home_id UUID)
   734→RETURNS BOOLEAN AS $$
   735→BEGIN
   736→    RETURN EXISTS (
   737→        SELECT 1 FROM public.home_members 
   738→        WHERE home_id = p_home_id AND user_id = auth.uid()
   739→    );
   740→END;
   741→$$ LANGUAGE plpgsql SECURITY DEFINER STABLE;
   742→
   743→-- Vérifie si l'utilisateur est owner/admin d'un home donné
   744→CREATE OR REPLACE FUNCTION is_home_admin(p_home_id UUID)
   745→RETURNS BOOLEAN AS $$
   746→BEGIN
   747→    RETURN EXISTS (
   748→        SELECT 1 FROM public.home_members 
   749→        WHERE home_id = p_home_id AND user_id = auth.uid() 
   750→        AND role IN ('owner', 'admin')
   751→    );
   752→END;
   753→$$ LANGUAGE plpgsql SECURITY DEFINER STABLE;

  1264→-- Met à jour rating_avg et total_reviews sur les merchants quand un avis est ajouté
  1265→CREATE OR REPLACE FUNCTION update_merchant_rating()
  1266→RETURNS TRIGGER AS $$
  1267→BEGIN
  1268→    IF TG_OP = 'INSERT' THEN
  1269→        UPDATE public.merchants 
  1270→        SET rating_avg = (
  1271→            SELECT AVG(r.rating) FROM public.reviews r WHERE r.professional_id = NEW.professional_id
  1272→        ),
  1273→        total_reviews = (
  1274→            SELECT COUNT(*) FROM public.reviews r WHERE r.professional_id = NEW.professional_id
  1275→        )
  1276→        WHERE id = NEW.professional_id;
  1277→        RETURN NEW;
  1278→    ELSIF TG_OP = 'UPDATE' THEN
  1279→        UPDATE public.merchants 
  1280→        SET rating_avg = (
  1281→            SELECT AVG(r.rating) FROM public.reviews r WHERE r.professional_id = NEW.professional_id
  1282→        ),
  1283→        total_reviews = (
  1284→            SELECT COUNT(*) FROM public.reviews r WHERE r.professional_id = NEW.professional_id
  1285→        )
  1286→        WHERE id = NEW.professional_id;
  1287→        RETURN NEW;
  1288→    ELSIF TG_OP = 'DELETE' THEN
  1289→        UPDATE public.merchants 
  1290→        SET rating_avg = (
  1291→            SELECT AVG(r.rating) FROM public.reviews r WHERE r.professional_id = OLD.professional_id
  1292→        ),
  1293→        total_reviews = (
  1294→            SELECT COUNT(*) FROM public.reviews r WHERE r.professional_id = OLD.professional_id
  1295→        )
  1296→        WHERE id = OLD.professional_id;
  1297→        RETURN OLD;
  1298→    END IF;
  1299→    RETURN NULL;
  1300→END;
