    37→/* ─── Reusable fade-in-up wrapper ─── */
    38→function FadeIn({
    39→  children,
    40→  delay = 0,
    41→  className = '',
    42→}: {
    43→  children: React.ReactNode;
    44→  delay?: number;
    45→  className?: string;
    46→}) {
    47→  return (
    48→    <motion.div
    49→      initial={{ opacity: 0, y: 30 }}
    50→      whileInView={{ opacity: 1, y: 0 }}
    51→      viewport={{ once: true, margin: '-50px' }}
    52→      transition={{ duration: 0.6, delay }}
    53→      className={className}
    54→    >
    55→      {children}
    56→    </motion.div>
    57→  );
    58→}
    59→
    60→/* ─── Section wrapper ─── */
    61→function Section({
    62→  children,
    63→  className = '',
    64→}: {
    65→  children: React.ReactNode;
    66→  className?: string;
    67→}) {
    68→  const ref = useRef(null);
    69→  const isInView = useInView(ref, { once: true, margin: '-50px' });
    70→
    71→  return (
    72→    <motion.section
    73→      ref={ref}
    74→      initial={{ opacity: 0, y: 30 }}
    75→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
    76→      transition={{ duration: 0.6 }}
    77→      className={`py-20 md:py-28 px-4 md:px-8 ${className}`}
    78→    >
    79→      <div className="max-w-7xl mx-auto">{children}</div>
    80→    </motion.section>
    81→  );
    82→}
    83→
    84→/* ─── Section Title ─── */
    85→function SectionTitle({
    86→  title,
    87→  subtitle,
    88→}: {
    89→  title: string;
    90→  subtitle?: string;
    91→}) {
    92→  return (
    93→    <div className="text-center mb-16">
    94→      <FadeIn>
    95→        <h2 className="text-3xl md:text-4xl lg:text-5xl font-bold mb-4">

    97→        </h2>
    98→      </FadeIn>
    99→      {subtitle && (
   100→        <FadeIn delay={0.1}>
   101→          <p className="text-gray-400 text-lg md:text-xl">{subtitle}</p>
   102→        </FadeIn>
   103→      )}
   104→    </div>
   105→  );
   106→}
