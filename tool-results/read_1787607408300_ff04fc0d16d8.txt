   133→
   134→export function GenerateBatch() {
   135→  /* ---- state ---- */
   136→  const [quantity, setQuantity] = useState(10);
   137→  const [design, setDesign] = useState<DesignConfig>(DEFAULT_DESIGN);
   138→  const [batchName, setBatchName] = useState('');
   139→  const [activeTemplate, setActiveTemplate] = useState<string | null>(null);
   140→  const [isGenerating, setIsGenerating] = useState(false);
   141→  const [isExporting, setIsExporting] = useState(false);
   142→  const [generatedCodes, setGeneratedCodes] = useState<string[]>([]);
   143→  const [batchId, setBatchId] = useState<string | null>(null);
   144→
   145→  /* ---- refs ---- */
   146→  const previewRef = useRef<HTMLDivElement>(null);
   147→  const qrInstanceRef = useRef<any>(null);
   148→
   149→  /* ---- QR preview ---- */
   150→  const updateQrPreview = useCallback(async () => {
   151→    if (!previewRef.current) return;
   152→    try {
   153→      const QRCodeStyling = (await import('qr-code-styling')).default;
   154→
   155→      const opts: any = {
   156→        width: 280,
   157→        height: 280,
   158→        type: 'canvas',
   159→        data: 'https://qrdomotik.com/activate/QR-XXXXXXXX',
   160→        dotsOptions: {
   161→          color: design.dotsColor,
   162→          type: design.dotsType,
   163→        },
   164→        backgroundOptions: {
   165→          color: design.backgroundColor,
   166→        },
   167→        cornersSquareOptions: {
   168→          color: design.dotsColor,
   169→          type: design.cornersSquareType,
   170→        },
   171→        cornersDotOptions: {
   172→          color: design.dotsColor,
   173→          type: design.cornersDotType,
   174→        },
   175→        qrOptions: {
   176→          errorCorrectionLevel: design.errorCorrectionLevel,
   177→        },
   178→        imageOptions: {
   179→          crossOrigin: 'anonymous',
   180→          margin: 8,
   181→          imageSize: 0.35,
   182→        },
   183→      };
   184→
   185→      if (design.logoPreset) {
   186→        opts.image = getPresetLogoDataUrl(design.logoPreset);
   187→      }
   188→
   189→      if (qrInstanceRef.current) {
   190→        qrInstanceRef.current.update(opts);
   191→      } else {
   192→        previewRef.current.innerHTML = '';
   193→        const instance = new QRCodeStyling(opts);
   194→        instance.append(previewRef.current);
   195→        qrInstanceRef.current = instance;
   196→      }
   197→    } catch (err) {
   198→      console.error('QR preview error:', err);
   199→    }
   200→  }, [design]);
   201→
   202→  useEffect(() => {
   203→    const timer = setTimeout(updateQrPreview, 100);
   204→    return () => clearTimeout(timer);
   205→  }, [updateQrPreview]);
   206→
   207→  /* ---- template ---- */
   208→  const applyTemplate = (key: string) => {
   209→    const tpl = BATCH_TEMPLATES[key];
   210→    if (!tpl) return;
   211→    setQuantity(tpl.quantity);
   212→    setDesign((prev) => ({ ...prev, ...tpl.design }));
   213→    setActiveTemplate(key);
   214→    qrInstanceRef.current = null;
   215→    toast.success(`Template « ${tpl.label} » appliqué`);
   216→  };
   217→
   218→  const resetTemplate = () => {
   219→    setActiveTemplate(null);
   220→    setQuantity(10);
   221→    setDesign(DEFAULT_DESIGN);
   222→    qrInstanceRef.current = null;
   223→  };
   224→
   225→  /* ---- generate batch ---- */
   226→  const handleGenerate = async () => {
   227→    setIsGenerating(true);
   228→    try {
   229→      const codes = generateUniqueCodes(quantity);
   230→      const designConfig = JSON.stringify(design);
   231→
   232→      const res = await fetch('/api/admin/batches', {
   233→        method: 'POST',
   234→        headers: { 'Content-Type': 'application/json' },
   235→        body: JSON.stringify({
   236→          quantity,
   237→          designConfig,
   238→          batchName: batchName || undefined,
   239→          activationCodes: codes,
   240→        }),
   241→      });
   242→
   243→      if (!res.ok) {
   244→        const err = await res.json().catch(() => ({}));
   245→        throw new Error(err.error || 'Erreur lors de la génération');
   246→      }
   247→
   248→      const data = await res.json();
   249→      setGeneratedCodes(codes);
   250→      setBatchId(data.batch?.id || null);
   251→      toast.success(`${quantity} QR codes générés avec succès !`);
   252→    } catch (err: any) {
   253→      toast.error(err.message || 'Erreur lors de la génération');
   254→    } finally {
   255→      setIsGenerating(false);
   256→    }
   257→  };
   258→
   259→  /* ---- export PDF ---- */
   260→  const handleExportPdf = async () => {
   261→    if (generatedCodes.length === 0) return;
   262→    setIsExporting(true);
   263→    try {
   264→      const QRCodeStyling = (await import('qr-code-styling')).default;
   265→      const qrCodes: QrCodeForPdf[] = [];
   266→
   267→      for (const code of generatedCodes) {
   268→        const opts: any = {
   269→          width: 400,
   270→          height: 400,
   271→          type: 'canvas',
   272→          data: `https://qrdomotik.com/activate/${code}`,
   273→          dotsOptions: { color: design.dotsColor, type: design.dotsType },
   274→          backgroundOptions: { color: design.backgroundColor },
   275→          cornersSquareOptions: { color: design.dotsColor, type: design.cornersSquareType },
   276→          cornersDotOptions: { color: design.dotsColor, type: design.cornersDotType },
   277→          qrOptions: { errorCorrectionLevel: design.errorCorrectionLevel },
   278→          imageOptions: { crossOrigin: 'anonymous', margin: 8, imageSize: 0.35 },
   279→        };
   280→        if (design.logoPreset) {
   281→          opts.image = getPresetLogoDataUrl(design.logoPreset);
   282→        }
   283→        const qr = new QRCodeStyling(opts);
   284→        const blob: Blob = await new Promise((resolve, reject) => {
   285→          qr.download({ name: 'qr', extension: 'png' as any });
   286→          qr.getRawData('png').then((d: any) => {
   287→            if (d) resolve(d);
   288→            else reject(new Error('No data'));
   289→          }).catch(reject);
   290→        });
   291→        const imageUrl = URL.createObjectURL(blob);
   292→        qrCodes.push({ code, imageUrl });
   293→      }
   294→
   295→      await downloadPdf({
   296→        qrCodes,
   297→        format: 'a4',
   298→        batchName: batchName || `Lot ${batchId?.slice(0, 8)}`,
   299→      });
   300→
   301→      // Cleanup object URLs
   302→      qrCodes.forEach((qr) => URL.revokeObjectURL(qr.imageUrl));
   303→      toast.success('PDF téléchargé !');
   304→    } catch (err: any) {
   305→      toast.error('Erreur lors de l\'export PDF');
   306→      console.error(err);
   307→    } finally {
   308→      setIsExporting(false);
   309→    }
   310→  };
   311→
   312→  /* ---- design updaters ---- */
   313→  const updateDesign = (partial: Partial<DesignConfig>) => {
   314→    setDesign((prev) => ({ ...prev, ...partial }));
   315→    qrInstanceRef.current = null;
   316→  };
   317→
   318→  const presetIcon = (icon: string) => {
   319→    switch (icon) {
   320→      case 'wifi': return <Wifi className="h-4 w-4" />;
   321→      case 'home': return <Home className="h-4 w-4" />;
   322→      case 'list': return <List className="h-4 w-4" />;
   323→      case 'bell': return <Bell className="h-4 w-4" />;
   324→      case 'shield': return <ShieldCheck className="h-4 w-4" />;
   325→      default: return <ImageIcon className="h-4 w-4" />;
   326→    }
   327→  };
   328→
   329→  /* ================================================================ */
   330→  /*  RENDER                                                            */
   331→  /* ================================================================ */
   332→  return (
   333→    <div className="space-y-6">
   334→      {/* Header */}
   335→      <div>
   336→        <h2 className="text-2xl font-bold tracking-tight">Générer un lot de QR codes</h2>
   337→        <p className="text-muted-foreground">
   338→          Créez un lot de QR codes physiques avec un design personnalisé, puis exportez-les en PDF pour l'impression.
   339→        </p>
   340→      </div>
   341→
   342→      <div className="grid grid-cols-1 lg:grid-cols-5 gap-6">
   343→        {/* ====== LEFT: FORM ====== */}
   344→        <div className="lg:col-span-3 space-y-6">
   345→          {/* Templates */}
   346→          <Card>
   347→            <CardHeader className="pb-3">
   348→              <CardTitle className="flex items-center gap-2 text-base">
   349→                <Sparkles className="h-4 w-4 text-amber-500" />
   350→                Templates prêts à l'emploi
   351→              </CardTitle>
   352→              <CardDescription>Choisissez un template pour pré-remplir la configuration</CardDescription>
   353→            </CardHeader>
   354→            <CardContent>
   355→              <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   356→                {Object.entries(BATCH_TEMPLATES).map(([key, tpl]) => (
   357→                  <button
   358→                    key={key}
   359→                    onClick={() => applyTemplate(key)}
   360→                    className={`relative flex flex-col items-start gap-1 rounded-lg border-2 p-4 text-left transition-all hover:shadow-md ${
   361→                      activeTemplate === key
   362→                        ? 'border-primary bg-primary/5'
   363→                        : 'border-border hover:border-primary/40'
   364→                    }`}
   365→                  >
   366→                    {activeTemplate === key && (
   367→                      <div className="absolute top-2 right-2">
   368→                        <Check className="h-4 w-4 text-primary" />
   369→                      </div>
   370→                    )}
   371→                    <span className="font-semibold text-sm">{tpl.label}</span>
   372→                    <span className="text-xs text-muted-foreground">{tpl.description}</span>
   373→                    <Badge variant="secondary" className="mt-1 w-fit text-xs">
   374→                      {tpl.quantity} codes
   375→                    </Badge>
   376→                  </button>
   377→                ))}
   378→              </div>
   379→              {activeTemplate && (
   380→                <Button variant="ghost" size="sm" className="mt-3" onClick={resetTemplate}>
   381→                  <RefreshCw className="mr-1 h-3 w-3" />
   382→                  Réinitialiser
   383→                </Button>
   384→              )}
   385→            </CardContent>
   386→          </Card>
   387→
   388→          {/* Configuration Form */}
   389→          <Card>
   390→            <CardHeader className="pb-3">
   391→              <CardTitle className="flex items-center gap-2 text-base">
   392→                <Layout className="h-4 w-4" />
   393→                Configuration du lot
   394→              </CardTitle>
   395→            </CardHeader>
   396→            <CardContent className="space-y-5">
   397→              <div className="grid grid-cols-2 gap-4">
   398→                <div className="space-y-2">
   399→                  <Label htmlFor="quantity">Quantité</Label>
   400→                  <Select value={String(quantity)} onValueChange={(v) => setQuantity(Number(v))}>
   401→                    <SelectTrigger id="quantity">
   402→                      <SelectValue />
   403→                    </SelectTrigger>
   404→                    <SelectContent>
   405→                      <SelectItem value="10">10 QR codes</SelectItem>
   406→                      <SelectItem value="15">15 QR codes</SelectItem>
   407→                      <SelectItem value="20">20 QR codes</SelectItem>
   408→                    </SelectContent>
   409→                  </Select>
   410→                </div>
   411→                <div className="space-y-2">
   412→                  <Label htmlFor="batchName">Nom du lot (optionnel)</Label>
   413→                  <Input
   414→                    id="batchName"
   415→                    placeholder="ex: Pack Dakar Jan 2026"
   416→                    value={batchName}
   417→                    onChange={(e) => setBatchName(e.target.value)}
   418→                  />
   419→                </div>
   420→              </div>
   421→            </CardContent>
   422→          </Card>
   423→
   424→          {/* Design Form */}
   425→          <Card>
   426→            <CardHeader className="pb-3">
   427→              <CardTitle className="flex items-center gap-2 text-base">
   428→                <Palette className="h-4 w-4" />
   429→                Design du QR Code
   430→              </CardTitle>
   431→              <CardDescription>Personnalisez l'apparence de vos QR codes</CardDescription>
   432→            </CardHeader>
   433→            <CardContent>
   434→              <Tabs defaultValue="style" className="w-full">
   435→                <TabsList className="grid w-full grid-cols-2">
   436→                  <TabsTrigger value="style">Style & Couleurs</TabsTrigger>
   437→                  <TabsTrigger value="logo">Logo central</TabsTrigger>
   438→                </TabsList>
   439→
   440→                <TabsContent value="style" className="space-y-5 mt-4">
   441→                  {/* Colors */}
   442→                  <div className="grid grid-cols-2 gap-4">
   443→                    <div className="space-y-2">
   444→                      <Label>Couleur des points</Label>
   445→                      <div className="flex items-center gap-3">
   446→                        <input
   447→                          type="color"
   448→                          value={design.dotsColor}
   449→                          onChange={(e) => updateDesign({ dotsColor: e.target.value })}
   450→                          className="h-10 w-14 cursor-pointer rounded-md border border-input bg-transparent"
