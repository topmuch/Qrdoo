   116→/*  SVG → PNG conversion utility (browser-only, zero native deps)       */
   117→/* ------------------------------------------------------------------ */
   118→
   119→/** Convert an inline SVG element to a PNG data-URL */
   120→function svgElementToPngDataUrl(svgEl: SVGSVGElement, size: number): Promise<string> {
   121→  return new Promise((resolve, reject) => {
   122→    const svgData = new XMLSerializer().serializeToString(svgEl);
   123→    // Use data URL instead of blob — more reliable for Image loading
   124→    const svgUrl = `data:image/svg+xml;charset=utf-8,${encodeURIComponent(svgData)}`;
   125→    const img = new Image();
   126→    img.onload = () => {
   127→      const canvas = document.createElement('canvas');
   128→      canvas.width = size;
   129→      canvas.height = size;
   130→      const ctx = canvas.getContext('2d');
   131→      if (!ctx) { reject(new Error('No 2d context')); return; }
   132→      ctx.drawImage(img, 0, 0, size, size);
   133→      resolve(canvas.toDataURL('image/png'));
   134→    };
   135→    img.onerror = () => reject(new Error('SVG image load failed'));
   136→    img.src = svgUrl;
   137→  });
   138→}

   140→/**
   141→ * Fallback QR PNG generator using qrcode.react (pure JS, no native deps).
   142→ * Renders QRCodeSVG into a hidden container, extracts the SVG, converts to PNG.
   143→ */
   144→async function generateFallbackQrPng(
   145→  data: string,
   146→  size: number,
   147→  fgColor: string,
   148→  bgColor: string,
   149→  level: string,
   150→): Promise<string> {
   151→  const container = document.createElement('div');
   152→  container.style.cssText = 'position:fixed;left:-9999px;top:-9999px;';
   153→  document.body.appendChild(container);
   154→
   155→  const root = createRoot(container);
   156→  root.render(
   157→    createElement(QRCodeSVG, {
   158→      value: data,
   159→      size,
   160→      bgColor,
   161→      fgColor,
   162→      level: level as 'L' | 'M' | 'Q' | 'H',
   163→    }),
   164→  );
   165→
   166→  // Wait for React to commit the render
   167→  await new Promise((r) => setTimeout(r, 150));
   168→
   169→  const svgEl = container.querySelector('svg');
   170→  if (!svgEl) {
   171→    root.unmount();
   172→    document.body.removeChild(container);
   173→    throw new Error('Fallback: SVG element not rendered');
   174→  }
   175→
   176→  // Ensure the SVG has xmlns for proper PNG conversion
   177→  if (!svgEl.getAttribute('xmlns')) {
   178→    svgEl.setAttribute('xmlns', 'http://www.w3.org/2000/svg');
   179→  }
   180→
   181→  const pngUrl = await svgElementToPngDataUrl(svgEl, size);
   182→  root.unmount();
   183→  document.body.removeChild(container);
   184→  return pngUrl;
   185→}

   208→export function GenerateBatch() {
   209→  /* ---- state ---- */
   210→  const [quantity, setQuantity] = useState(10);
   211→  const [design, setDesign] = useState<DesignConfig>(DEFAULT_DESIGN);
   212→  const [batchName, setBatchName] = useState('');
   213→  const [activeTemplate, setActiveTemplate] = useState<string | null>(null);
   214→  const [isGenerating, setIsGenerating] = useState(false);
   215→  const [isExporting, setIsExporting] = useState(false);
   216→  const [generatedCodes, setGeneratedCodes] = useState<string[]>([]);
   217→  const [batchId, setBatchId] = useState<string | null>(null);
   218→
   219→  /* ---- refs ---- */
   220→  const previewRef = useRef<HTMLDivElement>(null);
   221→  const qrInstanceRef = useRef<any>(null);
   222→  const fallbackRef = useRef(false);
   223→
   224→  /* ---- state ---- */
   225→  const [showFallback, setShowFallback] = useState(false);
   226→
   227→  /* ---- QR preview ---- */
   228→  const updateQrPreview = useCallback(async () => {
   229→    if (fallbackRef.current || !previewRef.current) return;
   230→    try {
   231→      const QRCodeStyling = (await import('qr-code-styling')).default;
   232→
   233→      const opts: any = {
   234→        width: 280,
   235→        height: 280,
   236→        type: 'svg',
   237→        data: 'https://qrdomotik.com/activate/QR-XXXXXXXX',
   238→        dotsOptions: {
   239→          color: design.dotsColor,
   240→          type: design.dotsType,
   241→        },
   242→        backgroundOptions: {
   243→          color: design.backgroundColor,
   244→        },
   245→        cornersSquareOptions: {
   246→          color: design.dotsColor,
   247→          type: design.cornersSquareType,
   248→        },
   249→        cornersDotOptions: {
   250→          color: design.dotsColor,
   251→          type: design.cornersDotType,
   252→        },
   253→        qrOptions: {
   254→          errorCorrectionLevel: design.errorCorrectionLevel,
   255→        },
   256→        imageOptions: {
   257→          crossOrigin: 'anonymous',
   258→          margin: 8,
   259→          imageSize: 0.35,
   260→        },
   261→      };
   262→
   263→      if (design.logoPreset) {
   264→        opts.image = getPresetLogoDataUrl(design.logoPreset);
   265→      }
   266→
   267→      if (qrInstanceRef.current) {
   268→        qrInstanceRef.current.update(opts);
   269→      } else {
   270→        previewRef.current.innerHTML = '';
   271→        const instance = new QRCodeStyling(opts);
   272→        instance.append(previewRef.current);
   273→        qrInstanceRef.current = instance;
   274→      }
   275→    } catch (err) {
   276→      console.error('QR preview error — switching to SVG fallback:', err);
   277→      fallbackRef.current = true;
   278→      setShowFallback(true);
   279→      if (previewRef.current) previewRef.current.innerHTML = '';
   280→      qrInstanceRef.current = null;
   281→    }
   282→  }, [design]);
   283→
   284→  useEffect(() => {
   285→    const timer = setTimeout(updateQrPreview, 100);
   286→    return () => clearTimeout(timer);
   287→  }, [updateQrPreview]);
   288→
   289→  /* ---- template ---- */
   290→  const applyTemplate = (key: string) => {
   291→    const tpl = BATCH_TEMPLATES[key];
   292→    if (!tpl) return;
   293→    setQuantity(tpl.quantity);
   294→    setDesign((prev) => ({ ...prev, ...tpl.design }));
   295→    setActiveTemplate(key);
   296→    qrInstanceRef.current = null;
   297→    toast.success(`Template « ${tpl.label} » appliqué`);
   298→  };
   299→
   300→  const resetTemplate = () => {
   301→    setActiveTemplate(null);
   302→    setQuantity(10);
   303→    setDesign(DEFAULT_DESIGN);
   304→    qrInstanceRef.current = null;
   305→  };
   306→
   307→  /* ---- generate batch ---- */
   308→  const handleGenerate = async () => {
   309→    setIsGenerating(true);
   310→    try {
   311→      const codes = generateUniqueCodes(quantity);
   312→      const designConfig = JSON.stringify(design);
   313→
   314→      const res = await fetch('/api/admin/batches', {
   315→        method: 'POST',
   316→        headers: { 'Content-Type': 'application/json' },
   317→        body: JSON.stringify({
   318→          quantity,
   319→          designConfig,
   320→          batchName: batchName || undefined,
   321→          activationCodes: codes,
   322→        }),
   323→      });
   324→
   325→      if (!res.ok) {
   326→        const err = await res.json().catch(() => ({}));
   327→        throw new Error(err.error || 'Erreur lors de la génération');
   328→      }
   329→
   330→      const data = await res.json();
   331→      setGeneratedCodes(codes);
   332→      setBatchId(data.id || null);
   333→      toast.success(`${quantity} QR codes générés avec succès !`);
   334→    } catch (err: any) {
   335→      toast.error(err.message || 'Erreur lors de la génération');
   336→    } finally {
   337→      setIsGenerating(false);
   338→    }
   339→  };
   340→
   341→  /* ---- export PDF ---- */
   342→  const handleExportPdf = async () => {
   343→    if (generatedCodes.length === 0) return;
   344→    setIsExporting(true);
   345→    try {
   346→      const qrCodes: QrCodeForPdf[] = [];
   347→
   348→      for (const code of generatedCodes) {
   349→        const data = `https://qrdomotik.com/activate/${code}`;
   350→        const imageUrl = await generateFallbackQrPng(
   351→          data, 400, design.dotsColor, design.backgroundColor, design.errorCorrectionLevel,
   352→        );
   353→        qrCodes.push({ code, imageUrl });
   354→      }
   355→
   356→      await downloadPdf({
   357→        qrCodes,
   358→        format: 'a4',
   359→        batchName: batchName || `Lot ${batchId?.slice(0, 8)}`,
   360→      });
   361→
   362→      toast.success('PDF téléchargé !');
   363→    } catch (err: any) {
   364→      toast.error("Erreur lors de l'export PDF");
   365→      console.error(err);
   366→    } finally {
   367→      setIsExporting(false);
   368→    }
   369→  };
   370→
   371→  /* ---- design updaters ---- */
   372→  const updateDesign = (partial: Partial<DesignConfig>) => {
   373→    setDesign((prev) => ({ ...prev, ...partial }));
   374→    qrInstanceRef.current = null;
   375→  };
   376→
   377→  const presetIcon = (icon: string) => {
   378→    switch (icon) {
   379→      case 'wifi': return <Wifi className="h-4 w-4" />;
   380→      case 'home': return <Home className="h-4 w-4" />;
   381→      case 'list': return <List className="h-4 w-4" />;
   382→      case 'bell': return <Bell className="h-4 w-4" />;
   383→      case 'shield': return <ShieldCheck className="h-4 w-4" />;
   384→      default: return <ImageIcon className="h-4 w-4" />;
   385→    }
   386→  };
   387→
   388→  /* ================================================================ */
   389→  /*  RENDER                                                            */
   390→  /* ================================================================ */
   391→  return (
   392→    <div className="space-y-6">
   393→      {/* Header */}
   394→      <div>
   395→        <h2 className="text-2xl font-bold tracking-tight">Générer un lot de QR codes</h2>
   396→        <p className="text-muted-foreground">
   397→          Créez un lot de QR codes physiques avec un design personnalisé, puis exportez-les en PDF pour l'impression.
   398→        </p>
   399→      </div>
   400→
   401→      <div className="grid grid-cols-1 lg:grid-cols-5 gap-6">
   402→        {/* ====== LEFT: FORM ====== */}
   403→        <div className="lg:col-span-3 space-y-6">
   404→          {/* Templates */}
   405→          <Card>
   406→            <CardHeader className="pb-3">
   407→              <CardTitle className="flex items-center gap-2 text-base">
   408→                <Sparkles className="h-4 w-4 text-amber-500" />
   409→                Templates prêts à l'emploi
   410→              </CardTitle>
   411→              <CardDescription>Choisissez un template pour pré-remplir la configuration</CardDescription>
   412→            </CardHeader>
   413→            <CardContent>
   414→              <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   415→                {Object.entries(BATCH_TEMPLATES).map(([key, tpl]) => (
   416→                  <button
   417→                    key={key}
   418→                    onClick={() => applyTemplate(key)}
   419→                    className={`relative flex flex-col items-start gap-1 rounded-lg border-2 p-4 text-left transition-all hover:shadow-md ${
   420→                      activeTemplate === key
   421→                        ? 'border-primary bg-primary/5'
   422→                        : 'border-border hover:border-primary/40'
   423→                    }`}
   424→                  >
   425→                    {activeTemplate === key && (
   426→                      <div className="absolute top-2 right-2">
   427→                        <Check className="h-4 w-4 text-primary" />
   428→                      </div>
   429→                    )}
   430→                    <span className="font-semibold text-sm">{tpl.label}</span>
   431→                    <span className="text-xs text-muted-foreground">{tpl.description}</span>
   432→                    <Badge variant="secondary" className="mt-1 w-fit text-xs">
   433→                      {tpl.quantity} codes
   434→                    </Badge>
   435→                  </button>
   436→                ))}
   437→              </div>
   438→              {activeTemplate && (
   439→                <Button variant="ghost" size="sm" className="mt-3" onClick={resetTemplate}>
   440→                  <RefreshCw className="mr-1 h-3 w-3" />
   441→                  Réinitialiser
   442→                </Button>
   443→              )}
   444→            </CardContent>
   445→          </Card>
   446→
   447→          {/* Configuration Form */}
   448→          <Card>
   449→            <CardHeader className="pb-3">
   450→              <CardTitle className="flex items-center gap-2 text-base">
