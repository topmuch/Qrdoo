   424→   424→    // Check each code sequentially
   425→   425→    for (const code of codes) {

   428→   428→        const data: CheckCodeResponse = await res.json();

   430→   430→          valid.push(code);
   431→   431→        } else {

   437→   437→    }

   439→   439→    setBatchValidCodes(valid);
   440→   440→    setBatchInvalidCodes(invalid);
   441→   441→    setBatchValidated(true);
