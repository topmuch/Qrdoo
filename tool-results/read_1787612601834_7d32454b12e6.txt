    40→/* ─── Reusable fade-in-up wrapper ─── */
    41→function FadeIn({
    42→  children,
    43→  delay = 0,
    44→  className = '',
    45→}: {
    46→  children: React.ReactNode;
    47→  delay?: number;
    48→  className?: string;
    49→}) {
    50→  return (
    51→    <motion.div
    52→      initial={{ opacity: 0, y: 30 }}
    53→      whileInView={{ opacity: 1, y: 0 }}
    54→      viewport={{ once: true, margin: '-50px' }}
    55→      transition={{ duration: 0.6, delay }}
    56→      className={className}
    57→    >
    58→      {children}
    59→    </motion.div>
    60→  );
    61→}

    63→/* ─── Sticky Navbar ─── */
    64→function Navbar({ onGoToDashboard }: { onGoToDashboard: () => void }) {
    65→  const [mobileOpen, setMobileOpen] = useState(false);
    66→  const [scrolled, setScrolled] = useState(false);
    67→
    68→  useEffect(() => {
    69→    const onScroll = () => setScrolled(window.scrollY > 20);
    70→    window.addEventListener('scroll', onScroll, { passive: true });
    71→    return () => window.removeEventListener('scroll', onScroll);
    72→  }, []);
    73→
    74→  const links = [
    75→    { label: 'Accueil', href: '#hero' },
    76→    { label: 'A propos', href: '#avantages' },
    77→    { label: 'Tarifs', href: '#pricing' },
    78→    { label: 'Contactez nous', href: '#cta-final' },
    79→  ];
    80→
    81→  const handleNavClick = (e: React.MouseEvent<HTMLAnchorElement>, href: string) => {
    82→    e.preventDefault();
    83→    setMobileOpen(false);
    84→    const target = document.querySelector(href);
    85→    if (target) {
    86→      target.scrollIntoView({ behavior: 'smooth' });
    87→    }
    88→  };
    89→
    90→  return (
    91→    <nav
    92→      className={`fixed top-0 left-0 right-0 z-50 transition-all duration-300 ${
    93→        scrolled
    94→          ? 'bg-[#0a0f1e]/95 backdrop-blur-md border-b border-white/10 shadow-lg shadow-black/20'
    95→          : 'bg-[#0a0f1e]/80 backdrop-blur-md border-b border-white/10'
    96→      }`}
    97→    >
    98→      <div className="max-w-7xl mx-auto px-4 md:px-8 flex items-center justify-between h-16">
    99→        {/* Logo */}
   100→        <a
   101→          href="#hero"
   102→          onClick={(e) => handleNavClick(e, '#hero')}
   103→          className="flex items-center gap-2 group"
   104→        >
   105→          <QrCode className="w-6 h-6 text-[#2563EB] group-hover:scale-110 transition-transform" />
   106→          <span className="font-bold text-lg tracking-tight">QR Domotik</span>
   107→        </a>
   108→
   109→        {/* Desktop links */}
   110→        <div className="hidden md:flex items-center gap-8">
   111→          {links.map((link) => (
   112→            <a
   113→              key={link.href}
   114→              href={link.href}
   115→              onClick={(e) => handleNavClick(e, link.href)}
   116→              className="text-sm text-gray-300 hover:text-white transition-colors"
   117→            >
   118→              {link.label}
   119→            </a>
   120→          ))}
   121→          <button
   122→            onClick={onGoToDashboard}
   123→            className="ml-2 px-5 py-2 rounded-lg bg-[#2563EB] text-white text-sm font-medium hover:bg-[#1d4ed8] transition-colors"
   124→          >
   125→            Se connecter
   126→          </button>
   127→        </div>
   128→
   129→        {/* Mobile hamburger */}
   130→        <button
   131→          onClick={() => setMobileOpen(!mobileOpen)}
   132→          className="md:hidden p-2 text-gray-300 hover:text-white transition-colors"
   133→          aria-label="Menu"
   134→        >
   135→          {mobileOpen ? <X className="w-6 h-6" /> : <Menu className="w-6 h-6" />}
   136→        </button>
   137→      </div>
   138→
   139→      {/* Mobile menu */}
   140→      <div
   141→        className={`md:hidden overflow-hidden transition-all duration-300 ${
   142→          mobileOpen ? 'max-h-80 opacity-100' : 'max-h-0 opacity-0'
   143→        }`}
   144→      >
   145→        <div className="px-4 pb-4 space-y-1 bg-[#0a0f1e]/95 backdrop-blur-md border-t border-white/5">
   146→          {links.map((link) => (
   147→            <a
   148→              key={link.href}
   149→              href={link.href}
   150→              onClick={(e) => handleNavClick(e, link.href)}
   151→              className="block py-3 px-3 text-gray-300 hover:text-white hover:bg-white/5 rounded-lg transition-colors"
   152→            >
   153→              {link.label}
   154→            </a>
   155→          ))}
   156→          <button
   157→            onClick={() => {
   158→              setMobileOpen(false);
   159→              onGoToDashboard();
   160→            }}
   161→            className="w-full mt-2 px-5 py-3 rounded-lg bg-[#2563EB] text-white text-sm font-medium hover:bg-[#1d4ed8] transition-colors"
   162→          >
   163→            Se connecter
   164→          </button>
   165→        </div>
   166→      </div>
   167→    </nav>
   168→  );
   169→}

   171→/* ─── Section wrapper ─── */
   172→function Section({
   173→  children,
   174→  className = '',
   175→  id,
   176→}: {
   177→  children: React.ReactNode;
   178→  className?: string;
   179→  id?: string;
   180→}) {
   181→  const ref = useRef(null);
   182→  const isInView = useInView(ref, { once: true, margin: '-50px' });
   183→
   184→  return (
   185→    <motion.section
   186→      id={id}
   187→      ref={ref}
   188→      initial={{ opacity: 0, y: 30 }}
   189→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
   190→      transition={{ duration: 0.6 }}
   191→      className={`py-20 md:py-28 px-4 md:px-8 scroll-mt-16 ${className}`}
   192→    >
   193→      <div className="max-w-7xl mx-auto">{children}</div>
   194→    </motion.section>
   195→  );
   196→}
   197→

   219→  );
   220→}
