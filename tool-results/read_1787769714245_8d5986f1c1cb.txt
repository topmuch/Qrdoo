   113→  const [merchants, setMerchants] = useState<MerchantData[]>([]);
   114→  const [promos, setPromos] = useState<PromoData[]>([]);
   115→  const [flashSales, setFlashSales] = useState<FlashSaleData[]>([]);

   117→  const [transactions, setTransactions] = useState<TransactionData[]>([]);
   118→  const [loading, setLoading] = useState(true);
   119→
   120→  const [mSearch, setMSearch] = useState('');
   121→  const [mCategory, setMCategory] = useState('all');
   122→  const [pMerchant, setPMerchant] = useState('all');
   123→  const [pStatus, setPStatus] = useState('all');
   124→  const [cStatus, setCStatus] = useState('all');
   125→  const [cMerchant, setCMerchant] = useState('all');
   126→  const [tType, setTType] = useState('all');
   127→  const [tStatus, setTStatus] = useState('all');
   128→
   129→  const [merchantDlg, setMerchantDlg] = useState(false);
   130→  const [promoDlg, setPromoDlg] = useState(false);
   131→  const [flashDlg, setFlashDlg] = useState(false);
   132→  const [couponDlg, setCouponDlg] = useState(false);
   133→  const [editId, setEditId] = useState<string | null>(null);
   134→
   135→  const [mf, setMf] = useState({ name: '', category: '', address: '', phone: '', description: '', website: '' });
   136→  const [pf, setPf] = useState({ merchantId: '', title: '', description: '', originalPrice: '', promoPrice: '', validUntil: '', category: '' });
   137→  const [ff, setFf] = useState({ merchantId: '', promoId: '', title: '', flashPrice: '', startsAt: '', endsAt: '', maxRedemptions: '', costEuros: '0.5' });
   138→  const [cf, setCf] = useState({ merchantId: '', discountType: 'percentage', discountValue: '', maxUses: '1', validUntil: '' });
   139→
   140→  const [submitting, setSubmitting] = useState(false);
   141→
   142→  const [now, setNow] = useState(Date.now());
   143→  const cdRef = useRef<ReturnType<typeof setInterval> | null>(null);
   144→  useEffect(() => {
   145→    cdRef.current = setInterval(() => setNow(Date.now()), 1000);
   146→    return () => { if (cdRef.current) clearInterval(cdRef.current); };
   147→  }, []);
   148→
   149→  // ── Fetch ────────────────────────────────────────────────────
   150→  const fetchAll = useCallback(async () => {
   151→    try {
   152→      setLoading(true);
   153→      const [m, p, f, c, t] = await Promise.all([
   154→        fetch('/api/client/merchants').then(r => r.json()).then(d => Array.isArray(d) ? d : d.data ?? []),
   155→        fetch('/api/client/promos').then(r => r.json()).then(d => Array.isArray(d) ? d : d.data ?? []),
   156→        fetch('/api/client/flash-sales?status=all').then(r => r.json()).then(d => Array.isArray(d) ? d : d.data ?? []),
   157→        fetch('/api/client/coupons?status=all').then(r => r.json()).then(d => Array.isArray(d) ? d : d.data ?? []),
   158→        fetch('/api/client/transactions?limit=100').then(r => r.json()).then(d => d.transactions ?? d.data ?? []),

   160→      setMerchants(m); setPromos(p); setFlashSales(f); setCoupons(c); setTransactions(t);
   161→    } catch { toast.error('Erreur lors du chargement'); } finally { setLoading(false); }
   162→  }, []);
   163→
   164→  useEffect(() => { fetchAll(); }, [fetchAll]);
   165→
   166→  // ── API helper ───────────────────────────────────────────────
   167→  const api = async (url: string, method: string, body?: unknown) => {
   168→    setSubmitting(true);
   169→    try {
   170→      const res = await fetch(url, { method, headers: { 'Content-Type': 'application/json' }, body: body ? JSON.stringify(body) : undefined });
   171→      if (!res.ok) { const d = await res.json().catch(() => ({})); throw new Error(d.error || 'Erreur'); }
   172→      return true;
   173→    } finally { setSubmitting(false); }
   174→  };
   175→
   176→  const rstM = () => setMf({ name: '', category: '', address: '', phone: '', description: '', website: '' });
   177→  const rstP = () => setPf({ merchantId: '', title: '', description: '', originalPrice: '', promoPrice: '', validUntil: '', category: '' });
   178→  const rstF = () => setFf({ merchantId: '', promoId: '', title: '', flashPrice: '', startsAt: '', endsAt: '', maxRedemptions: '', costEuros: '0.5' });

   180→
   181→  // ── Merchant CRUD ────────────────────────────────────────────
   182→  const saveMerchant = async () => {
   183→    if (!mf.name.trim()) return;
   184→    const body = { name: mf.name.trim(), category: mf.category || undefined, address: mf.address || undefined, phone: mf.phone || undefined, description: mf.description || undefined, website: mf.website || undefined, openingHours: '{}' };
   185→    if (editId) { await api(`/api/client/merchants/${editId}`, 'PATCH', body); toast.success('Commerçant mis à jour'); }
   186→    else { await api('/api/client/merchants', 'POST', body); toast.success('Commerçant créé'); }
   187→    setMerchantDlg(false); setEditId(null); rstM(); fetchAll();
   188→  };
   189→  const deleteMerchant = (id: string) => api(`/api/client/merchants/${id}`, 'DELETE').then(() => { toast.success('Supprimé'); fetchAll(); });
   190→  const toggleMerchant = (m: MerchantData, field: 'isActive' | 'isVerified') =>
   191→    api(`/api/client/merchants/${m.id}`, 'PATCH', { [field]: !m[field] }).then(() => { toast.success('Mis à jour'); fetchAll(); });
   192→  const editMerchant = (m: MerchantData) => {
   193→    setMf({ name: m.name, category: m.category || '', address: m.address || '', phone: m.phone || '', description: m.description || '', website: m.website || '' });
   194→    setEditId(m.id); setMerchantDlg(true);
   195→  };
   196→
   197→  // ── Promo CRUD ───────────────────────────────────────────────
   198→  const savePromo = async () => {
   199→    if (!pf.title.trim()) return;
   200→    const body = { merchantId: pf.merchantId || undefined, title: pf.title.trim(), description: pf.description || undefined, originalPrice: pf.originalPrice ? parseFloat(pf.originalPrice) : undefined, promoPrice: pf.promoPrice ? parseFloat(pf.promoPrice) : undefined, validUntil: pf.validUntil || undefined, category: pf.category || undefined, source: 'local' };
   201→    if (editId) { await api(`/api/client/promos/${editId}`, 'PATCH', body); toast.success('Promotion mise à jour'); }
   202→    else { await api('/api/client/promos', 'POST', body); toast.success('Promotion créée'); }
   203→    setPromoDlg(false); setEditId(null); rstP(); fetchAll();
   204→  };
   205→  const deletePromo = (id: string) => api(`/api/client/promos/${id}`, 'DELETE').then(() => { toast.success('Supprimé'); fetchAll(); });
   206→  const editPromo = (p: PromoData) => {
   207→    setPf({ merchantId: p.merchantId || '', title: p.title, description: p.description || '', originalPrice: p.originalPrice?.toString() || '', promoPrice: p.promoPrice?.toString() || '', validUntil: p.validUntil?.split('T')[0] || '', category: p.category || '' });
   208→    setEditId(p.id); setPromoDlg(true);
   209→  };
   210→
   211→  // ── Flash CRUD ───────────────────────────────────────────────
   212→  const saveFlash = async () => {
   213→    if (!ff.merchantId || !ff.flashPrice || !ff.startsAt || !ff.endsAt) return;
   214→    const body = { merchantId: ff.merchantId, promoId: ff.promoId || undefined, title: ff.title || undefined, flashPrice: parseFloat(ff.flashPrice), startsAt: ff.startsAt, endsAt: ff.endsAt, maxRedemptions: ff.maxRedemptions ? parseInt(ff.maxRedemptions) : undefined, costEuros: parseFloat(ff.costEuros) || 0.5 };
   215→    await api('/api/client/flash-sales', 'POST', body);
   216→    toast.success('Vente flash créée'); setFlashDlg(false); rstF(); fetchAll();
   217→  };
   218→  const flashAction = (id: string, action: 'active' | 'cancelled' | 'delete') => {
   219→    if (action === 'delete') api(`/api/client/flash-sales/${id}`, 'DELETE').then(() => { toast.success('Supprimé'); fetchAll(); });
   220→    else api(`/api/client/flash-sales/${id}`, 'PATCH', { status: action }).then(() => { toast.success(action === 'active' ? 'Activée' : 'Annulée'); fetchAll(); });
   221→  };
   222→
   223→  // ── Coupon CRUD ──────────────────────────────────────────────
   224→  const saveCoupon = async () => {
   225→    if (!cf.merchantId || !cf.discountValue) return;
   226→    const body = { merchantId: cf.merchantId, discountType: cf.discountType, discountValue: parseFloat(cf.discountValue), maxUses: parseInt(cf.maxUses) || 1, validUntil: cf.validUntil || undefined };
   227→    await api('/api/client/coupons', 'POST', body);
   228→    toast.success('Coupon créé'); setCouponDlg(false); rstC(); fetchAll();
   229→  };

   231→

   233→  const filteredMerchants = merchants.filter(m => {
   234→    if (mCategory !== 'all' && m.category !== mCategory) return false;
   235→    if (mSearch && !m.name.toLowerCase().includes(mSearch.toLowerCase()) && !(m.category || '').toLowerCase().includes(mSearch.toLowerCase())) return false;
   236→    return true;
   237→  });

   239→    if (pMerchant !== 'all' && p.merchantId !== pMerchant) return false;

   241→    if (pStatus === 'expired' && isPromoActive(p)) return false;
   242→    return true;
   243→  });

   245→    if (cStatus !== 'all' && c.status !== cStatus) return false;
   246→    if (cMerchant !== 'all' && c.merchantId !== cMerchant) return false;
   247→    return true;
   248→  });

   250→    if (tType !== 'all' && t.type !== tType) return false;
   251→    if (tStatus !== 'all' && t.status !== tStatus) return false;
   252→    return true;

   259→    const h = Math.floor(diff / 3600000), m = Math.floor((diff % 3600000) / 60000), s = Math.floor((diff % 60000) / 1000);
   260→    return `${h}h ${m}m ${s}s`;
   261→  };
   262→
   263→  const PROMO_STATUS_ACTIVE = { label: 'Active', cls: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-300 border-0' };
   264→  const PROMO_STATUS_EXPIRED = { label: 'Expirée', cls: 'bg-slate-100 text-slate-600 dark:bg-slate-800 dark:text-slate-400 border-0' };
   265→
   266→  if (loading) return (
   267→    <div className="space-y-4">
   268→      <Skeleton className="h-8 w-64" />
   269→      <div className="grid grid-cols-3 gap-4">{Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-24" />)}</div>
   270→      <Skeleton className="h-96" />
   271→    </div>
   272→  );
   273→
   274→  return (
   275→    <Card>
   276→      <CardHeader>
   277→        <CardTitle className="flex items-center gap-2"><Store className="h-5 w-5" /> Marketplace</CardTitle>
   278→        <CardDescription>Gestion des commerçants, promotions, ventes flash, coupons et transactions</CardDescription>
   279→      </CardHeader>
   280→      <CardContent>
   281→        <Tabs defaultValue="merchants">
   282→          <TabsList className="mb-4 w-full flex-wrap h-auto gap-1">
   283→            <TabsTrigger value="merchants" className="gap-1.5"><Store className="h-3.5 w-3.5" /><span className="hidden sm:inline">Commerçants</span></TabsTrigger>
   284→            <TabsTrigger value="promos" className="gap-1.5"><Tag className="h-3.5 w-3.5" /><span className="hidden sm:inline">Promotions</span></TabsTrigger>
   285→            <TabsTrigger value="flash" className="gap-1.5"><Zap className="h-3.5 w-3.5" /><span className="hidden sm:inline">Ventes Flash</span></TabsTrigger>
   286→            <TabsTrigger value="coupons" className="gap-1.5"><Ticket className="h-3.5 w-3.5" /><span className="hidden sm:inline">Coupons</span></TabsTrigger>
   287→            <TabsTrigger value="transactions" className="gap-1.5"><Receipt className="h-3.5 w-3.5" /><span className="hidden sm:inline">Transactions</span></TabsTrigger>
   288→          </TabsList>
   289→
   290→          {/* ── Commerçants ──────────────────────────────────────── */}
   291→          <TabsContent value="merchants" className="space-y-4">
   292→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   293→              <StatCard label="Total" value={merchants.length} icon={Store} />
   294→              <StatCard label="Vérifiés" value={merchants.filter(m => m.isVerified).length} icon={ShieldCheck} />
   295→              <StatCard label="Actifs" value={merchants.filter(m => m.isActive).length} icon={CheckCircle} />
   296→            </div>
   297→            <div className="flex flex-col sm:flex-row gap-3 items-start sm:items-center justify-between">
   298→              <div className="flex gap-2 flex-wrap">
   299→                <div className="relative"><Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" /><Input placeholder="Rechercher..." value={mSearch} onChange={e => setMSearch(e.target.value)} className="pl-9 w-56" /></div>
   300→                <Select value={mCategory} onValueChange={setMCategory}><SelectTrigger className="w-40"><SelectValue placeholder="Catégorie" /></SelectTrigger><SelectContent><SelectItem value="all">Toutes</SelectItem>{CATEGORIES.map(c => <SelectItem key={c} value={c}>{c}</SelectItem>)}</SelectContent></Select>
   301→              </div>
   302→              <Dialog open={merchantDlg} onOpenChange={o => { setMerchantDlg(o); if (!o) { setEditId(null); rstM(); } }}>
   303→                <DialogTrigger asChild><Button size="sm"><Plus className="mr-1.5 h-4 w-4" />Ajouter</Button></DialogTrigger>
   304→                <DialogContent><DialogHeader><DialogTitle>{editId ? 'Modifier le commerçant' : 'Nouveau commerçant'}</DialogTitle><DialogDescription>{editId ? 'Modifier les informations' : 'Ajouter un commerçant'}</DialogDescription></DialogHeader>
   305→                  <div className="grid gap-3 py-2">
   306→                    <div className="grid gap-1.5"><Label>Nom *</Label><Input value={mf.name} onChange={e => setMf(f => ({ ...f, name: e.target.value }))} /></div>
   307→                    <div className="grid gap-1.5"><Label>Catégorie</Label><Select value={mf.category} onValueChange={v => setMf(f => ({ ...f, category: v }))}><SelectTrigger><SelectValue placeholder="Choisir" /></SelectTrigger><SelectContent>{CATEGORIES.map(c => <SelectItem key={c} value={c}>{c}</SelectItem>)}</SelectContent></Select></div>
   308→                    <div className="grid gap-1.5"><Label>Adresse</Label><Input value={mf.address} onChange={e => setMf(f => ({ ...f, address: e.target.value }))} /></div>
   309→                    <div className="grid gap-1.5"><Label>Téléphone</Label><Input value={mf.phone} onChange={e => setMf(f => ({ ...f, phone: e.target.value }))} /></div>
   310→                    <div className="grid gap-1.5"><Label>Site web</Label><Input value={mf.website} onChange={e => setMf(f => ({ ...f, website: e.target.value }))} /></div>

   313→                  <DialogFooter><Button onClick={saveMerchant} disabled={submitting || !mf.name.trim()}>{submitting && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}{editId ? 'Enregistrer' : 'Créer'}</Button></DialogFooter>
   314→                </DialogContent>
   315→              </Dialog>
   316→            </div>
