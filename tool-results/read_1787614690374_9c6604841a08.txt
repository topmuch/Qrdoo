    39→/* ─── Reusable fade-in-up wrapper ─── */
    40→function FadeIn({
    41→  children,
    42→  delay = 0,
    43→  className = '',
    44→}: {
    45→  children: React.ReactNode;
    46→  delay?: number;
    47→  className?: string;
    48→}) {
    49→  return (
    50→    <motion.div
    51→      initial={{ opacity: 0, y: 30 }}
    52→      whileInView={{ opacity: 1, y: 0 }}
    53→      viewport={{ once: true, margin: '-50px' }}
    54→      transition={{ duration: 0.6, delay }}
    55→      className={className}
    56→    >
    57→      {children}
    58→    </motion.div>
    59→  );
    60→}

    62→/* ─── Sticky Navbar ─── */
    63→function Navbar({ onGoToDashboard }: { onGoToDashboard: () => void }) {
    64→  const [mobileOpen, setMobileOpen] = useState(false);
    65→  const [scrolled, setScrolled] = useState(false);
    66→
    67→  useEffect(() => {
    68→    const onScroll = () => setScrolled(window.scrollY > 20);
    69→    window.addEventListener('scroll', onScroll, { passive: true });
    70→    return () => window.removeEventListener('scroll', onScroll);
    71→  }, []);
    72→
    73→  const links = [
    74→    { label: 'Accueil', href: '#hero' },
    75→    { label: 'A propos', href: '#avantages' },
    76→    { label: 'Tarifs', href: '#pricing' },
    77→    { label: 'Contactez nous', href: '#cta-final' },
    78→  ];
    79→
    80→  const handleNavClick = (e: React.MouseEvent<HTMLAnchorElement>, href: string) => {
    81→    e.preventDefault();
    82→    setMobileOpen(false);
    83→    const target = document.querySelector(href);
    84→    if (target) {
    85→      target.scrollIntoView({ behavior: 'smooth' });
    86→    }
    87→  };
    88→
    89→  return (
    90→    <nav
    91→      className={`fixed top-0 left-0 right-0 z-50 transition-all duration-300 ${
    92→        scrolled
    93→          ? 'bg-[#0a0f1e]/95 backdrop-blur-md border-b border-white/10 shadow-lg shadow-black/20'
    94→          : 'bg-[#0a0f1e]/80 backdrop-blur-md border-b border-white/10'
    95→      }`}
    96→    >
    97→      <div className="max-w-7xl mx-auto px-4 md:px-8 flex items-center justify-between h-16">
    98→        {/* Logo */}
    99→        <a
   100→          href="#hero"
   101→          onClick={(e) => handleNavClick(e, '#hero')}
   102→          className="flex items-center gap-2 group"
   103→        >
   104→          <QrCode className="w-6 h-6 text-[#2563EB] group-hover:scale-110 transition-transform" />
   105→          <span className="font-bold text-lg tracking-tight">QR Domotik</span>
   106→        </a>
   107→
   108→        {/* Desktop links */}
   109→        <div className="hidden md:flex items-center gap-8">
   110→          {links.map((link) => (
   111→            <a
   112→              key={link.href}
   113→              href={link.href}
   114→              onClick={(e) => handleNavClick(e, link.href)}
   115→              className="text-sm text-gray-300 hover:text-white transition-colors"
   116→            >
   117→              {link.label}
   118→            </a>
   119→          ))}
   120→          <button
   121→            onClick={onGoToDashboard}
   122→            className="ml-2 px-5 py-2 rounded-lg bg-[#2563EB] text-white text-sm font-medium hover:bg-[#1d4ed8] transition-colors"
   123→          >
   124→            Se connecter
   125→          </button>
   126→        </div>
   127→
   128→        {/* Mobile hamburger */}
   129→        <button
   130→          onClick={() => setMobileOpen(!mobileOpen)}
   131→          className="md:hidden p-2 text-gray-300 hover:text-white transition-colors"
   132→          aria-label="Menu"
   133→        >
   134→          {mobileOpen ? <X className="w-6 h-6" /> : <Menu className="w-6 h-6" />}
   135→        </button>
   136→      </div>
   137→
   138→      {/* Mobile menu */}
   139→      <div
   140→        className={`md:hidden overflow-hidden transition-all duration-300 ${
   141→          mobileOpen ? 'max-h-80 opacity-100' : 'max-h-0 opacity-0'
   142→        }`}
   143→      >
   144→        <div className="px-4 pb-4 space-y-1 bg-[#0a0f1e]/95 backdrop-blur-md border-t border-white/5">
   145→          {links.map((link) => (
   146→            <a
   147→              key={link.href}
   148→              href={link.href}
   149→              onClick={(e) => handleNavClick(e, link.href)}
   150→              className="block py-3 px-3 text-gray-300 hover:text-white hover:bg-white/5 rounded-lg transition-colors"
   151→            >
   152→              {link.label}
   153→            </a>
   154→          ))}
   155→          <button
   156→            onClick={() => {
   157→              setMobileOpen(false);
   158→              onGoToDashboard();
   159→            }}
   160→            className="w-full mt-2 px-5 py-3 rounded-lg bg-[#2563EB] text-white text-sm font-medium hover:bg-[#1d4ed8] transition-colors"
   161→          >
   162→            Se connecter
   163→          </button>
   164→        </div>
   165→      </div>
   166→    </nav>
   167→  );
   168→}

   170→/* ─── Section wrapper ─── */
   171→function Section({
   172→  children,
   173→  className = '',
   174→  id,
   175→}: {
   176→  children: React.ReactNode;
   177→  className?: string;
   178→  id?: string;
   179→}) {
   180→  const ref = useRef(null);
   181→  const isInView = useInView(ref, { once: true, margin: '-50px' });
   182→
   183→  return (
   184→    <motion.section
   185→      id={id}
   186→      ref={ref}
   187→      initial={{ opacity: 0, y: 30 }}
   188→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
   189→      transition={{ duration: 0.6 }}
   190→      className={`py-20 md:py-28 px-4 md:px-8 scroll-mt-16 ${className}`}
   191→    >
   192→      <div className="max-w-7xl mx-auto">{children}</div>
   193→    </motion.section>
   194→  );
   195→}
