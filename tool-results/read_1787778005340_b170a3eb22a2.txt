    18→function FadeIn({ children, delay = 0, className = '' }: { children: React.ReactNode; delay?: number; className?: string }) {
    19→  return (
    20→    <motion.div initial={{ opacity: 0, y: 30 }} whileInView={{ opacity: 1, y: 0 }} viewport={{ once: true, margin: '-50px' }} transition={{ duration: 0.6, delay }} className={className}>
    21→      {children}
    22→    </motion.div>
    23→  );
    24→}

    26→function Navbar({ onGoToDashboard }: { onGoToDashboard: () => void }) {
    27→  const [mobileOpen, setMobileOpen] = useState(false);
    28→  const [scrolled, setScrolled] = useState(false);
    29→
    30→  useEffect(() => {
    31→    const onScroll = () => setScrolled(window.scrollY > 20);
    32→    window.addEventListener('scroll', onScroll, { passive: true });
    33→    return () => window.removeEventListener('scroll', onScroll);
    34→  }, []);
    35→
    36→  const links = [
    37→    { label: 'Accueil', href: '#hero' },
    38→    { label: 'A propos', href: '#avantages' },
    39→    { label: 'Tarifs', href: '#pricing' },
    40→    { label: 'Contactez nous', href: '#cta-final' },
    41→  ];
    42→
    43→  const handleNavClick = (e: React.MouseEvent<HTMLAnchorElement>, href: string) => {
    44→    e.preventDefault();
    45→    setMobileOpen(false);
    46→    const target = document.querySelector(href);
    47→    if (target) target.scrollIntoView({ behavior: 'smooth' });
    48→  };
    49→
    50→  const navBg = scrolled ? 'bg-white/90 backdrop-blur-md border-b border-gray-200 shadow-sm' : 'bg-transparent';
    51→  const txtColor = scrolled ? 'text-gray-900' : 'text-gray-600';
    52→
    53→  return (
    54→    <nav className={`fixed top-0 left-0 right-0 z-50 transition-all duration-300 ${navBg}`}>
    55→      <div className="max-w-7xl mx-auto px-4 md:px-8 flex items-center justify-between h-16">
    56→        <a href="#hero" onClick={(e) => handleNavClick(e, '#hero')} className="flex items-center gap-2 group">
    57→          <div className="w-8 h-8 rounded-lg bg-violet-600 flex items-center justify-center">
    58→            <QrCode className="w-5 h-5 text-white group-hover:scale-110 transition-transform" />
    59→          </div>
    60→          <span className={`font-bold text-lg tracking-tight ${scrolled ? 'text-gray-900' : 'text-gray-800'}`}>QR Domotik</span>
    61→        </a>
    62→        <div className="hidden md:flex items-center gap-8">
    63→          {links.map((link) => (
    64→            <a key={link.href} href={link.href} onClick={(e) => handleNavClick(e, link.href)} className={`text-sm ${txtColor} hover:text-gray-900 transition-colors`}>
    65→              {link.label}
    66→            </a>
    67→          ))}
    68→          <button onClick={onGoToDashboard} className="ml-2 px-5 py-2.5 rounded-xl bg-violet-600 text-white text-sm font-medium hover:bg-violet-700 transition-colors shadow-sm shadow-violet-600/25">
    69→            Se connecter
    70→          </button>
    71→        </div>
    72→        <button onClick={() => setMobileOpen(!mobileOpen)} className="md:hidden p-2 text-gray-600 hover:text-gray-900 transition-colors" aria-label="Menu">
    73→          {mobileOpen ? <X className="w-6 h-6" /> : <Menu className="w-6 h-6" />}
    74→        </button>
    75→      </div>
    76→      <div className={`md:hidden overflow-hidden transition-all duration-300 ${mobileOpen ? 'max-h-80 opacity-100' : 'max-h-0 opacity-0'}`}>
    77→        <div className="px-4 pb-4 space-y-1 bg-white/95 backdrop-blur-md border-t border-gray-100">
    78→          {links.map((link) => (
    79→            <a key={link.href} href={link.href} onClick={(e) => handleNavClick(e, link.href)} className="block py-3 px-3 text-gray-600 hover:text-gray-900 hover:bg-gray-50 rounded-lg transition-colors">
    80→              {link.label}
    81→            </a>
    82→          ))}
    83→          <button onClick={() => { setMobileOpen(false); onGoToDashboard(); }} className="w-full mt-2 px-5 py-3 rounded-xl bg-violet-600 text-white text-sm font-medium hover:bg-violet-700 transition-colors">
    84→            Se connecter
    85→          </button>
    86→        </div>
    87→      </div>
    88→    </nav>
    89→  );
    90→}
