   423→  // ---- Batch validation ----
   424→  async function handleBatchValidate() {
   425→    const raw = batchCodes.trim();
   426→    if (!raw) return;
   427→    const codes = raw
   428→      .split(/[\n,]/)
   429→      .map((c) => c.trim().toUpperCase().replace(/[^A-Z0-9]/g, ''))
   430→      .filter((c) => c.length > 0);
   431→
   432→    setBatchValidating(true);
   433→    setBatchValidCodes([]);
   434→    setBatchInvalidCodes([]);
   435→
   436→    const valid: string[] = [];
   437→    const invalid: string[] = [];
   438→
   439→    // Check each code sequentially
   440→    for (const code of codes) {
   441→      try {
   442→        const res = await fetch(`/api/client/check-code?code=${encodeURIComponent(code)}`);
   443→        const data: CheckCodeResponse = await res.json();
   444→        if (data.valid) {
   445→          valid.push(code);
   446→        } else {
   447→          invalid.push(code);
   448→        }
   449→      } catch {
   450→        invalid.push(code);
   451→      }
   452→    }
   453→
   454→    setBatchValidCodes(valid);
   455→    setBatchInvalidCodes(invalid);
   456→    setBatchValidated(true);
   457→    setBatchValidating(false);
   458→  }
