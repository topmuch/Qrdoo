    39→/* ─── Reusable fade-in-up wrapper ─── */
    40→function FadeIn({
    41→  children,
    42→  delay = 0,
    43→  className = '',
    44→}: {
    45→  children: React.ReactNode;
    46→  delay?: number;
    47→  className?: string;
    48→}) {
    49→  return (
    50→    <motion.div
    51→      initial={{ opacity: 0, y: 30 }}
    52→      whileInView={{ opacity: 1, y: 0 }}
    53→      viewport={{ once: true, margin: '-50px' }}
    54→      transition={{ duration: 0.6, delay }}
    55→      className={className}
    56→    >
    57→      {children}
    58→    </motion.div>
    59→  );
    60→}

    62→/* ─── Sticky Navbar ─── */
    63→function Navbar({ onGoToDashboard }: { onGoToDashboard: () => void }) {
    64→  const [[mmobileOpen, setMobileOpen] = useState(false);
    65→  const [scrolled, setScrolled] = useState(false);
    66→
    67→  useEffect(() => {
    68→    const onScroll = () => setScrolled(window.scrollY > 20);
    69→    window.addEventListener('scroll', onScroll, { passive: true });
    70→    return () => window.removeEventListener('scroll', onScroll);
    71→  }, []);
    72→
    73→  const links = [
    74→    { label: 'Accueil', href: '#hero' },
    75→    { label: 'A propos', href: '#avantages' },
    76→    { label: 'Tarifs', href: '#pricing' },
    77→    { label: 'Contactez nous', href: '#cta-final' },
    78→  ];
    79→
    80→  const handleNavClick = (e: React.MouseEvent<HTMLAnchorElement>, href: string) => {
    81→    e.preventDefault();
    82→    setMobileOpen(false);
    83→    const target = document.querySelector(href);
    84→    if (target) {
    85→      target.scrollIntoView({ behavior: 'smooth' });
    86→    }
    87→  };
    88→
    89→  return (
    90→    <nav
    91→      className={`fixed top-0 left-0 right-0 z-50 transition-all duration-300 ${
    92→        scrolled
    93→          ? 'bg-white/90 backdrop-blur-md border-b border-gray-200 shadow-sm'
    94→          : 'bg-transparent'
    95→      }`
    96→    >
    97→      <div className="max-w-7xl mx-auto px-4 md:px-8 flex items-center justify-between h-16">
    98→        <a
    99→          href="#hero"
   100→          onClick={(e) => handleNavClick(e, '#hero')}
   101→          className="flex items-center gap-2 group"
   102→        >
   103→          <div className="w-8 h-8 rounded-lg bg-violet-600 flex items-center justify-center">
   104→            <QrCode className="w-5 h-5 text-white group-hover:scale-110 transition-transform" />
   105→          </div>
   106→          <span className="font-bold text-lg tracking-tight text-gray-900">QR Domotik</span>
   107→        </a>
   108→
   109→        <div className="hidden md:flex items-center gap-8">
   110→          {links.map((link) => (
   111→            <a
   112→              key={link.href}
   113→              href={link.href}
   114→              onClick={(e) => handleNavClick(e, link.href)}
   115→              className="text-sm text-gray-600 hover:text-gray-900 transition-colors"
   116→            >
   117→              {link.label}
   118→            </a>
   119→          ))}
   120→          <button
   121→            onClick={onGoToDashboard}
   122→            className="ml-2 px-5 py-2.5 rounded-xl bg-violet-600 text-white text-sm font-medium hover:bg-violet-700 transition-colors shadow-sm shadow-violet-600/25"
   123→          >
   124→            Se connecter
   125→          </button>
   126→        </div>
   127→
   128→        <button
   129→          onClick={() => setMobileOpen(!mobileOpen)}
   130→          className="md:hidden p-2 text-gray-600 hover:text-gray-900 transition-colors"
   131→          aria-label="Menu"
   132→        >
   133→          {mobileOpen ? <X className="w-6 h-6" /> : <Menu className="w-6 h-6" />}
   134→        </button>
   135→      </div>
   136→
   137→      <div
   138→        className={`md:hidden overflow-hidden transition-all duration-300 ${
   139→          mobileOpen ? 'max-h-80 opacity-100' : 'max-h-0 opacity-0'
   140→        }`
   141→      >
   142→        <div className="px-4 pb-4 space-y-1 bg-white/95 backdrop-blur-md border-t border-gray-100">
   143→          {links.map((link) => (
   144→            <a
   145→              key={link.href}
   146→              href={link.href}
   147→              onClick={(e) => handleNavClick(e, link.href)}
   148→              className="block py-3 px-3 text-gray-600 hover:text-gray-900 hover:bg-gray-50 rounded-lg transition-colors"
   149→            >
   150→              {link.label}
   151→            </a>
   152→          ))}
   153→          <button
   154→            onClick={() => { setMobileOpen(false); onGoToDashboard(); }}
   155→            className="w-full mt-2 px-5 py-3 rounded-xl bg-violet-600 text-white text-sm font-medium hover:bg-violet-700 transition-colors"
   156→          >
   157→            Se connecter
   158→          </button>
   159→        </div>
   160→      </div>
   161→    </nav>
   162→  );
   163→}

   165→/* ─── Section wrapper ─── */
   166→function Section({
   167→  children,
   168→  className = '',
   169→  id,
   170→}: {
   171→  children: React.ReactNode;
   172→  className?: string;
   173→  id?: string;
   174→}) {
   175→  const ref = useRef(null);
   176→  const isInView = useInView(ref, { once: true, margin: '-50px' });
   177→
   178→  return (
   179→    <motion.section
   180→      id={id}
   181→      ref={ref}
   182→      initial={{ opacity: 0, y: 30 }}
   183→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
   184→      transition={{ duration: 0.6 }}
   185→      className={`py-20 md:py-28 px-4 md:px-8 scroll-mt-16 ${className}`}
   186→    >
   187→      <div className="max-w-7xl mx-auto">{children}</div>
   188→    </motion.section>
   189→  );
   190→}
