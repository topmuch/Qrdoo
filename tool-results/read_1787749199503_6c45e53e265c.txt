   102→// ── Component ──────────────────────────────────────────────────
   103→export function StockManager() {
   104→  // Data state
   105→  const [homeId, setHomeId] = useState<string | null>(null);
   106→  const [products, setProducts] = useState<ProductData[]>([]);
   107→  const [dlcProducts, setDlcProducts] = useState<ProductData[]>([]);
   108→  const [stockProducts, setStockProducts] = useState<ProductData[]>([]);
   109→  const [loading, setLoading] = useState(true);
   110→  const [tabLoading, setTabLoading] = useState(false);
   111→
   112→  // UI state
   113→  const [activeTab, setActiveTab] = useState<TabType>('products');
   114→  const [expandedProducts, setExpandedProducts] = useState<Set<string>>(new Set());
   115→
   116→  // Dialog state
   117→  const [addDialogOpen, setAddDialogOpen] = useState(false);
   118→  const [instanceDialogProduct, setInstanceDialogProduct] = useState<ProductData | null>(null);
   119→
   120→  // Form state
   121→  const [formName, setFormName] = useState('');
   122→  const [formCategory, setFormCategory] = useState<string>('');
   123→  const [formMinStock, setFormMinStock] = useState('1');
   124→  const [formCurrentStock, setFormCurrentStock] = useState('0');
   125→  const [submitting, setSubmitting] = useState(false);
   126→
   127→  // Instance form state
   128→  const [instPurchaseDate, setInstPurchaseDate] = useState('');
   129→  const [instExpiryDate, setInstExpiryDate] = useState('');
   130→  const [instanceSubmitting, setInstanceSubmitting] = useState(false);
   131→
   132→  // Alert check state
   133→  const [checkingAlerts, setCheckingAlerts] = useState(false);
   134→
   135→  // ── Fetch data ──────────────────────────────────────────────
   136→  const fetchProducts = useCallback(async (home: string, filter: string = 'all') => {
   137→    try {
   138→      const res = await fetch(`/api/client/products?homeId=${home}&filter=${filter}`);
   139→      const data = await res.json();
   140→      return (data.products || []) as ProductData[];
   141→    } catch {
   142→      return [];
   143→    }
   144→  }, []);
   145→
   146→  const fetchAllData = useCallback(async () => {
   147→    try {
   148→      const homesRes = await fetch('/api/client/homes');
   149→      const homesData = await homesRes.json();
   150→      const homes: HomeData[] = homesData.homes || [];
   151→      const firstHomeId = homes[0]?.id;
   152→
   153→      if (!firstHomeId) {
   154→        setLoading(false);
   155→        return;
   156→      }
   157→
   158→      setHomeId(firstHomeId);
   159→
   160→      const [allProds] = await Promise.all([
   161→        fetchProducts(firstHomeId, 'all'),
   162→      ]);
   163→
   164→      setProducts(allProds);
   165→    } catch {
   166→      toast.error('Erreur lors du chargement des produits');
   167→    } finally {
   168→      setLoading(false);
   169→    }
   170→  }, [fetchProducts]);
   171→
   172→  useEffect(() => {
   173→    fetchAllData();
   174→  }, [fetchAllData]);
   175→
   176→  const fetchTabData = useCallback(async (tab: TabType) => {
   177→    if (!homeId) return;
   178→    setTabLoading(true);
   179→    try {
   180→      const filterMap: Record<TabType, string> = {
   181→        products: 'all',
   182→        dlc: 'dlc',
   183→        stock: 'stock',
   184→      };
   185→      const data = await fetchProducts(homeId, filterMap[tab]);
   186→      if (tab === 'products') setProducts(data);
   187→      else if (tab === 'dlc') setDlcProducts(data);
   188→      else if (tab === 'stock') setStockProducts(data);
   189→    } catch {
   190→      toast.error('Erreur lors du chargement');
   191→    } finally {
   192→      setTabLoading(false);
   193→    }
   194→  }, [homeId, fetchProducts]);
   195→
   196→  useEffect(() => {
   197→    if (homeId) fetchTabData(activeTab);
   198→  }, [activeTab, homeId, fetchTabData]);
   199→
   200→  // ── Helpers ─────────────────────────────────────────────────
   201→  const toggleExpand = (productId: string) => {
   202→    setExpandedProducts((prev) => {
   203→      const next = new Set(prev);
   204→      if (next.has(productId)) next.delete(productId);
   205→      else next.add(productId);
   206→      return next;
   207→    });
   208→  };
   209→
   210→  const getDaysUntilExpiry = (expiryDate: string | null): number | null => {
   211→    if (!expiryDate) return null;
   212→    const expiry = new Date(expiryDate);
   213→    const now = new Date();
   214→    now.setHours(0, 0, 0, 0);
   215→    expiry.setHours(0, 0, 0, 0);
   216→    return Math.ceil((expiry.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));
   217→  };
   218→
   219→  const getDlcColor = (days: number): string => {
   220→    if (days === 0) return 'bg-red-100 text-red-700 dark:bg-red-900/30 dark:text-red-300 border-0';
   221→    if (days === 1) return 'bg-orange-100 text-orange-700 dark:bg-orange-900/30 dark:text-orange-300 border-0';
   222→    if (days === 2) return 'bg-yellow-100 text-yellow-700 dark:bg-yellow-900/30 dark:text-yellow-300 border-0';
   223→    return 'bg-amber-50 text-amber-600 dark:bg-amber-950/30 dark:text-amber-300 border-0';
   224→  };
   225→
   226→  const getFreshInstances = (product: ProductData) =>
   227→    product.productInstances.filter((i) => i.status === 'fresh' && i.expiryDate);
   228→
   229→  const getStockColor = (current: number, min: number) =>
   230→    current > min ? 'text-emerald-600 dark:text-emerald-400' : 'text-red-600 dark:text-red-400';
   231→
   232→  const getStockProgressColor = (current: number, min: number) => {
   233→    if (current <= 0) return 'bg-red-500';
   234→    if (current <= min) return 'bg-amber-500';
   235→    return 'bg-emerald-500';
   236→  };
   237→
   238→  // ── Handlers ────────────────────────────────────────────────
   239→  const resetForm = () => {
   240→    setFormName('');
   241→    setFormCategory('');
   242→    setFormMinStock('1');
   243→    setFormCurrentStock('0');
   244→  };
   245→
   246→  const handleCreateProduct = async () => {
   247→    if (!formName.trim() || !homeId) return;
   248→
   249→    setSubmitting(true);
   250→    try {
   251→      const res = await fetch('/api/client/products', {
   252→        method: 'POST',
   253→        headers: { 'Content-Type': 'application/json' },
   254→        body: JSON.stringify({
   255→          homeId,
   256→          name: formName.trim(),
   257→          category: formCategory || undefined,
   258→          minStockThreshold: parseInt(formMinStock) || 1,
   259→          currentStock: parseInt(formCurrentStock) || 0,
   260→        }),
   261→      });
   262→      if (!res.ok) throw new Error();
   263→      toast.success('Produit ajouté avec succès');
   264→      setAddDialogOpen(false);
   265→      resetForm();
   266→      fetchTabData(activeTab);
   267→    } catch {
   268→      toast.error('Erreur lors de l\'ajout du produit');
   269→    } finally {
   270→      setSubmitting(false);
   271→    }
   272→  };
   273→
   274→  const handleDeleteProduct = async (productId: string) => {
   275→    if (!confirm('Supprimer ce produit et toutes ses instances ?')) return;
   276→
   277→    try {
   278→      const res = await fetch(`/api/client/products/${productId}`, { method: 'DELETE' });
   279→      if (!res.ok) throw new Error();
   280→      toast.success('Produit supprimé');
   281→      fetchTabData(activeTab);
   282→    } catch {
   283→      toast.error('Erreur lors de la suppression');
   284→    }
   285→  };
   286→
   287→  const handleUpdateStock = async (productId: string, delta: number, currentStock: number) => {
   288→    const newStock = Math.max(0, currentStock + delta);
   289→    try {
   290→      const res = await fetch(`/api/client/products/${productId}`, {
   291→        method: 'PATCH',
   292→        headers: { 'Content-Type': 'application/json' },
   293→        body: JSON.stringify({ currentStock: newStock }),
   294→      });
   295→      if (!res.ok) throw new Error();
   296→      fetchTabData(activeTab);
   297→    } catch {
   298→      toast.error('Erreur lors de la mise à jour du stock');
   299→    }
   300→  };
   301→
   302→  const handleAddInstance = async () => {
   303→    if (!instanceDialogProduct || !homeId) return;
   304→
   305→    setInstanceSubmitting(true);
   306→    try {
   307→      const res = await fetch('/api/client/products/instances', {
   308→        method: 'POST',
   309→        headers: { 'Content-Type': 'application/json' },
   310→        body: JSON.stringify({
   311→          productId: instanceDialogProduct.id,
   312→          homeId,
   313→          purchaseDate: instPurchaseDate || undefined,
   314→          expiryDate: instExpiryDate || undefined,
   315→        }),
   316→      });
   317→      if (!res.ok) throw new Error();
   318→      toast.success('Instance ajoutée avec succès');
   319→      setInstanceDialogProduct(null);
   320→      setInstPurchaseDate('');
   321→      setInstExpiryDate('');
   322→      fetchTabData(activeTab);
   323→    } catch {
   324→      toast.error('Erreur lors de l\'ajout de l\'instance');
   325→    } finally {
   326→      setInstanceSubmitting(false);
   327→    }
   328→  };
   329→
   330→  const handleCheckAlerts = async () => {
   331→    if (!homeId) return;
   332→
   333→    setCheckingAlerts(true);
   334→    try {
   335→      const res = await fetch('/api/client/products/check-alerts', {
   336→        method: 'POST',
   337→        headers: { 'Content-Type': 'application/json' },
   338→        body: JSON.stringify({ homeId }),
   339→      });
   340→      if (!res.ok) throw new Error();
   341→      const data = await res.json();
   342→      toast.success(
   343→        `${data.dlcAlerts} alertes DLC, ${data.stockAlerts} alertes stock, ${data.notifications} notifications envoyées`
   344→      );
   345→      // Refresh data after alert check
   346→      fetchTabData(activeTab);
   347→    } catch {
   348→      toast.error('Erreur lors de la vérification des alertes');
   349→    } finally {
   350→      setCheckingAlerts(false);
   351→    }
   352→  };
   353→
   354→  // ── Loading skeleton ────────────────────────────────────────
   355→  if (loading) {
   356→    return (
   357→      <div className="space-y-6">
   358→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   359→          {Array.from({ length: 3 }).map((_, i) => (
   360→            <Skeleton key={i} className="h-28 rounded-2xl" />
   361→          ))}
   362→        </div>
   363→        <Skeleton className="h-10 rounded-xl" />
   364→        <Skeleton className="h-64 rounded-xl" />
   365→      </div>
   366→    );
   367→  }
   368→
   369→  if (!homeId) {
   370→    return (
   371→      <div className="flex flex-col items-center justify-center py-20 text-center">
   372→        <div className="rounded-2xl border-2 border-dashed border-muted-foreground/25 p-12 max-w-md">
   373→          <Package className="h-12 w-12 mx-auto mb-4 text-muted-foreground" />
   374→          <p className="text-lg font-semibold">Aucune maison</p>
   375→          <p className="mt-2 text-sm text-muted-foreground">
   376→            Créez d'abord une maison pour gérer vos stocks.
   377→          </p>
   378→        </div>
   379→      </div>
   380→    );
   381→  }
   382→
   383→  // ── Computed ────────────────────────────────────────────────
   384→  const totalProducts = products.length;
   385→  const lowStockCount = products.filter((p) => p.currentStock <= p.minStockThreshold).length;
   386→  const shoppingListCount = products.filter((p) => p.isOnShoppingList).length;
   387→
   388→  const currentTabProducts =
   389→    activeTab === 'products' ? products :
   390→    activeTab === 'dlc' ? dlcProducts :
   391→    stockProducts;
   392→
   393→  // ── Render ──────────────────────────────────────────────────
   394→  return (
   395→    <div className="space-y-6">
   396→      {/* Section 1 - Header + Stats */}
   397→      <div>
   398→        <div className="flex items-center gap-3 mb-1">
   399→          <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-gradient-to-br from-orange-500 to-amber-600">
   400→            <Package className="h-5 w-5 text-white" />
   401→          </div>
   402→          <div>
   403→            <h2 className="text-2xl font-bold tracking-tight">Gestion du Stock</h2>
   404→            <p className="text-sm text-muted-foreground">
   405→              Gérez vos produits, suivez les DLC et les stocks bas
   406→            </p>
   407→          </div>
   408→        </div>
   409→      </div>
   410→
   411→      {/* Stat cards */}
   412→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   413→        {/* Total products */}
   414→        <div className="relative overflow-hidden rounded-2xl bg-gradient-to-br from-orange-400 via-orange-500 to-amber-600 p-5 text-white shadow-lg">
   415→          <div className="flex items-center justify-between">
   416→            <div className="space-y-1">
   417→              <p className="text-sm font-medium text-white/80">Produits</p>
   418→              <p className="text-3xl font-bold tabular-nums">{totalProducts}</p>
   419→            </div>
   420→            <div className="flex h-12 w-12 items-center justify-center rounded-xl bg-white/20 backdrop-blur-sm">
   421→              <Package className="h-6 w-6" />
   422→            </div>
   423→          </div>
   424→          <div className="absolute -right-4 -top-4 h-24 w-24 rounded-full bg-white/10" />
   425→          <div className="absolute -right-2 -bottom-6 h-16 w-16 rounded-full bg-white/5" />
   426→        </div>
   427→
   428→        {/* Low stock */}
   429→        <div className="relative overflow-hidden rounded-2xl bg-gradient-to-br from-red-400 via-red-500 to-rose-600 p-5 text-white shadow-lg">
   430→          <div className="flex items-center justify-between">
   431→            <div className="space-y-1">
   432→              <p className="text-sm font-medium text-white/80">Stock bas</p>
   433→              <p className="text-3xl font-bold tabular-nums">{lowStockCount}</p>
   434→            </div>
   435→            <div className="flex h-12 w-12 items-center justify-center rounded-xl bg-white/20 backdrop-blur-sm">
   436→              <AlertTriangle className="h-6 w-6" />
   437→            </div>
   438→          </div>
   439→          <div className="absolute -right-4 -top-4 h-24 w-24 rounded-full bg-white/10" />
   440→          <div className="absolute -right-2 -bottom-6 h-16 w-16 rounded-full bg-white/5" />
   441→        </div>
   442→
   443→        {/* Shopping list */}
   444→        <div className="relative overflow-hidden rounded-2xl bg-gradient-to-br from-emerald-400 via-emerald-500 to-teal-600 p-5 text-white shadow-lg">
   445→          <div className="flex items-center justify-between">
   446→            <div className="space-y-1">
   447→              <p className="text-sm font-medium text-white/80">Liste de courses</p>
   448→              <p className="text-3xl font-bold tabular-nums">{shoppingListCount}</p>
   449→            </div>
   450→            <div className="flex h-12 w-12 items-center justify-center rounded-xl bg-white/20 backdrop-blur-sm">
   451→              <ShoppingCart className="h-6 w-6" />
   452→            </div>
   453→          </div>
   454→          <div className="absolute -right-4 -top-4 h-24 w-24 rounded-full bg-white/10" />
   455→          <div className="absolute -right-2 -bottom-6 h-16 w-16 rounded-full bg-white/5" />
   456→        </div>
   457→      </div>
   458→
   459→      {/* Section 2 - Tab buttons + Actions */}
   460→      <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3">
   461→        <div className="flex items-center gap-2">
   462→          {([
   463→            { key: 'products' as TabType, label: 'Produits', icon: Package },
   464→            { key: 'dlc' as TabType, label: 'Alertes DLC', icon: Clock },
   465→            { key: 'stock' as TabType, label: 'Stock Bas', icon: AlertTriangle },
   466→          ]).map(({ key, label, icon: Icon }) => (
   467→            <Button
   468→              key={key}
   469→              variant={activeTab === key ? 'default' : 'outline'}
   470→              size="sm"
   471→              onClick={() => setActiveTab(key)}
   472→              className={
   473→                activeTab === key
   474→                  ? 'bg-gradient-to-r from-orange-500 to-amber-600 hover:from-orange-600 hover:to-amber-700 text-white shadow-sm'
   475→                  : ''
   476→              }
   477→            >
   478→              <Icon className="h-3.5 w-3.5 mr-1.5" />
   479→              {label}
   480→            </Button>
   481→          ))}
   482→        </div>
   483→
   484→        <div className="flex items-center gap-2">
   485→          {/* Add product dialog */}
   486→          <Dialog open={addDialogOpen} onOpenChange={(open) => { setAddDialogOpen(open); if (!open) resetForm(); }}>
   487→            <DialogTrigger asChild>
   488→              <Button className="gap-2 bg-gradient-to-r from-orange-500 to-amber-600 hover:from-orange-600 hover:to-amber-700 text-white shadow-md">
   489→                <Plus className="h-4 w-4" />
   490→                Ajouter
   491→              </Button>
   492→            </DialogTrigger>
   493→            <DialogContent className="sm:max-w-[480px]">
   494→              <DialogHeader>
   495→                <DialogTitle className="flex items-center gap-2">
   496→                  <Package className="h-5 w-5 text-orange-500" />
   497→                  Nouveau produit
   498→                </DialogTitle>
   499→              </DialogHeader>
   500→
   501→              <div className="space-y-4 py-2">
   502→                <div className="space-y-2">
   503→                  <Label htmlFor="product-name">Nom *</Label>
   504→                  <Input
   505→                    id="product-name"
   506→                    placeholder="Ex: Lait demi-écrémé"
   507→                    value={formName}
   508→                    onChange={(e) => setFormName(e.target.value)}
   509→                  />
   510→                </div>
   511→
   512→                <div className="space-y-2">
   513→                  <Label>Catégorie</Label>
   514→                  <Select value={formCategory} onValueChange={setFormCategory}>
   515→                    <SelectTrigger className="w-full">
   516→                      <SelectValue placeholder="Sélectionner une catégorie" />
   517→                    </SelectTrigger>
   518→                    <SelectContent>
   519→                      {CATEGORIES.map((cat) => (
   520→                        <SelectItem key={cat} value={cat}>
   521→                          {cat}
   522→                        </SelectItem>
   523→                      ))}
   524→                    </SelectContent>
   525→                  </Select>
   526→                </div>
   527→
