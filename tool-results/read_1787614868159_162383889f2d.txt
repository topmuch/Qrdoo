   116→/*  SVG → PNG conversion utility (browser-only, zero native deps)       */
   117→/* ------------------------------------------------------------------ */
   118→
   119→/** Convert an inline SVG element to a PNG data-URL */
   120→function svgElementToPngDataUrl(svgEl: SVGSVGElement, size: number): Promise<string> {
   121→  return new Promise((resolve, reject) => {
   122→    const svgData = new XMLSerializer().serializeToString(svgEl);
   123→    // Use data URL instead of blob — more reliable for Image loading
   124→    const svgUrl = `data:image/svg+xml;charset=utf-8,${encodeURIComponent(svgData)}`;
   125→    const img = new Image();
   126→    img.onload = () => {
   127→      const canvas = document.createElement('canvas');
   128→      canvas.width = size;
   129→      canvas.height = size;
   130→      const ctx = canvas.getContext('2d');
   131→      if (!ctx) { reject(new Error('No 2d context')); return; }
   132→      ctx.drawImage(img, 0, 0, size, size);
   133→      resolve(canvas.toDataURL('image/png'));
   134→    };
   135→    img.onerror = () => reject(new Error('SVG image load failed'));
   136→    img.src = svgUrl;
   137→  });
   138→}

   140→/**
   141→ * Fallback QR PNG generator using qrcode.react (pure JS, no native deps).
   142→ * Renders QRCodeSVG into a hidden container, extracts the SVG, converts to PNG.
   143→ */
   144→async function generateFallbackQrPng(
   145→  data: string,
   146→  size: number,
   147→  fgColor: string,
   148→  bgColor: string,
   149→  level: string,
   150→): Promise<string> {
   151→  const container = document.createElement('div');
   152→  container.style.cssText = 'position:fixed;left:-9999px;top:-9999px;';
   153→  document.body.appendChild(container);
   154→
   155→  const root = createRoot(container);
   156→  root.render(
   157→    createElement(QRCodeSVG, {
   158→      value: data,
   159→      size,
   160→      bgColor,
   161→      fgColor,
   162→      level: level as 'L' | 'M' | 'Q' | 'H',
   163→    }),
   164→  );
   165→
   166→  // Wait for React to commit the render
   167→  await new Promise((r) => setTimeout(r, 150));
   168→
   169→  const svgEl = container.querySelector('svg');
   170→  if (!svgEl) {
   171→    root.unmount();
   172→    document.body.removeChild(container);
   173→    throw new Error('Fallback: SVG element not rendered');
   174→  }
   175→
   176→  // Ensure the SVG has xmlns for proper PNG conversion
   177→  if (!svgEl.getAttribute('xmlns')) {
   178→    svgEl.setAttribute('xmlns', 'http://www.w3.org/2000/svg');
   179→  }
   180→
   181→  const pngUrl = await svgElementToPngDataUrl(svgEl, size);
   182→  root.unmount();
   183→  document.body.removeChild(container);
   184→  return pngUrl;
   185→}
