    42→    42→
    43→    43→interface Home {
    44→    44→  id: string;
    45→    45→  name: string;
    46→    46→}
    47→    47→
    48→    48→interface Automation {
    49→    49→  id: string;
    50→    50→  homeId: string;
    51→    51→  name: string;
    52→    52→  provider: string;
    53→    53→  baseUrl: string;
    54→    54→  apiToken: string | null;
    55→    55→  isActive: boolean;
    56→    56→  createdAt: string;
    57→    57→  updatedAt: string;
    58→    58→  home: Home;
    59→    59→}
    60→    60→
    61→    61→interface HACEntity {
    62→    62→  entity_id: string;
    63→    63→  state: string;
    64→    64→  attributes: {
    65→    65→    friendly_name?: string;
    66→    66→    [key: string]: unknown;
    67→    67→  };
    68→    68→}
    69→    69→
    70→    70→type ConnectionStatus = 'idle' | 'testing' | 'connected' | 'disconnected' | 'error';
    71→    71→
    72→    72→type ActionType = 'toggle' | 'turn_on' | 'turn_off' | 'get_state';
    73→    73→
    74→    74→export function AutomationsManager() {
    75→    75→  const [automations, setAutomations] = useState<Automation[]>([]);
    76→    76→  const [homes, setHomes] = useState<Home[]>([]);
    77→    77→  const [loading, setLoading] = useState(true);
    78→    78→  const [dialogOpen, setDialogOpen] = useState(false);
    79→    79→  const [saving, setSaving] = useState(false);
    80→    80→  const [selectedHomeId, setSelectedHomeId] = useState('');
    81→    81→  const [newName, setNewName] = useState('');
    82→    82→  const [newProvider, setNewProvider] = useState('home_assistant');
    83→    83→  const [newBaseUrl, setNewBaseUrl] = useState('');
    84→    84→  const [newApiToken, setNewApiToken] = useState('');
    85→    85→  // Connection status per automation id
    86→    86→  const [connectionStatuses, setConnectionStatuses] = useState<Record<string, ConnectionStatus>>({});
    87→    87→  const [connectionMessages, setConnectionMessages] = useState<Record<string, string>>({});
    88→    88→  const [expandedAutomation, setExpandedAutomation] = useState<string | null>(null);
    89→    89→  const [entities, setEntities] = useState<Record<string, HACEntity[]>>({});
    90→    90→  const [loadingEntities, setLoadingEntities] = useState<Record<string, boolean>>({});
    91→    91→  const [triggeringEntity, setTriggeringEntity] = useState<string | null>(null);
    92→    92→  const [triggerMessage, setTriggerMessage] = useState<Record<string, string>>({});
    93→    93→
    94→    94→  const fetchAutomations = useCallback(async () => {
    95→    95→    try {
    96→    96→      const res = await fetch('/api/client/automations');
    97→    97→      const data = await res.json();
    98→    98→      setAutomations(data.automations || []);
    99→    99→    } catch {
   100→   100→      // empty
   101→   101→    }
   102→   102→    setLoading(false);
   103→   103→  }, []);
   104→   104→
   105→   105→  const fetchHomes = useCallback(async () => {
   106→   106→    try {
   107→   107→      const res = await fetch('/api/client/homes');
   108→   108→      const data = await res.json();
   109→   109→      setHomes(data.homes || []);
   110→   110→      if (data.homes?.length > 0 && !selectedHomeId) {
   111→   111→        setSelectedHomeId(data.homes[0].id);
   112→   112→      }
   113→   113→    } catch {
   114→   114→      // empty
   115→   115→    }
   116→   116→  }, [selectedHomeId]);
   117→   117→
   118→   118→  useEffect(() => {
   119→   119→    fetchHomes();
   120→   120→    fetchAutomations();
   121→   121→  }, [fetchHomes, fetchAutomations]);
   122→   122→
   123→   123→  const handleCreate = async () => {
   124→   124→    if (!newName || !newBaseUrl) return;
   125→   125→    setSaving(true);
   126→   126→    try {
   127→   127→      const res = await fetch('/api/client/automations', {
   128→   128→        method: 'POST',
   129→   129→        headers: { 'Content-Type': 'application/json' },
   130→   130→        body: JSON.stringify({
   131→   131→          homeId: selectedHomeId,
   132→   132→          name: newName,
   133→   133→          provider: newProvider,
   134→   134→          baseUrl: newBaseUrl,
   135→   135→          apiToken: newApiToken || undefined,
   136→   136→        }),
   137→   137→      });
   138→   138→      if (res.ok) {
   139→   139→        setDialogOpen(false);
   140→   140→        setNewName('');
   141→   141→        setNewBaseUrl('');
   142→   142→        setNewApiToken('');
   143→   143→        fetchAutomations();
   144→   144→      }
   145→   145→    } catch {
   146→   146→      // empty
   147→   147→    }
   148→   148→    setSaving(false);
   149→   149→  };
   150→   150→
   151→   151→  const handleDelete = async (id: string) => {
   152→   152→    if (!confirm('Supprimer cette automatisation ?')) return;
   153→   153→    try {
   154→   154→      await fetch(`/api/client/automations/${id}`, { method: 'DELETE' });
   155→   155→      fetchAutomations();
   156→   156→    } catch {
   157→   157→      // empty
   158→   158→    }
   159→   159→  };
   160→   160→
   161→   161→  const handleToggleActive = async (automation: Automation) => {
   162→   162→    try {
   163→   163→      await fetch(`/api/client/automations/${automation.id}`, {
   164→   164→        method: 'PUT',
   165→   165→        headers: { 'Content-Type': 'application/json' },
   166→   166→        body: JSON.stringify({ isActive: !automation.isActive }),
   167→   167→      });
   168→   168→      fetchAutomations();
   169→   169→    } catch {
   170→   170→      // empty
   171→   171→    }
   172→   172→  };
   173→   173→
   174→   174→  const handleTestConnection = async (automation: Automation) => {
   175→   175→    setConnectionStatuses((prev) => ({ ...prev, [automation.id]: 'testing' }));
   176→   176→    setConnectionMessages((prev) => ({ ...prev, [automation.id]: 'Test en cours...' }));
   177→   177→    try {
   178→   178→      const res = await fetch(`/api/client/automations/${automation.id}/test`, {
   179→   179→        method: 'POST',
   180→   180→      });
   181→   181→      const data = await res.json();
   182→   182→      setConnectionStatuses((prev) => ({
   183→   183→        ...prev,
   184→   184→        [automation.id]: data.success ? 'connected' : 'disconnected',
   185→   185→      }));
   186→   186→      setConnectionMessages((prev) => ({ ...prev, [automation.id]: data.message }));
   187→   187→    } catch {
   188→   188→      setConnectionStatuses((prev) => ({ ...prev, [automation.id]: 'error' }));
   189→   189→      setConnectionMessages((prev) => ({ ...prev, [automation.id]: 'Erreur réseau' }));
   190→   190→    }
   191→   191→  };
   192→   192→
   193→   193→  const fetchEntities = async (automation: Automation) => {
   194→   194→    if (automation.provider !== 'home_assistant') return;
   195→   195→    setLoadingEntities((prev) => ({ ...prev, [automation.id]: true }));
   196→   196→    try {
   197→   197→      const controller = new AbortController();
   198→   198→      const timeout = setTimeout(() => controller.abort(), 10000);
   199→   199→      const res = await fetch(`${automation.baseUrl}/api/states`, {
   200→   200→        headers: {
   201→   201→          Authorization: `Bearer ${automation.apiToken}`,
   202→   202→          'Content-Type': 'application/json',
   203→   203→        },
   204→   204→        signal: controller.signal,
   205→   205→      });
   206→   206→      clearTimeout(timeout);
   207→   207→      if (res.ok) {
   208→   208→        const data = await res.json();
   209→   209→        setEntities((prev) => ({ ...prev, [automation.id]: data.slice(0, 50) }));
   210→   210→      }
   211→   211→    } catch {
   212→   212→      // CORS may block direct fetch
   213→   213→    }
   214→   214→    setLoadingEntities((prev) => ({ ...prev, [automation.id]: false }));
   215→   215→  };
   216→   216→
   217→   217→  const handleToggleExpand = async (automation: Automation) => {
   218→   218→    if (expandedAutomation === automation.id) {
   219→   219→      setExpandedAutomation(null);
   220→   220→    } else {
   221→   221→      setExpandedAutomation(automation.id);
   222→   222→      if (automation.provider === 'home_assistant' && !entities[automation.id]) {
   223→   223→        // Try fetching entities via our API proxy (trigger with get_state action)
   224→   224→        // For HA we can list states via test or we try direct
   225→   225→        fetchEntities(automation);
   226→   226→      }
   227→   227→    }
   228→   228→  };
   229→   229→
   230→   230→  const handleTriggerEntity = async (automation: Automation, entityId: string, action: ActionType) => {
   231→   231→    const key = `${automation.id}-${entityId}`;
   232→   232→    setTriggeringEntity(key);
   233→   233→    setTriggerMessage((prev) => ({ ...prev, [key]: 'Exécution...' }));
   234→   234→    try {
   235→   235→      const res = await fetch(`/api/client/automations/${automation.id}/trigger`, {
   236→   236→        method: 'POST',
   237→   237→        headers: { 'Content-Type': 'application/json' },
   238→   238→        body: JSON.stringify({ entityId, action }),
   239→   239→      });
   240→   240→      const data = await res.json();
   241→   241→      setTriggerMessage((prev) => ({ ...prev, [key]: data.message }));
   242→   242→      // Refresh entities after action
   243→   243→      if (automation.provider === 'home_assistant') {
   244→   244→        fetchEntities(automation);
   245→   245→      }
   246→   246→    } catch {
   247→   247→      setTriggerMessage((prev) => ({ ...prev, [key]: 'Erreur réseau' }));
   248→   248→    }
   249→   249→    setTriggeringEntity(null);
   250→   250→  };
   251→   251→
   252→   252→  const getConnectionIcon = (automationId: string) => {
   253→   253→    const status = connectionStatuses[automationId];
   254→   254→    switch (status) {
   255→   255→      case 'connected':
   256→   256→        return <CheckCircle2 className="h-4 w-4 text-emerald-500" />;
   257→   257→      case 'disconnected':
   258→   258→      case 'error':
   259→   259→        return <XCircle className="h-4 w-4 text-red-500" />;
   260→   260→      case 'testing':
   261→   261→        return <Loader2 className="h-4 w-4 text-amber-500 animate-spin" />;
   262→   262→      default:
   263→   263→        return <div className="h-2 w-2 rounded-full bg-muted-foreground/30" />;
   264→   264→    }
   265→   265→  };
   266→   266→
   267→   267→  if (loading) {
   268→   268→    return (
   269→   269→      <div className="flex items-center justify-center py-20">
   270→   270→        <div className="h-8 w-8 border-2 border-violet-300/30 border-t-violet-500 rounded-full animate-spin" />
   271→   271→      </div>
   272→   272→    );
   273→   273→  }
   274→   274→
   275→   275→  return (
   276→   276→    <div className="space-y-6">
   277→   277→      {/* Header */}
   278→   278→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   279→   279→        <div>
   280→   280→          <h3 className="text-lg font-semibold">Domotique (Home Assistant / Jeedom)</h3>
   281→   281→          <p className="text-sm text-muted-foreground">
   282→   282→            Connectez et contrôlez vos appareils domotiques
   283→   283→          </p>
   284→   284→        </div>
   285→   285→        <Button
   286→   286→          onClick={() => setDialogOpen(true)}
   287→   287→          className="bg-violet-600 hover:bg-violet-700"
   288→   288→          disabled={homes.length === 0}
   289→   289→        >
   290→   290→          <Plus className="h-4 w-4 mr-2" />
   291→   291→          Ajouter une automatisation
   292→   292→        </Button>
   293→   293→      </div>
   294→   294→
   295→   295→      {homes.length === 0 && (
   296→   296→        <Card>
   297→   297→          <CardContent className="p-6 text-center">
   298→   298→            <p className="text-muted-foreground">Créez d'abord une maison pour ajouter des automatisations.</p>
   299→   299→          </CardContent>
   300→   300→        </Card>
   301→   301→      )}
   302→   302→
   303→   303→      {/* Automations list */}
   304→   304→      {automations.length === 0 && homes.length > 0 && (
   305→   305→        <Card>
   306→   306→          <CardContent className="flex flex-col items-center justify-center py-16 gap-4">
   307→   307→          <div className="rounded-2xl bg-violet-100 dark:bg-violet-500/10 p-4">
   308→   308→            <Plug className="h-8 w-8 text-violet-600" />
   309→   309→          </div>
   310→   310→          <div className="text-center">
   311→   311→            <p className="font-medium">Aucune automatisation configurée</p>
   312→   312→            <p className="text-sm text-muted-foreground mt-1">
   313→   313→              Connectez Home Assistant ou Jeedom pour commencer
   314→   314→            </p>
   315→   315→          </div>
   316→   316→          </CardContent>
   317→   317→        </Card>
   318→   318→      )}
   319→   319→
   320→   320→      <div className="grid gap-4">
   321→   321→        {automations.map((automation) => {
   322→   322→          const isExpanded = expandedAutomation === automation.id;
   323→   323→          const haEntities = entities[automation.id] || [];
   324→   324→
   325→   325→          return (
   326→   326→            <Card key={automation.id} className={!automation.isActive ? 'opacity-60' : ''}>
   327→   327→              <CardContent className="p-4">
   328→   328→                <div className="flex items-center gap-3">
   329→   329→                  {/* Provider icon */}
   330→   330→                  <div className={`rounded-lg p-2 ${
   331→   331→                    automation.provider === 'home_assistant'
   332→   332→                      ? 'bg-sky-100 dark:bg-sky-500/10'
   333→   333→                      : 'bg-amber-100 dark:bg-amber-500/10'
   334→   334→                  }`>
   335→   335→                    {automation.provider === 'home_assistant' ? (
   336→   336→                      <HomeAssistantLogo className="h-6 w-6 text-sky-600" />
   337→   337→                    ) : (
   338→   338→                      <JeedomLogo className="h-6 w-6 text-amber-600" />
   339→   339→                    )}
   340→   340→                  </div>
   341→   341→
   342→   342→                  {/* Info */}
   343→   343→                  <div className="flex-1 min-w-0">
   344→   344→                    <div className="flex items-center gap-2">
   345→   345→                      <h4 className="font-medium truncate">{automation.name}</h4>
   346→   346→                      {getConnectionIcon(automation.id)}
   347→   347→                    </div>
   348→   348→                    <div className="flex items-center gap-2 mt-0.5">
   349→   349→                      <span className="text-xs text-muted-foreground truncate">
   350→   350→                        {automation.home.name}
   351→   351→                      </span>
   352→   352→                      <span className="text-xs text-muted-foreground">·</span>
   353→   353→                      <Badge variant="secondary" className="text-[10px] px-1.5 py-0">
   354→   354→                        {automation.provider === 'home_assistant' ? 'HA' : 'Jeedom'}
   355→   355→                      </Badge>
   356→   356→                      <span className="text-xs text-muted-foreground truncate">
   357→   357→                        {automation.baseUrl}
   358→   358→                      </span>
   359→   359→                    </div>
   360→   360→                    {connectionMessages[automation.id] && connectionStatuses[automation.id] !== 'idle' && (
   361→   361→                      <p className={`text-xs mt-1 ${
   362→   362→                        connectionStatuses[automation.id] === 'connected'
   363→   363→                          ? 'text-emerald-600'
   364→   364→                          : 'text-red-500'
   365→   365→                      }`}>
   366→   366→                        {connectionMessages[automation.id]}
   367→   367→                      </p>
   368→   368→                    )}
   369→   369→                  </div>
   370→   370→
   371→   371→                  {/* Actions */}
   372→   372→                  <div className="flex items-center gap-1">
   373→   373→                    <Button
   374→   374→                      variant="ghost"
   375→   375→                      size="icon"
   376→   376→                      className="h-8 w-8"
   377→   377→                      onClick={() => handleTestConnection(automation)}
   378→   378→                      disabled={connectionStatuses[automation.id] === 'testing'}
   379→   379→                      title="Tester la connexion"
   380→   380→                    >
   381→   381→                      <Wifi className="h-4 w-4" />
   382→   382→                    </Button>
   383→   383→                    {automation.provider === 'home_assistant' && (
   384→   384→                      <Button
   385→   385→                        variant="ghost"
   386→   386→                        size="icon"
   387→   387→                        className="h-8 w-8"
   388→   388→                        onClick={() => handleToggleExpand(automation)}
   389→   389→                        title="Parcourir les entités"
   390→   390→                      >
   391→   391→                        {isExpanded ? <ChevronUp className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   392→   392→                      </Button>
   393→   393→                    )}
   394→   394→                    <Switch
   395→   395→                      checked={automation.isActive}
   396→   396→                      onCheckedChange={() => handleToggleActive(automation)}
   397→   397→                    />
   398→   398→                    <Button
   399→   399→                      variant="ghost"
   400→   400→                      size="icon"
   401→   401→                      className="h-8 w-8 text-red-400 hover:text-red-600"
   402→   402→                      onClick={() => handleDelete(automation.id)}
   403→   403→                    >
   404→   404→                      <Trash2 className="h-4 w-4" />
   405→   405→                    </Button>
   406→   406→                  </div>
   407→   407→                </div>
   408→   408→
   409→   409→                {/* Expanded entity browser for HA */}
   410→   410→                {isExpanded && automation.provider === 'home_assistant' && (
   411→   411→                  <div className="mt-4 border-t pt-4">
   412→   412→                    <div className="flex items-center justify-between mb-3">
   413→   413→                      <h5 className="text-sm font-medium flex items-center gap-2">
   414→   414→                        <Server className="h-4 w-4" />
   415→   415→                        Entités ({haEntities.length})
   416→   416→                      </h5>
   417→   417→                      <Button
   418→   418→                        variant="outline"
   419→   419→                        size="sm"
   420→   420→                        onClick={() => fetchEntities(automation)}
   421→   421→                        disabled={loadingEntities[automation.id]}
   422→   422→                      >
   423→   423→                        <RefreshCw className={`h-3.5 w-3.5 mr-1 ${loadingEntities[automation.id] ? 'animate-spin' : ''}`} />
   424→   424→                        Rafraîchir
   425→   425→                      </Button>
   426→   426→                    </div>
   427→   427→
   428→   428→                    {loadingEntities[automation.id] ? (
   429→   429→                      <div className="flex items-center justify-center py-8">
   430→   430→                        <Loader2 className="h-5 w-5 animate-spin text-muted-foreground" />
   431→   431→                      </div>
   432→   432→                    ) : haEntities.length === 0 ? (
   433→   433→                      <div className="flex items-center justify-center py-8">
   434→   434→                        <p className="text-sm text-muted-foreground">
   435→   435→                          Aucune entité trouvée. Assurez-vous que l'API est accessible.
   436→   436→                        </p>
   437→   437→                      </div>
   438→   438→                    ) : (
   439→   439→                      <div className="max-h-96 overflow-y-auto space-y-2 pr-1">
   440→   440→                        {haEntities.map((entity) => {
   441→   441→                          const triggerKey = `${automation.id}-${entity.entity_id}`;
   442→   442→                          const isOn = ['on', 'open', 'unlocked'].includes(entity.state);
   443→   443→                          const domain = entity.entity_id.split('.')[0];
   444→   444→                          const showToggle = ['light', 'switch', 'fan', 'binary_sensor', 'cover', 'lock'].includes(domain);
   445→   445→
   446→   446→                          return (
   447→   447→                            <div
   448→   448→                              key={entity.entity_id}
   449→   449→                              className="flex items-center gap-2 rounded-lg border p-2.5 text-sm"
   450→   450→                            >
   451→   451→                              <div className={`h-2.5 w-2.5 rounded-full shrink-0 ${
   452→   452→                                isOn ? 'bg-emerald-500' : 'bg-gray-300 dark:bg-gray-600'
   453→   453→                              }`} />
   454→   454→                              <div className="flex-1 min-w-0">
   455→   455→                                <p className="font-medium truncate">
   456→   456→                                  {entity.attributes.friendly_name || entity.entity_id}
   457→   457→                                </p>
   458→   458→                                <p className="text-xs text-muted-foreground truncate">
   459→   459→                                  {entity.entity_id} · {entity.state}
   460→   460→                                </p>
   461→   461→                              </div>
   462→   462→                              <div className="flex items-center gap-1 shrink-0">
   463→   463→                                {showToggle && (
   464→   464→                                  <Button
   465→   465→                                    variant="outline"
   466→   466→                                    size="sm"
   467→   467→                                    className="h-7 text-xs"
   468→   468→                                    onClick={() => handleTriggerEntity(automation, entity.entity_id, 'toggle')}
