     1→'use client';
     2→
     3→import { useState, useEffect, useCallback } from 'react';
     4→import {
     5→  Server,
     6→  Wifi,
     7→  Plug,
     8→  Zap,
     9→  RefreshCw,
    10→  Plus,
    11→  Trash2,
    12→  CheckCircle2,
    13→  XCircle,
    14→  Power,
    15→  PowerOff,
    16→  ChevronDown,
    17→  ChevronUp,
    18→  Loader2,
    19→  Eye,
    20→} from 'lucide-react';
    21→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card';
    22→import { Button } from '@/components/ui/button';
    23→import { Input } from '@/components/ui/input';
    24→import { Badge } from '@/components/ui/badge';
    25→import { Switch } from '@/components/ui/switch';
    26→import { Label } from '@/components/ui/label';
    27→import {
    28→  Dialog,
    29→  DialogContent,
    30→  DialogHeader,
    31→  DialogTitle,
    32→  DialogDescription,
    33→  DialogFooter,
    34→} from '@/components/ui/dialog';
    35→import {
    36→  Select,
    37→  SelectContent,
    38→  SelectItem,
    39→  SelectTrigger,
    40→  SelectValue,
    41→} from '@/components/ui/select';
    42→
    43→interface Home {
    44→  id: string;
    45→  name: string;
    46→}
    47→
    48→interface Automation {
    49→  id: string;
    50→  homeId: string;
    51→  name: string;
    52→  provider: string;
    53→  baseUrl: string;
    54→  apiToken: string | null;
    55→  isActive: boolean;
    56→  createdAt: string;
    57→  updatedAt: string;
    58→  home: Home;
    59→}
    60→
    61→interface HACEntity {
    62→  entity_id: string;
    63→  state: string;
    64→  attributes: {
    65→    friendly_name?: string;
    66→    [key: string]: unknown;
    67→  };
    68→}
    69→
    70→type ConnectionStatus = 'idle' | 'testing' | 'connected' | 'disconnected' | 'error';
    71→
    72→type ActionType = 'toggle' | 'turn_on' | 'turn_off' | 'get_state';
    73→
    74→export function AutomationsManager() {
    75→  const [automations, setAutomations] = useState<Automation[]>([]);
    76→  const [homes, setHomes] = useState<Home[]>([]);
    77→  const [loading, setLoading] = useState(true);
    78→  const [dialogOpen, setDialogOpen] = useState(false);
    79→  const [saving, setSaving] = useState(false);
    80→  const [selectedHomeId, setSelectedHomeId] = useState('');
    81→  const [newName, setNewName] = useState('');
    82→  const [newProvider, setNewProvider] = useState('home_assistant');
    83→  const [newBaseUrl, setNewBaseUrl] = useState('');
    84→  const [newApiToken, setNewApiToken] = useState('');
    85→  // Connection status per automation id
    86→  const [connectionStatuses, setConnectionStatuses] = useState<Record<string, ConnectionStatus>>({});
    87→  const [connectionMessages, setConnectionMessages] = useState<Record<string, string>>({});
    88→  const [expandedAutomation, setExpandedAutomation] = useState<string | null>(null);
    89→  const [entities, setEntities] = useState<Record<string, HACEntity[]>>({});
    90→  const [loadingEntities, setLoadingEntities] = useState<Record<string, boolean>>({});
    91→  const [triggeringEntity, setTriggeringEntity] = useState<string | null>(null);
    92→  const [triggerMessage, setTriggerMessage] = useState<Record<string, string>>({});
    93→
    94→  const fetchAutomations = useCallback(async () => {
    95→    try {
    96→      const res = await fetch('/api/client/automations');
    97→      const data = await res.json();
    98→      setAutomations(data.automations || []);
    99→    } catch {
   100→      // empty
   101→    }
   102→    setLoading(false);
   103→  }, []);
   104→
   105→  const fetchHomes = useCallback(async () => {
   106→    try {
   107→      const res = await fetch('/api/client/homes');
   108→      const data = await res.json();
   109→      setHomes(data.homes || []);
   110→      if (data.homes?.length > 0 && !selectedHomeId) {
   111→        setSelectedHomeId(data.homes[0].id);
   112→      }
   113→    } catch {
   114→      // empty
   115→    }
   116→  }, [selectedHomeId]);
   117→
   118→  useEffect(() => {
   119→    fetchHomes();
   120→    fetchAutomations();
   121→  }, [fetchHomes, fetchAutomations]);
   122→
   123→  const handleCreate = async () => {
   124→    if (!newName || !newBaseUrl) return;
   125→    setSaving(true);
   126→    try {
   127→      const res = await fetch('/api/client/automations', {
   128→        method: 'POST',
   129→        headers: { 'Content-Type': 'application/json' },
   130→        body: JSON.stringify({
   131→          homeId: selectedHomeId,
   132→          name: newName,
   133→          provider: newProvider,
   134→          baseUrl: newBaseUrl,
   135→          apiToken: newApiToken || undefined,
   136→        }),
   137→      });
   138→      if (res.ok) {
   139→        setDialogOpen(false);
   140→        setNewName('');
   141→        setNewBaseUrl('');
   142→        setNewApiToken('');
   143→        fetchAutomations();
   144→      }
   145→    } catch {
   146→      // empty
   147→    }
   148→    setSaving(false);
   149→  };
   150→
   151→  const handleDelete = async (id: string) => {
   152→    if (!confirm('Supprimer cette automatisation ?')) return;
   153→    try {
   154→      await fetch(`/api/client/automations/${id}`, { method: 'DELETE' });
   155→      fetchAutomations();
   156→    } catch {
   157→      // empty
   158→    }
   159→  };
   160→
   161→  const handleToggleActive = async (automation: Automation) => {
   162→    try {
   163→      await fetch(`/api/client/automations/${automation.id}`, {
   164→        method: 'PUT',
   165→        headers: { 'Content-Type': 'application/json' },
   166→        body: JSON.stringify({ isActive: !automation.isActive }),
   167→      });
   168→      fetchAutomations();
   169→    } catch {
   170→      // empty
   171→    }
   172→  };
   173→
   174→  const handleTestConnection = async (automation: Automation) => {
   175→    setConnectionStatuses((prev) => ({ ...prev, [automation.id]: 'testing' }));
   176→    setConnectionMessages((prev) => ({ ...prev, [automation.id]: 'Test en cours...' }));
   177→    try {
   178→      const res = await fetch(`/api/client/automations/${automation.id}/test`, {
   179→        method: 'POST',
   180→      });
   181→      const data = await res.json();
   182→      setConnectionStatuses((prev) => ({
   183→        ...prev,
   184→        [automation.id]: data.success ? 'connected' : 'disconnected',
   185→      }));
   186→      setConnectionMessages((prev) => ({ ...prev, [automation.id]: data.message }));
   187→    } catch {
   188→      setConnectionStatuses((prev) => ({ ...prev, [automation.id]: 'error' }));
   189→      setConnectionMessages((prev) => ({ ...prev, [automation.id]: 'Erreur réseau' }));
   190→    }
   191→  };
   192→
   193→  const fetchEntities = async (automation: Automation) => {
   194→    if (automation.provider !== 'home_assistant') return;
   195→    setLoadingEntities((prev) => ({ ...prev, [automation.id]: true }));
   196→    try {
   197→      const controller = new AbortController();
   198→      const timeout = setTimeout(() => controller.abort(), 10000);
   199→      const res = await fetch(`${automation.baseUrl}/api/states`, {
   200→        headers: {
   201→          Authorization: `Bearer ${automation.apiToken}`,
   202→          'Content-Type': 'application/json',
   203→        },
   204→        signal: controller.signal,
   205→      });
   206→      clearTimeout(timeout);
   207→      if (res.ok) {
   208→        const data = await res.json();
   209→        setEntities((prev) => ({ ...prev, [automation.id]: data.slice(0, 50) }));
   210→      }
   211→    } catch {
   212→      // CORS may block direct fetch
   213→    }
   214→    setLoadingEntities((prev) => ({ ...prev, [automation.id]: false }));
   215→  };
   216→
   217→  const handleToggleExpand = async (automation: Automation) => {
   218→    if (expandedAutomation === automation.id) {
   219→      setExpandedAutomation(null);
   220→    } else {
   221→      setExpandedAutomation(automation.id);
   222→      if (automation.provider === 'home_assistant' && !entities[automation.id]) {
   223→        // Try fetching entities via our API proxy (trigger with get_state action)
   224→        // For HA we can list states via test or we try direct
   225→        fetchEntities(automation);
   226→      }
   227→    }
   228→  };
   229→
   230→  const handleTriggerEntity = async (automation: Automation, entityId: string, action: ActionType) => {
   231→    const key = `${automation.id}-${entityId}`;
   232→    setTriggeringEntity(key);
   233→    setTriggerMessage((prev) => ({ ...prev, [key]: 'Exécution...' }));
   234→    try {
   235→      const res = await fetch(`/api/client/automations/${automation.id}/trigger`, {
   236→        method: 'POST',
   237→        headers: { 'Content-Type': 'application/json' },
   238→        body: JSON.stringify({ entityId, action }),
   239→      });
   240→      const data = await res.json();
   241→      setTriggerMessage((prev) => ({ ...prev, [key]: data.message }));
   242→      // Refresh entities after action
   243→      if (automation.provider === 'home_assistant') {
   244→        fetchEntities(automation);
   245→      }
   246→    } catch {
   247→      setTriggerMessage((prev) => ({ ...prev, [key]: 'Erreur réseau' }));
   248→    }
   249→    setTriggeringEntity(null);
   250→  };
   251→
   252→  const getConnectionIcon = (automationId: string) => {
   253→    const status = connectionStatuses[automationId];
   254→    switch (status) {
   255→      case 'connected':
   256→        return <CheckCircle2 className="h-4 w-4 text-emerald-500" />;
   257→      case 'disconnected':
   258→      case 'error':
   259→        return <XCircle className="h-4 w-4 text-red-500" />;
   260→      case 'testing':
   261→        return <Loader2 className="h-4 w-4 text-amber-500 animate-spin" />;
   262→      default:
   263→        return <div className="h-2 w-2 rounded-full bg-muted-foreground/30" />;
   264→    }
   265→  };
   266→
   267→  if (loading) {
   268→    return (
   269→      <div className="flex items-center justify-center py-20">
   270→        <div className="h-8 w-8 border-2 border-violet-300/30 border-t-violet-500 rounded-full animate-spin" />
   271→      </div>
   272→    );
   273→  }
   274→
   275→  return (
   276→    <div className="space-y-6">
   277→      {/* Header */}
   278→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   279→        <div>
   280→          <h3 className="text-lg font-semibold">Domotique (Home Assistant / Jeedom)</h3>
   281→          <p className="text-sm text-muted-foreground">
   282→            Connectez et contrôlez vos appareils domotiques
   283→          </p>
   284→        </div>
   285→        <Button
   286→          onClick={() => setDialogOpen(true)}
   287→          className="bg-violet-600 hover:bg-violet-700"
   288→          disabled={homes.length === 0}
   289→        >
   290→          <Plus className="h-4 w-4 mr-2" />
   291→          Ajouter une automatisation
   292→        </Button>
   293→      </div>
   294→
   295→      {homes.length === 0 && (
   296→        <Card>
   297→          <CardContent className="p-6 text-center">
   298→            <p className="text-muted-foreground">Créez d'abord une maison pour ajouter des automatisations.</p>
   299→          </CardContent>
   300→        </Card>
   301→      )}
   302→
   303→      {/* Automations list */}
   304→      {automations.length === 0 && homes.length > 0 && (
   305→        <Card>
   306→          <CardContent className="flex flex-col items-center justify-center py-16 gap-4">
   307→          <div className="rounded-2xl bg-violet-100 dark:bg-violet-500/10 p-4">
   308→            <Plug className="h-8 w-8 text-violet-600" />
   309→          </div>
   310→          <div className="text-center">
   311→            <p className="font-medium">Aucune automatisation configurée</p>
   312→            <p className="text-sm text-muted-foreground mt-1">
   313→              Connectez Home Assistant ou Jeedom pour commencer
   314→            </p>
   315→          </div>
   316→          </CardContent>
   317→        </Card>
   318→      )}
   319→
   320→      <div className="grid gap-4">
   321→        {automations.map((automation) => {
   322→          const isExpanded = expandedAutomation === automation.id;
   323→          const haEntities = entities[automation.id] || [];
   324→
   325→          return (
   326→            <Card key={automation.id} className={!automation.isActive ? 'opacity-60' : ''}>
   327→              <CardContent className="p-4">
   328→                <div className="flex items-center gap-3">
   329→                  {/* Provider icon */}
   330→                  <div className={automation.provider === 'home_assistant' ? 'rounded-lg p-2 bg-sky-100 dark:bg-sky-500/10' : 'rounded-lg p-2 bg-amber-100 dark:bg-amber-500/10'}>
   331→                    {automation.provider === 'home_assistant' ? (
   332→                      <Server className="h-6 w-6 text-sky-600" />
   333→                    ) : (
   334→                      <Zap className="h-6 w-6 text-amber-600" />
   335→                    )}
   336→                  </div>
   337→
   338→                  {/* Info */}
   339→                  <div className="flex-1 min-w-0">
   340→                    <div className="flex items-center gap-2">
   341→                      <h4 className="font-medium truncate">{automation.name}</h4>
   342→                      {getConnectionIcon(automation.id)}
   343→                    </div>
   344→                    <div className="flex items-center gap-2 mt-0.5">
   345→                      <span className="text-xs text-muted-foreground truncate">
   346→                        {automation.home.name}
   347→                      </span>
   348→                      <span className="text-xs text-muted-foreground">·</span>
   349→                      <Badge variant="secondary" className="text-[10px] px-1.5 py-0">
   350→                        {automation.provider === 'home_assistant' ? 'HA' : 'Jeedom'}
   351→                      </Badge>
   352→                      <span className="text-xs text-muted-foreground truncate">
   353→                        {automation.baseUrl}
   354→                      </span>
   355→                    </div>
   356→                    {connectionMessages[automation.id] && connectionStatuses[automation.id] !== 'idle' && (
   357→                      <p className={`text-xs mt-1 ${
   358→                        connectionStatuses[automation.id] === 'connected'
   359→                          ? 'text-emerald-600'
   360→                          : 'text-red-500'
   361→                      }`}>
   362→                        {connectionMessages[automation.id]}
   363→                      </p>
   364→                    )}
   365→                  </div>
   366→
   367→                  {/* Actions */}
   368→                  <div className="flex items-center gap-1">
   369→                    <Button
   370→                      variant="ghost"
   371→                      size="icon"
   372→                      className="h-8 w-8"
   373→                      onClick={() => handleTestConnection(automation)}
   374→                      disabled={connectionStatuses[automation.id] === 'testing'}
   375→                      title="Tester la connexion"
   376→                    >
   377→                      <Wifi className="h-4 w-4" />
   378→                    </Button>
   379→                    {automation.provider === 'home_assistant' && (
   380→                      <Button
   381→                        variant="ghost"
   382→                        size="icon"
   383→                        className="h-8 w-8"
   384→                        onClick={() => handleToggleExpand(automation)}
   385→                        title="Parcourir les entités"
   386→                      >
   387→                        {isExpanded ? <ChevronUp className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   388→                      </Button>
   389→                    )}
   390→                    <Switch
   391→                      checked={automation.isActive}
   392→                      onCheckedChange={() => handleToggleActive(automation)}
   393→                    />
   394→                    <Button
   395→                      variant="ghost"
   396→                      size="icon"
   397→                      className="h-8 w-8 text-red-400 hover:text-red-600"
   398→                      onClick={() => handleDelete(automation.id)}
   399→                    >
   400→                      <Trash2 className="h-4 w-4" />
   401→                    </Button>
   402→                  </div>
   403→                </div>
   404→
   405→                {/* Expanded entity browser for HA */}
   406→                {isExpanded && automation.provider === 'home_assistant' && (
   407→                  <div className="mt-4 border-t pt-4">
   408→                    <div className="flex items-center justify-between mb-3">
   409→                      <h5 className="text-sm font-medium flex items-center gap-2">
   410→                        <Server className="h-4 w-4" />
   411→                        Entités ({haEntities.length})
   412→                      </h5>
   413→                      <Button
   414→                        variant="outline"
   415→                        size="sm"
   416→                        onClick={() => fetchEntities(automation)}
   417→                        disabled={loadingEntities[automation.id]}
   418→                      >
   419→                        <RefreshCw className={`h-3.5 w-3.5 mr-1 ${loadingEntities[automation.id] ? 'animate-spin' : ''}`} />
   420→                        Rafraîchir
   421→                      </Button>
   422→                    </div>
   423→
   424→                    {loadingEntities[automation.id] ? (
   425→                      <div className="flex items-center justify-center py-8">
   426→                        <Loader2 className="h-5 w-5 animate-spin text-muted-foreground" />
   427→                      </div>
   428→                    ) : haEntities.length === 0 ? (
   429→                      <div className="flex items-center justify-center py-8">
   430→                        <p className="text-sm text-muted-foreground">
   431→                          Aucune entité trouvée. Assurez-vous que l'API est accessible.
   432→                        </p>
   433→                      </div>
   434→                    ) : (
   435→                      <div className="max-h-96 overflow-y-auto space-y-2 pr-1">
   436→                        {haEntities.map((entity) => {
   437→                          const triggerKey = `${automation.id}-${entity.entity_id}`;
   438→                          const isOn = ['on', 'open', 'unlocked'].includes(entity.state);
   439→                          const domain = entity.entity_id.split('.')[0];
   440→                          const showToggle = ['light', 'switch', 'fan', 'binary_sensor', 'cover', 'lock'].includes(domain);
   441→
   442→                          return (
   443→                            <div
   444→                              key={entity.entity_id}
   445→                              className="flex items-center gap-2 rounded-lg border p-2.5 text-sm"
   446→                            >
   447→                              <div className={`h-2.5 w-2.5 rounded-full shrink-0 ${
   448→                                isOn ? 'bg-emerald-500' : 'bg-gray-300 dark:bg-gray-600'
   449→                              }`} />
   450→                              <div className="flex-1 min-w-0">
   451→                                <p className="font-medium truncate">
   452→                                  {entity.attributes.friendly_name || entity.entity_id}
   453→                                </p>
   454→                                <p className="text-xs text-muted-foreground truncate">
   455→                                  {entity.entity_id} · {entity.state}
   456→                                </p>
   457→                              </div>
   458→                              <div className="flex items-center gap-1 shrink-0">
   459→                                {showToggle && (
   460→                                  <Button
   461→                                    variant="outline"
   462→                                    size="sm"
   463→                                    className="h-7 text-xs"
   464→                                    onClick={() => handleTriggerEntity(automation, entity.entity_id, 'toggle')}
   465→                                    disabled={triggeringEntity === triggerKey}
   466→                                  >
   467→                                    {triggeringEntity === triggerKey ? (
   468→                                      <Loader2 className="h-3 w-3 animate-spin" />
   469→                                    ) : isOn ? (
   470→                                      <PowerOff className="h-3 w-3" />
   471→                                    ) : (
   472→                                      <Power className="h-3 w-3" />
   473→                                    )}
   474→                                  </Button>
   475→                                )}
   476→                                <Button
   477→                                  variant="ghost"
   478→                                  size="sm"
   479→                                  className="h-7 text-xs"
   480→                                  onClick={() => handleTriggerEntity(automation, entity.entity_id, 'get_state')}
   481→                                  disabled={triggeringEntity === triggerKey}
   482→                                >
   483→                                  <Eye className="h-3 w-3" />
   484→                                </Button>
   485→                              </div>
   486→                              {triggerMessage[triggerKey] && (
   487→                                <p className="text-[10px] text-muted-foreground max-w-[120px] truncate">
   488→                                  {triggerMessage[triggerKey]}
   489→                                </p>
   490→                              )}
   491→                            </div>
   492→                          );
   493→                        })}
   494→                      </div>
   495→                    )}
   496→                  </div>
   497→                )}
   498→              </CardContent>
   499→            </Card>
   500→          );
   501→        })}
   502→      </div>
   503→
   504→      {/* Create Dialog */}
   505→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   506→        <DialogContent className="sm:max-w-md">
   507→          <DialogHeader>
   508→            <DialogTitle>Nouvelle automatisation</DialogTitle>
   509→            <DialogDescription>
   510→              Connectez Home Assistant ou Jeedom à votre maison
   511→            </DialogDescription>
   512→          </DialogHeader>
   513→
   514→          <div className="space-y-4 py-2">
   515→            {/* Home selector */}
   516→            <div className="space-y-2">
   517→              <Label>Maison</Label>
   518→              <Select value={selectedHomeId} onValueChange={setSelectedHomeId}>
   519→                <SelectTrigger>
   520→                  <SelectValue placeholder="Sélectionner une maison" />
   521→                </SelectTrigger>
   522→                <SelectContent>
   523→                  {homes.map((home) => (
   524→                    <SelectItem key={home.id} value={home.id}>
   525→                      {home.name}
   526→                    </SelectItem>
   527→                  ))}
   528→                </SelectContent>
   529→              </Select>
   530→            </div>
   531→
   532→            {/* Provider selector with logos */}
   533→            <div className="space-y-2">
   534→              <Label>Fournisseur</Label>
   535→              <div className="grid grid-cols-2 gap-3">
   536→                <button
   537→                  type="button"
   538→                  onClick={() => setNewProvider('home_assistant')}
   539→                  className={`flex items-center gap-3 rounded-lg border-2 p-3 transition-all ${
   540→                    newProvider === 'home_assistant'
   541→                      ? 'border-sky-500 bg-sky-50 dark:bg-sky-500/10'
   542→                      : 'border-muted hover:border-muted-foreground/30'
   543→                  }`}
   544→                >
   545→                  <HomeAssistantLogo className={`h-8 w-8 ${
   546→                    newProvider === 'home_assistant' ? 'text-sky-600' : 'text-muted-foreground'
   547→                  }`} />
   548→                  <div className="text-left">
   549→                    <p className="text-sm font-medium">Home Assistant</p>
   550→                    <p className="text-[11px] text-muted-foreground">HA</p>
   551→                  </div>
   552→                </button>
   553→                <button
   554→                  type="button"
   555→                  onClick={() => setNewProvider('jeedom')}
   556→                  className={`flex items-center gap-3 rounded-lg border-2 p-3 transition-all ${
   557→                    newProvider === 'jeedom'
   558→                      ? 'border-amber-500 bg-amber-50 dark:bg-amber-500/10'
   559→                      : 'border-muted hover:border-muted-foreground/30'
   560→                  }`}
   561→                >
   562→                  <JeedomLogo className={`h-8 w-8 ${
   563→                    newProvider === 'jeedom' ? 'text-amber-600' : 'text-muted-foreground'
   564→                  }`} />
   565→                  <div className="text-left">

   633→              Créer

   635→          </DialogFooter>
