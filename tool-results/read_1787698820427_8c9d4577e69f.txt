   484→  // ---- Batch validation ----
   485→  async function handleBatchValidate() {
   486→    const raw = batchCodes.trim();
   487→    if (!raw) return;
   488→    const codes = raw
   489→      .split(/[\n,]/)
   490→      .map((c) => c.trim().toUpperCase().replace(/[^A-Z0-9]/g, ''))
   491→      .filter((c) => c.length > 0);
   492→
   493→    setBatchValidating(true);
   494→    setBatchValidCodes([]);
   495→    setBatchInvalidCodes([]);
   496→
   497→    const valid: string[] = [];
   498→    const invalid: string[] = [];
   499→
   500→    // Check each code sequentially
   501→    for (const code of codes) {
   502→      try {
   503→        const res = await fetch(`/api/client/check-code?code=${encodeURIComponent(code)}`);
   504→        const data: CheckCodeResponse = await res.json();
   505→        if (data.valid) {
   506→          valid.push(code);
   507→        } else {
   508→          invalid.push(code);
   509→        }
   510→      } catch {
   511→        invalid.push(code);
   512→      }
   513→    }
   514→
   515→    setBatchValidCodes(valid);
   516→    setBatchInvalidCodes(invalid);
   517→    setBatchValidated(true);
   518→    setBatchValidating(false);
   519→  }
