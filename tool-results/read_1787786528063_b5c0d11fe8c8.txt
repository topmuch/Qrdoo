   116→// ── Voice Player Component ──
   117→function VoicePlayer({ msg }: { msg: VoiceMsg }) {
   118→  const audioRef = useRef<HTMLAudioElement>(null);
   119→  const [playing, setPlaying] = useState(false);
   120→  const [progress, setProgress] = useState(0);
   121→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   122→
   123→  const togglePlay = () => {
   124→    if (!audioRef.current) return;
   125→    if (playing) {
   126→      audioRef.current.pause();
   127→      if (intervalRef.current != null) clearInterval(intervalRef.current);
   128→    } else {
   129→      audioRef.current.play().catch(() => {});
   130→      intervalRef.current = setInterval(() => {
   131→        if (audioRef.current) {
   132→          const pct = audioRef.current.duration ? (audioRef.current.currentTime / audioRef.current.duration) * 100 : 0;
   133→          setProgress(pct);
   134→        }
   135→      }, 200);
   136→    }
   137→    setPlaying(!playing);
   138→  };
   139→
   140→  useEffect(() => {
   141→    const onEnded = () => {
   142→      setPlaying(false);
   143→      setProgress(0);
   144→      if (intervalRef.current != null) clearInterval(intervalRef.current);
   145→    };
   146→    const el = audioRef.current;
   147→    el?.addEventListener('ended', onEnded);
   148→    return () => { el?.removeEventListener('ended', onEnded); if (intervalRef.current != null) clearInterval(intervalRef.current); };
   149→  }, []);
   150→
   151→  return (
   152→    <div className="flex items-center gap-3 rounded-2xl bg-white/10 backdrop-blur-xl border border-white/20 px-4 py-3">
   153→      <audio ref={audioRef} src={msg.audioUrl} preload="metadata" />
   154→      <button onClick={togglePlay} className="h-10 w-10 rounded-full bg-white/15 flex items-center justify-center shrink-0">
   155→        {playing ? <Pause className="h-4 w-4 text-white" /> : <Play className="h-4 w-4 text-white ml-0.5" />}
   156→      </button>
   157→      <div className="flex-1 min-w-0">
   158→        <div className="flex items-center justify-between mb-1.5">
   159→          <p className="text-sm font-medium text-white truncate">{msg.senderName}</p>
   160→          <span className="text-[10px] text-white/40 shrink-0 ml-2">{msg.durationSec}s</span>
   161→        </div>
   162→        <div className="h-1.5 rounded-full bg-white/10 overflow-hidden">
   163→          <motion.div
   164→            className="h-full rounded-full bg-white/60"
   165→            animate={{ width: `${progress}%` }}
   166→            transition={{ duration: 0.2 }}
   167→          />
   168→        </div>
   169→      </div>
   170→      <span className="text-[10px] text-white/30 shrink-0">
   171→        {new Date(msg.createdAt).toLocaleDateString('fr-FR', { day: 'numeric', month: 'short' })}
   172→      </span>
   173→    </div>
   174→  );
   175→}

   177→// ── Voice Recorder Component ──
   178→function VoiceRecorder({ slug, onSent }: { slug: string; onSent: () => void }) {
   179→  const [recording, setRecording] = useState(false);
   180→  const [duration, setDuration] = useState(0);
   181→  const [senderName, setSenderName] = useState('');
   182→  const [showNameInput, setShowNameInput] = useState(false);
   183→  const [uploading, setUploading] = useState(false);
   184→  const mediaRecorderRef = useRef<MediaRecorder | null>(null);
   185→  const chunksRef = useRef<Blob[]>([]);
   186→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   187→  const startTimeRef = useRef<number>(0);
   188→
   189→  const startRecording = async () => {
   190→    try {
   191→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
   192→      const mr = new MediaRecorder(stream, { mimeType: 'audio/webm;codecs=opus' });
   193→      chunksRef.current = [];
   194→      mr.ondataavailable = (e) => { if (e.data.size > 0) chunksRef.current.push(e.data); };
   195→      mr.onstop = () => {
   196→        stream.getTracks().forEach((t) => t.stop());
   197→        if (chunksRef.current.length > 0) setShowNameInput(true);
   198→        else toast.error('Enregistrement vide');
   199→      };
   200→      mr.start();
   201→      mediaRecorderRef.current = mr;
   202→      startTimeRef.current = Date.now();
   203→      setDuration(0);
   204→      setRecording(true);
   205→      timerRef.current = setInterval(() => {
   206→        const elapsed = Math.floor((Date.now() - startTimeRef.current) / 1000);
   207→        setDuration(elapsed);
   208→        if (elapsed >= 30) stopRecording();
   209→      }, 200);
   210→    } catch {
   211→      toast.error('Accès au micro refusé');
   212→    }
   213→  };
   214→
   215→  const stopRecording = () => {
   216→    if (mediaRecorderRef.current?.state === 'recording') mediaRecorderRef.current.stop();
   217→    if (timerRef.current != null) clearInterval(timerRef.current);
   218→    setRecording(false);
   219→  };
   220→
   221→  const cancelRecording = () => {
   222→    if (mediaRecorderRef.current?.state === 'recording') {
   223→      mediaRecorderRef.current.onstop = null;
   224→      mediaRecorderRef.current.stop();
   225→    }
   226→    if (timerRef.current != null) clearInterval(timerRef.current);
   227→    chunksRef.current = [];
   228→    setRecording(false);
   229→    setDuration(0);
   230→    setShowNameInput(false);
   231→  };
   232→
   233→  const handleSend = async () => {
   234→    if (chunksRef.current.length === 0) return;
   235→    setUploading(true);
   236→    try {
   237→      const blob = new Blob(chunksRef.current, { type: 'audio/webm' });
   238→      const formData = new FormData();
   239→      formData.append('audio', blob, 'voice.webm');
   240→      formData.append('senderName', senderName.trim() || 'Invité');
   241→      formData.append('durationSec', String(duration));
   242→      const res = await fetch(`/api/public/hub/${encodeURIComponent(slug)}/voice`, { method: 'POST', body: formData });
   243→      const json = await res.json();
   244→      if (!res.ok) throw new Error(json.error || 'Erreur');
   245→      toast.success('Message vocal envoyé !');
   246→      setShowNameInput(false); setDuration(0); setSenderName(''); chunksRef.current = [];
   247→      onSent();
   248→    } catch (e: unknown) {
   249→      const msg = e instanceof Error ? e.message : "Erreur d'envoi";
   250→      toast.error(msg);
   251→    } finally {
   252→      setUploading(false);
   253→    }
   254→  };
   255→
   256→  useEffect(() => {
   257→    return () => { if (timerRef.current != null) clearInterval(timerRef.current); };
   258→  }, []);
   259→
   260→  return (
   261→    <div className="space-y-3">
   262→      {recording && (
   263→        <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }}
   264→          className="rounded-2xl bg-red-500/15 border border-red-500/30 p-4 space-y-3">
   265→          <div className="flex items-center justify-between">
   266→            <div className="flex items-center gap-2">
   267→              <div className="h-3 w-3 rounded-full bg-red-500 animate-pulse" />
   268→              <span className="text-sm font-semibold text-red-300">Enregistrement en cours</span>
   269→            </div>
   270→            <span className="text-sm font-mono text-red-300">
   271→              {String(Math.floor(duration / 60)).padStart(2, '0')}:{String(duration % 60).padStart(2, '0')}
   272→            </span>
   273→          </div>
   274→          <div className="h-1 rounded-full bg-white/10 overflow-hidden">
   275→            <motion.div className="h-full rounded-full bg-red-500" animate={{ width: `${Math.min((duration / 30) * 100, 100)}%` }} />
   276→          </div>
   277→          <div className="flex gap-2">
   278→            <button onClick={cancelRecording} className="flex-1 h-10 rounded-xl bg-white/10 text-sm text-white/70">Annuler</button>
   279→            <button onClick={stopRecording} className="flex-1 h-10 rounded-xl bg-red-600 text-sm text-white font-semibold">Arrêter</button>
   280→          </div>
   281→        </motion.div>
   282→      )}
   283→
   284→      <AnimatePresence>
   285→        {showNameInput && !recording && (
   286→          <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0 }}
   287→            className="rounded-2xl bg-white/10 backdrop-blur-xl border border-white/20 p-4 space-y-3">
   288→            <p className="text-sm font-semibold text-white">Votre message ({duration}s)</p>
   289→            <input
   290→              type="text" placeholder="Votre nom (optionnel)" maxLength={50}
   291→              value={senderName} onChange={(e) => setSenderName(e.target.value)}
   292→              className="w-full h-10 rounded-xl bg-white/10 border border-white/20 px-3 text-sm text-white placeholder:text-white/40 focus:outline-none focus:border-white/40 transition-all"
   293→              autoFocus
   294→            />
   295→            <div className="flex gap-2">
   296→              <button onClick={() => { setShowNameInput(false); chunksRef.current = []; }}
   297→                className="flex-1 h-10 rounded-xl bg-white/10 text-sm text-white/70">Annuler</button>
   298→              <button onClick={handleSend} disabled={uploading}
   299→                className="flex-1 h-10 rounded-xl bg-white text-emerald-700 text-sm font-bold flex items-center justify-center gap-1.5 disabled:opacity-50">
   300→                {uploading ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
   301→                {uploading ? 'Envoi...' : 'Envoyer'}
   302→              </button>
   303→            </div>
   304→          </motion.div>
   305→        )}
   306→      </AnimatePresence>
   307→
   308→      {!recording && !showNameInput && (
   309→        <motion.button whileTap={{ scale: 0.97 }} onClick={startRecording}
   310→          className="w-full rounded-2xl bg-white/10 backdrop-blur-xl border border-white/20 p-4 flex items-center justify-center gap-3 hover:bg-white/15 transition-all">
   311→          <div className="h-12 w-12 rounded-full bg-gradient-to-br from-emerald-400 to-teal-500 flex items-center justify-center shadow-lg">
   312→            <Mic className="h-5 w-5 text-white" />
   313→          </div>
   314→          <div className="text-left">
   315→            <p className="text-sm font-semibold text-white">Laisser un message vocal</p>
   316→            <p className="text-xs text-white/40">Appuyez pour enregistrer (max 30s)</p>
   317→          </div>
   318→        </motion.button>
   319→      )}
   320→    </div>
   321→  );
   322→}

   388→// ── WiFi Card (Enhanced with copy) ──
   389→function WifiQuickCard({ content }: { content: Record<string, unknown> }) {
   390→  const [copied, setCopied] = useState(false);
   391→  const [showPw, setShowPw] = useState(false);
   392→  const networkName = (content.network_name as string) || 'Non configuré';
   393→  const password = (content.password as string) || '';
   394→  const securityType = (content.security_type as string) || '';
   395→
   396→  const copyPassword = async () => {
   397→    if (!password) return;
   398→    try {
   399→      await navigator.clipboard.writeText(password);
   400→      setCopied(true);
   401→      toast.success('Mot de passe copié !');
   402→      setTimeout(() => setCopied(false), 2000);
   403→    } catch {
   404→      toast.error('Impossible de copier');
   405→    }
   406→  };
   407→
   408→  return (
   409→    <motion.div
   410→      variants={itemVariants}
   411→      className="relative overflow-hidden rounded-2xl bg-white/10 backdrop-blur-xl border border-emerald-400/30 p-5"
   412→    >
   413→      {/* Decorative glow */}
   414→      <div className="absolute -top-8 -right-8 h-24 w-24 rounded-full bg-emerald-400/20 blur-2xl" />
   415→
   416→      <div className="relative z-10">
   417→        <div className="flex items-start gap-3.5">
   418→          <div className="h-11 w-11 rounded-xl bg-gradient-to-br from-emerald-400 to-teal-500 flex items-center justify-center shrink-0 shadow-lg">
   419→            <Wifi className="h-5 w-5 text-white" />
   420→          </div>
   421→          <div className="flex-1 min-w-0">
   422→            <p className="text-[11px] font-semibold text-emerald-300 uppercase tracking-wider">Wi-Fi</p>
   423→            <p className="text-base font-bold text-white truncate mt-0.5">{networkName}</p>
   424→            {securityType && (
   425→              <p className="text-[10px] text-white/40 mt-0.5">{securityType}</p>
   426→            )}
   427→          </div>
   428→        </div>
   429→
   430→        {password && (
   431→          <div className="mt-3 flex items-center gap-2">
   432→            <div className="flex-1 h-9 rounded-lg bg-white/10 border border-white/15 flex items-center px-3 gap-2">
   433→              <span className="text-sm font-mono text-white/80 flex-1 truncate">
   434→                {showPw ? password : '•'.repeat(Math.min(password.length, 16))}
   435→              </span>
   436→              <button
   437→                onClick={() => setShowPw(s => !s)}
   438→                className="p-1 text-white/40 hover:text-white/70 transition-colors"
   439→              >
   440→                {showPw ? <EyeOff className="h-3.5 w-3.5" /> : <Eye className="h-3.5 w-3.5" />}
   441→              </button>
   442→            </div>
   443→            <motion.button
   444→              whileTap={{ scale: 0.92 }}
   445→              onClick={copyPassword}
   446→              className="h-9 px-3 rounded-lg bg-emerald-500/20 border border-emerald-400/30 text-emerald-300 text-xs font-semibold flex items-center gap-1.5 hover:bg-emerald-500/30 transition-colors"
   447→            >
   448→              {copied ? <Check className="h-3.5 w-3.5" /> : <Copy className="h-3.5 w-3.5" />}
   449→              {copied ? 'Copié' : 'Copier'}
   450→            </motion.button>
   451→          </div>
   452→        )}
   453→      </div>
   454→    </motion.div>
   455→  );
   456→}
