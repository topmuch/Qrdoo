    94→  const { token } = use(params);
    95→  const { data: session } = useSession();

    97→
    98→  // Token state
    99→  const [tokenStatus, setTokenStatus] = useState<TokenStatus>('loading');
   100→  const [plaqueInfo, setPlaqueInfo] = useState<PlaqueInfo | null>(null);
   101→  const [claimedInfo, setClaimedInfo] = useState<{ hubSlug?: string; homeName?: string }>({});
   102→
   103→  // Step navigation
   104→  const [step, setStep] = useState<SetupStep>('welcome');
   105→  const [direction, setDirection] = useState(1);
   106→
   107→  // Account
   108→  const [fullName, setFullName] = useState('');
   109→  const [email, setEmail] = useState('');
   110→  const [password, setPassword] = useState('');
   111→  const [showPassword, setShowPassword] = useState(false);
   112→
   113→  // PIN
   114→  const [pinKey, setPinKey] = useState(0);

   116→
   117→  // Config

   119→  const [homeName, setHomeName] = useState('');
   120→  const [wifiSsid, setWifiSsid] = useState('');

   123→

   130→    try {
   131→      const res = await fetch(`/api/setup/${encodeURIComponent(token)}`);
   132→      const data = await res.json();

   134→        setTokenStatus('available');
   135→        setPlaqueInfo(data.plaque);
   136→      } else if (data.status === 'claimed') {
   137→        setTokenStatus('claimed');
   138→        setClaimedInfo({ hubSlug: data.hubSlug, homeName: data.homeName });
   139→      } else {
   140→        setTokenStatus('not_found');
   141→      }
   142→    } catch {
   143→      setTokenStatus('error');
   144→    }
   145→  }, [token]);
   146→
   147→  useEffect(() => { checkToken(); }, [checkToken]);
   148→
   149→  // Pre-fill if logged in
   150→  useEffect(() => {
   151→    if (session?.user) {
   152→      const u = session.user as Record<string, unknown>;
   153→      setEmail((u?.email as string) || '');
   154→      setFullName((u?.name as string) || '');
   155→    }
   156→  }, [session]);
   157→

   159→  const stepOrder: SetupStep[] = ['welcome', 'account', 'pin', 'config', 'success'];

   162→    setDirection(1);
   163→    const idx = stepOrder.indexOf(step);
   164→    if (idx < stepOrder.length - 1) setStep(stepOrder[idx + 1]);
   165→  };

   167→  const goBack = () => {
   168→    setDirection(-1);
   169→    const idx = stepOrder.indexOf(step);
   170→    if (idx > 0) setStep(stepOrder[idx - 1]);
   171→  };

   174→  const handleWelcomeNext = () => goNext();

   178→    if (!email.trim() || !email.includes('@')) { toast.error('Email invalide'); return; }

   180→      toast.error('Mot de passe requis (6+ caractères)'); return;
   181→    }

   183→  };

   192→    if (!homeName.trim()) { toast.error('Nommez votre logement'); return; }

   197→      const body: Record<string, string> = {
   198→        email: email.trim(),
   199→        fullName: fullName.trim(),
   200→        pin: pinValue,
   201→        homeName: homeName.trim(),
   202→        plan: selectedPlan,
   203→      };

   205→      if (!session?.user) {
   206→        body.password = password;
   207→      } else {
   208→        body.existingUserId = (session.user as Record<string, unknown>)?.id as string;
   209→      }

   212→        method: 'POST',
   213→        headers: { 'Content-Type': 'application/json' },
   214→        body: JSON.stringify(body),
   215→      });
   216→      const data = await res.json();
   217→
   218→      if (!res.ok) throw new Error(data.error || 'Erreur lors de la configuration');

   230→              headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
   231→              body: new URLSearchParams({
   232→                email: email.trim(),
   233→                password,
   234→                callbackUrl: '/',
   235→              }).toString(),
   236→            });

   239→        }, 3000);

   243→        }, 3000);
   244→      }
   245→    } catch (e: unknown) {
   246→      const msg = e instanceof Error ? e.message : 'Erreur';
   247→      toast.error(msg);

   249→      setSubmitting(false);

   254→  const currentIdx = stepOrder.indexOf(step);

   259→      <AnimatedGradient preset="setup">

   261→          <motion.div
   262→            initial={{ opacity: 0, scale: 0.8 }}

   264→            className="flex flex-col items-center gap-4"

   266→            <div className="relative">
   267→              <div className="h-16 w-16 rounded-2xl bg-white/20 backdrop-blur-xl border border-white/30 flex items-center justify-center">

   271→                <div className="h-2 w-2 rounded-full bg-white animate-pulse" />
   272→              </div>
   273→            </div>
   274→            <p className="text-white/70 font-medium">Vérification de votre plaque...</p>

   276→          </motion.div>
   277→        </div>
   278→      </AnimatedGradient>
   279→    );
   280→  }
   281→
   282→  // ── Error states ──
   283→  if (tokenStatus === 'not_found' || tokenStatus === 'error') {
   284→    const Icon = tokenStatus === 'error' ? AlertCircle : XCircle;
   285→    const msg = tokenStatus === 'not_found'
   286→      ? { title: 'Plaque non trouvée', desc: "Ce code d'activation n'existe pas. Vérifiez votre plaque et réessayez." }
   287→      : { title: 'Erreur', desc: 'Impossible de vérifier la plaque. Réessayez.' };
   288→    return (
   289→      <AnimatedGradient preset="setup">
   290→        <div className="min-h-screen flex items-center justify-center p-4">
   291→          <GlassCard className="w-full max-w-sm p-8 text-center" hover={false}>
   292→            <div className="mx-auto mb-4 flex h-16 w-16 items-center justify-center rounded-full bg-red-500/20">
   293→              <Icon className="h-8 w-8 text-red-400" />
   294→            </div>
   295→            <h1 className="text-xl font-bold text-white mb-2">{msg.title}</h1>

   297→            <p className="text-sm text-white/60 mb-6">{msg.desc}</p>

   299→              whileTap={{ scale: 0.97 }}
   300→              onClick={() => { setTokenStatus('loading'); checkToken(); }}

   302→            >
   303→              Réessayer

   315→        <div className="min-h-screen flex items-center justify-center p-4">

   324→              </p>

   327→            {claimedInfo.hubSlug && (
   328→              <motion.button
   329→                whileTap={{ scale: 0.97 }}
   330→                onClick={() => router.push(`/hub/${claimedInfo.hubSlug}`)}
   331→                className="w-full h-12 rounded-2xl bg-white text-violet-700 font-bold text-sm shadow-xl"
   332→              >
   333→                Accéder au Hub
   334→              </motion.button>
   335→            )}

   337→              whileTap={{ scale: 0.97 }}
   338→              onClick={() => router.push('/')}
   339→              className="w-full h-11 mt-3 rounded-2xl bg-white/10 border border-white/20 text-white/70 text-sm font-medium hover:bg-white/20 transition-all"
   340→            >
   341→              Aller à l'accueil
   342→            </motion.button>
   343→          </GlassCard>
   344→        </div>
   345→      </AnimatedGradient>

   351→    <AnimatedGradient preset="setup">
   352→      <FloatingParticles count={20} color="rgba(255,255,255,0.15)" size={3} duration={22} />
   353→
   354→      <div className="min-h-screen flex flex-col">
   355→        {/* Header with progress */}
   356→        <div className="w-full px-6 pt-6 pb-2">
   357→          <div className="max-w-md mx-auto">
   358→            <div className="flex items-center justify-between mb-4">
   359→              <div className="flex items-center gap-2.5">
   360→                <div className="h-8 w-8 rounded-lg bg-white/20 backdrop-blur-xl border border-white/30 flex items-center justify-center">
   361→                  <QrCode className="h-4 w-4 text-white" />
   362→                </div>
   363→                <span className="text-sm font-bold text-white tracking-tight">QR Domotik</span>
   364→              </div>

   367→              </span>

   369→            {/* Progress bar */}
   370→            {step !== 'success' && (
   371→              <div className="h-1 rounded-full bg-white/10 overflow-hidden">

   407→                    <Home className="w-12 h-12 text-white" />

   428→                  {/* Feature pills */}

   452→                  {/* Plaque badge */}
