   447→  // ---- Batch validation ----
   448→  async function handleBatchValidate() {
   449→    const raw = batchCodes.trim();
   450→    if (!raw) return;
   451→    const codes = raw
   452→      .split(/[\n,]/)
   453→      .map((c) => c.trim().toUpperCase().replace(/[^A-Z0-9]/g, ''))
   454→      .filter((c) => c.length > 0);
   455→
   456→    setBatchValidating(true);
   457→    setBatchValidCodes([]);
   458→    setBatchInvalidCodes([]);
   459→
   460→    const valid: string[] = [];
   461→    const invalid: string[] = [];
   462→
   463→    // Check each code sequentially
   464→    for (const code of codes) {
   465→      try {
   466→        const res = await fetch(`/api/client/check-code?code=${encodeURIComponent(code)}`);
   467→        const data: CheckCodeResponse = await res.json();
   468→        if (data.valid) {
   469→          valid.push(code);
   470→        } else {
   471→          invalid.push(code);
   472→        }
   473→      } catch {
   474→        invalid.push(code);
   475→      }
   476→    }
   477→
   478→    setBatchValidCodes(valid);
   479→    setBatchInvalidCodes(invalid);
   480→    setBatchValidated(true);
   481→    setBatchValidating(false);
   482→  }
