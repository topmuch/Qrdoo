   408→  // ---- Batch validation ----
   409→  async function handleBatchValidate() {
   410→    const raw = batchCodes.trim();
   411→    if (!raw) return;
   412→    const codes = raw
   413→      .split(/[\n,]/)
   414→      .map((c) => c.trim().toUpperCase().replace(/[^A-Z0-9]/g, ''))
   415→      .filter((c) => c.length > 0);
   416→
   417→    setBatchValidating(true);
   418→    setBatchValidCodes([]);
   419→    setBatchInvalidCodes([]);
   420→
   421→    const valid: string[] = [];
   422→    const invalid: string[] = [];
   423→
   424→    // Check each code sequentially
   425→    for (const code of codes) {
   426→      try {
   427→        const res = await fetch(`/api/client/check-code?code=${encodeURIComponent(code)}`);
   428→        const data: CheckCodeResponse = await res.json();
   429→        if (data.valid) {
   430→          valid.push(code);
   431→        } else {
   432→          invalid.push(code);
   433→        }
   434→      } catch {
   435→        invalid.push(code);
   436→      }
   437→    }
   438→
   439→    setBatchValidCodes(valid);
   440→    setBatchInvalidCodes(invalid);
   441→    setBatchValidated(true);
   442→    setBatchValidating(false);
   443→  }
