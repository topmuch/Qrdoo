   101→// ── Voice Player Component ──
   102→function VoicePlayer({ msg }: { msg: VoiceMsg }) {
   103→  const audioRef = useRef<HTMLAudioElement>(null);
   104→  const [playing, setPlaying] = useState(false);
   105→  const [progress, setProgress] = useState(0);
   106→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   107→
   108→  const togglePlay = () => {
   109→    if (!audioRef.current) return;
   110→    if (playing) {
   111→      audioRef.current.pause();
   112→      if (intervalRef.current != null) clearInterval(intervalRef.current);
   113→    } else {
   114→      audioRef.current.play().catch(() => {});
   115→      intervalRef.current = setInterval(() => {
   116→        if (audioRef.current) {
   117→          const pct = audioRef.current.duration ? (audioRef.current.currentTime / audioRef.current.duration) * 100 : 0;
   118→          setProgress(pct);
   119→        }
   120→      }, 200);
   121→    }
   122→    setPlaying(!playing);
   123→  };
   124→
   125→  useEffect(() => {
   126→    const onEnded = () => {
   127→      setPlaying(false);
   128→      setProgress(0);
   129→      if (intervalRef.current != null) clearInterval(intervalRef.current);
   130→    };
   131→    const el = audioRef.current;
   132→    el?.addEventListener('ended', onEnded);
   133→    return () => { el?.removeEventListener('ended', onEnded); if (intervalRef.current != null) clearInterval(intervalRef.current); };
   134→  }, []);
   135→
   136→  return (
   137→    <div className="flex items-center gap-3 rounded-2xl bg-white/10 backdrop-blur-xl border border-white/20 px-4 py-3">
   138→      <audio ref={audioRef} src={msg.audioUrl} preload="metadata" />
   139→      <button onClick={togglePlay} className="h-9 w-9 rounded-full bg-white/15 flex items-center justify-center shrink-0">
   140→        {playing ? <Pause className="h-4 w-4 text-white" /> : <Play className="h-4 w-4 text-white ml-0.5" />}
   141→      </button>
   142→      <div className="flex-1 min-w-0">
   143→        <div className="flex items-center justify-between mb-1">
   144→          <p className="text-sm font-medium text-white truncate">{msg.senderName}</p>
   145→          <span className="text-[10px] text-white/40 shrink-0 ml-2">{msg.durationSec}s</span>
   146→        </div>
   147→        <div className="h-1.5 rounded-full bg-white/10 overflow-hidden">
   148→          <div className="h-full rounded-full bg-white/60 transition-all duration-200" style={{ width: `${progress}%` }} />
   149→        </div>
   150→      </div>
   151→      <span className="text-[10px] text-white/30 shrink-0">
   152→        {new Date(msg.createdAt).toLocaleDateString('fr-FR', { day: 'numeric', month: 'short' })}
   153→      </span>
   154→    </div>
   155→  );
   156→}

   158→// ── Voice Recorder Component ──
   159→function VoiceRecorder({ slug, onSent }: { slug: string; onSent: () => void }) {
   160→  const [recording, setRecording] = useState(false);
   161→  const [duration, setDuration] = useState(0);
   162→  const [senderName, setSenderName] = useState('');
   163→  const [showNameInput, setShowNameInput] = useState(false);
   164→  const [uploading, setUploading] = useState(false);
   165→  const mediaRecorderRef = useRef<MediaRecorder | null>(null);
   166→  const chunksRef = useRef<Blob[]>([]);
   167→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   168→  const startTimeRef = useRef<number>(0);
   169→
   170→  const startRecording = async () => {
   171→    try {
   172→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
   173→      const mr = new MediaRecorder(stream, { mimeType: 'audio/webm;codecs=opus' });
   174→      chunksRef.current = [];
   175→      mr.ondataavailable = (e) => { if (e.data.size > 0) chunksRef.current.push(e.data); };
   176→      mr.onstop = () => {
   177→        stream.getTracks().forEach((t) => t.stop());
   178→        if (chunksRef.current.length > 0) setShowNameInput(true);
   179→        else toast.error('Enregistrement vide');
   180→      };
   181→      mr.start();
   182→      mediaRecorderRef.current = mr;
   183→      startTimeRef.current = Date.now();
   184→      setDuration(0);
   185→      setRecording(true);
   186→      timerRef.current = setInterval(() => {
   187→        const elapsed = Math.floor((Date.now() - startTimeRef.current) / 1000);
   188→        setDuration(elapsed);
   189→        if (elapsed >= 30) stopRecording();
   190→      }, 200);
   191→    } catch {
   192→      toast.error('Accès au micro refusé');
   193→    }
   194→  };
   195→
   196→  const stopRecording = () => {
   197→    if (mediaRecorderRef.current?.state === 'recording') mediaRecorderRef.current.stop();
   198→    if (timerRef.current != null) clearInterval(timerRef.current);
   199→    setRecording(false);
   200→  };
   201→
   202→  const cancelRecording = () => {
   203→    if (mediaRecorderRef.current?.state === 'recording') {
   204→      mediaRecorderRef.current.onstop = null;
   205→      mediaRecorderRef.current.stop();
   206→    }
   207→    if (timerRef.current != null) clearInterval(timerRef.current);
   208→    chunksRef.current = [];
   209→    setRecording(false);
   210→    setDuration(0);
   211→    setShowNameInput(false);
   212→  };
   213→
   214→  const handleSend = async () => {
   215→    if (chunksRef.current.length === 0) return;
   216→    setUploading(true);
   217→    try {
   218→      const blob = new Blob(chunksRef.current, { type: 'audio/webm' });
   219→      const formData = new FormData();
   220→      formData.append('audio', blob, 'voice.webm');
   221→      formData.append('senderName', senderName.trim() || 'Invité');
   222→      formData.append('durationSec', String(duration));
   223→      const res = await fetch(`/api/public/hub/${encodeURIComponent(slug)}/voice`, { method: 'POST', body: formData });
   224→      const json = await res.json();
   225→      if (!res.ok) throw new Error(json.error || 'Erreur');
   226→      toast.success('Message vocal envoyé !');
   227→      setShowNameInput(false); setDuration(0); setSenderName(''); chunksRef.current = [];
   228→      onSent();
   229→    } catch (e: unknown) {
   230→      const msg = e instanceof Error ? e.message : "Erreur d'envoi";
   231→      toast.error(msg);
   232→    } finally {
   233→      setUploading(false);
   234→    }
   235→  };
   236→
   237→  useEffect(() => {
   238→    return () => { if (timerRef.current != null) clearInterval(timerRef.current); };
   239→  }, []);
   240→
   241→  return (
   242→    <div className="space-y-3">
   243→      {recording && (
   244→        <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }}
   245→          className="rounded-2xl bg-red-500/10 border border-red-500/30 p-4 space-y-3">
   246→          <div className="flex items-center justify-between">
   247→            <div className="flex items-center gap-2">
   248→              <div className="h-3 w-3 rounded-full bg-red-500 animate-pulse" />
   249→              <span className="text-sm font-semibold text-red-300">Enregistrement en cours</span>
   250→            </div>
   251→            <span className="text-sm font-mono text-red-300">{String(Math.floor(duration / 60)).padStart(2, '0')}:{String(duration % 60).padStart(2, '0')}</span>
   252→          </div>
   253→          <div className="h-1 rounded-full bg-white/10 overflow-hidden">
   254→            <div className="h-full rounded-full bg-red-500 transition-all" style={{ width: `${Math.min((duration / 30) * 100, 100)}%` }} />
   255→          </div>
   256→          <div className="flex gap-2">
   257→            <button onClick={cancelRecording} className="flex-1 h-10 rounded-xl bg-white/10 text-sm text-white/70">Annuler</button>
   258→            <button onClick={stopRecording} className="flex-1 h-10 rounded-xl bg-red-600 text-sm text-white font-semibold">Arrêter</button>
   259→          </div>
   260→        </motion.div>
   261→      )}
   262→
   263→      <AnimatePresence>
   264→        {showNameInput && !recording && (
   265→          <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0 }}
   266→            className="rounded-2xl bg-white/10 backdrop-blur-xl border border-white/20 p-4 space-y-3">
   267→            <p className="text-sm font-semibold text-white">Votre message ({duration}s)</p>
   268→            <input
   269→              type="text" placeholder="Votre nom (optionnel)" maxLength={50}
   270→              value={senderName} onChange={(e) => setSenderName(e.target.value)}
   271→              className="w-full h-10 rounded-xl bg-white/10 border border-white/20 px-3 text-sm text-white placeholder:text-white/40 focus:outline-none focus:border-white/40 transition-all"
   272→              autoFocus
   273→            />
   274→            <div className="flex gap-2">
   275→              <button onClick={() => { setShowNameInput(false); chunksRef.current = []; }}
   276→                className="flex-1 h-10 rounded-xl bg-white/10 text-sm text-white/70">Annuler</button>
   277→              <button onClick={handleSend} disabled={uploading}
   278→                className="flex-1 h-10 rounded-xl bg-white text-emerald-700 text-sm font-bold flex items-center justify-center gap-1.5 disabled:opacity-50">
   279→                {uploading ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
   280→                {uploading ? 'Envoi...' : 'Envoyer'}
   281→              </button>
   282→            </div>
   283→          </motion.div>
   284→        )}
   285→      </AnimatePresence>
   286→
   287→      {!recording && !showNameInput && (
   288→        <motion.button whileTap={{ scale: 0.95 }} onClick={startRecording}
   289→          className="w-full rounded-2xl bg-white/10 backdrop-blur-xl border border-white/20 p-4 flex items-center justify-center gap-3 hover:bg-white/20 transition-all">
   290→          <div className="h-12 w-12 rounded-full bg-gradient-to-br from-emerald-400 to-teal-500 flex items-center justify-center shadow-lg">
   291→            <Mic className="h-5 w-5 text-white" />
   292→          </div>
   293→          <div className="text-left">
   294→            <p className="text-sm font-semibold text-white">Laisser un message vocal</p>
   295→            <p className="text-xs text-white/40">Appuyez pour enregistrer (max 30s)</p>
   296→          </div>
   297→        </motion.button>
   298→      )}
   299→    </div>
   300→  );
   301→}

   364→export function HubPageContent({ params }: { params: Promise<{ slug: string }> }) {

   372→  const [voiceMsgs, setVoiceMsgs] = useState<VoiceMsg[]>([]);
   373→
   374→  const fetchHub = useCallback(async () => {
   375→    try {
   376→      const res = await fetch(`/api/public/hub/${encodeURIComponent(slug)}`);
   377→      const json = await res.json();
   378→      if (!res.ok) throw new Error(json.error || 'Hub non trouvé');
   379→      setData(json);
   380→      setVoiceMsgs(json.voiceMessages || []);
   381→    } catch (e: unknown) {
   382→      const msg = e instanceof Error ? e.message : 'Erreur';
   383→      setError(msg);
   384→    } finally {
   385→      setLoading(false);
   386→    }
   387→  }, [slug]);

   389→  const refreshVoice = useCallback(async () => {
   390→    try {
   391→      const res = await fetch(`/api/public/hub/${encodeURIComponent(slug)}/voice?limit=10`);
   392→      if (res.ok) { const json = await res.json(); setVoiceMsgs(json.messages || []); }
   393→    } catch { /* silent */ }
   394→  }, [slug]);
   395→
   396→  useEffect(() => { fetchHub(); }, [fetchHub]);

   416→        }
   417→      })
   418→      .catch(() => { toast.error('Erreur réseau'); setPinModalFor(null); })
   419→      .finally(() => setPinVerifying(false));

   433→            </div>

   437→      </AnimatedGradient>

   440→
   441→  // ── Error state ──
   442→  if (error || !data) {
   443→    return (
   444→      <AnimatedGradient preset="hub-guest">
   445→        <div className="min-h-screen flex items-center justify-center p-4">
   446→          <GlassCard className="w-full max-w-sm p-8 text-center" hover={false}>

   462→  const totalGuestModules = data.guestRooms.reduce((s, r) => s + r.qrCodes.length, 0);

   741→function QuickWifiCard({ rooms }: { rooms: RoomInfo[] }) {
   742→  const wifiQr = rooms.flatMap((r) => r.qrCodes).find((qr) => qr.type === 'wifi');
   743→  if (!wifiQr?.content) return null;
   744→  const { network_name, password, security_type } = wifiQr.content as Record<string, string>;
   745→  return (
   746→    <motion.div variants={itemVariants} className="rounded-2xl bg-white/10 backdrop-blur-xl border border-emerald-400/30 p-4">
   747→      <div className="flex items-center gap-3">
   748→        <div className="h-11 w-11 rounded-xl bg-gradient-to-br from-emerald-400 to-teal-500 flex items-center justify-center shrink-0">
   749→          <Wifi className="h-5 w-5 text-white" />
   750→        </div>
   751→        <div className="flex-1 min-w-0">
   752→          <p className="text-xs text-emerald-300 font-medium">Wi-Fi</p>
   753→          <p className="text-sm font-bold text-white truncate">{network_name || 'Non configuré'}</p>
   754→          {password && (
   755→            <div className="flex items-center gap-2 mt-1">
   756→              <code className="text-xs font-mono text-emerald-200 bg-white/10 px-2 py-0.5 rounded">{password}</code>
