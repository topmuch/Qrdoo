   116→/*  SVG → PNG conversion utility (browser-only, zero native deps)       */

   118→
   119→/** Convert an SVG Blob to a PNG data-URL using the browser Canvas API */
   120→function svgBlobToPngDataUrl(svgBlob: Blob, size: number): Promise<string> {
   121→  return new Promise((resolve, reject) => {
   122→    const url = URL.createObjectURL(svgBlob);
   123→    const img = new Image();
   124→    img.onload = () => {
   125→      const canvas = document.createElement('canvas');
   126→      canvas.width = size;
   127→      canvas.height = size;
   128→      const ctx = canvas.getContext('2d');
   129→      if (!ctx) { URL.revokeObjectURL(url); reject(new Error('No 2d context')); return; }
   130→      ctx.drawImage(img, 0, 0, size, size);
   131→      URL.revokeObjectURL(url);
   132→      resolve(canvas.toDataURL('image/png'));
   133→    };
   134→    img.onerror = () => { URL.revokeObjectURL(url); reject(new Error('SVG image load failed')); };
   135→    img.src = url;
   136→  });
   137→}
   138→
   139→/** Convert an inline SVG element to a PNG data-URL */
   140→function svgElementToPngDataUrl(svgEl: SVGSVGElement, size: number): Promise<string> {
   141→  return new Promise((resolve, reject) => {
   142→    const svgData = new XMLSerializer().serializeToString(svgEl);
   143→    const svgBlob = new Blob([svgData], { type: 'image/svg+xml;charset=utf-8' });
   144→    svgBlobToPngDataUrl(svgBlob, size).then(resolve).catch(reject);
   145→  });
   146→}

   148→/**
   149→ * Fallback QR PNG generator using qrcode.react (pure JS, no native deps).
   150→ * Renders QRCodeSVG into a hidden container, extracts the SVG, converts to PNG.
   151→ */
   152→async function generateFallbackQrPng(
   153→  data: string,
   154→  size: number,
   155→  fgColor: string,
   156→  bgColor: string,
   157→  level: string,
   158→): Promise<string> {
   159→  const container = document.createElement('div');
   160→  container.style.cssText = 'position:fixed;left:-9999px;top:-9999px;';
   161→  document.body.appendChild(container);
   162→
   163→  const root = createRoot(container);
   164→  root.render(
   165→    createElement(QRCodeSVG, {
   166→      value: data,
   167→      size,
   168→      bgColor,
   169→      fgColor,
   170→      level: level as 'L' | 'M' | 'Q' | 'H',
   171→    }),
   172→  );
   173→
   174→  // Wait for React to commit the render
   175→  await new Promise((r) => setTimeout(r, 60));
   176→
   177→  const svgEl = container.querySelector('svg');
   178→  if (!svgEl) {
   179→    root.unmount();
   180→    document.body.removeChild(container);
   181→    throw new Error('Fallback: SVG element not rendered');
   182→  }
   183→
   184→  const pngUrl = await svgElementToPngDataUrl(svgEl, size);
   185→  root.unmount();
   186→  document.body.removeChild(container);
   187→  return pngUrl;
   188→}

   211→export function GenerateBatch() {
   212→  /* ---- state ---- */
   213→  const [quantity, setQuantity] = useState(10);
   214→  const [design, setDesign] = useState<DesignConfig>(DEFAULT_DESIGN);
   215→  const [batchName, setBatchName] = useState('');
   216→  const [activeTemplate, setActiveTemplate] = useState<string | null>(null);
   217→  const [isGenerating, setIsGenerating] = useState(false);
   218→  const [isExporting, setIsExporting] = useState(false);
   219→  const [generatedCodes, setGeneratedCodes] = useState<string[]>([]);
   220→  const [batchId, setBatchId] = useState<string | null>(null);
   221→
   222→  /* ---- refs ---- */
   223→  const previewRef = useRef<HTMLDivElement>(null);
   224→  const qrInstanceRef = useRef<any>(null);
   225→  const fallbackRef = useRef(false);
   226→
   227→  /* ---- state ---- */
   228→  const [showFallback, setShowFallback] = useState(false);
   229→
   230→  /* ---- QR preview ---- */
   231→  const updateQrPreview = useCallback(async () => {
   232→    if (fallbackRef.current || !previewRef.current) return;
   233→    try {
   234→      const QRCodeStyling = (await import('qr-code-styling')).default;
   235→
   236→      const opts: any = {
   237→        width: 280,
   238→        height: 280,
   239→        type: 'svg',
   240→        data: 'https://qrdomotik.com/activate/QR-XXXXXXXX',
   241→        dotsOptions: {
   242→          color: design.dotsColor,
   243→          type: design.dotsType,
   244→        },
   245→        backgroundOptions: {
   246→          color: design.backgroundColor,
   247→        },
   248→        cornersSquareOptions: {
   249→          color: design.dotsColor,
   250→          type: design.cornersSquareType,
   251→        },
   252→        cornersDotOptions: {
   253→          color: design.dotsColor,
   254→          type: design.cornersDotType,
   255→        },
   256→        qrOptions: {
   257→          errorCorrectionLevel: design.errorCorrectionLevel,
   258→        },
   259→        imageOptions: {
   260→          crossOrigin: 'anonymous',
   261→          margin: 8,
   262→          imageSize: 0.35,
   263→        },
   264→      };
   265→
   266→      if (design.logoPreset) {
   267→        opts.image = getPresetLogoDataUrl(design.logoPreset);
   268→      }
   269→
   270→      if (qrInstanceRef.current) {
   271→        qrInstanceRef.current.update(opts);
   272→      } else {
   273→        previewRef.current.innerHTML = '';
   274→        const instance = new QRCodeStyling(opts);
   275→        instance.append(previewRef.current);
   276→        qrInstanceRef.current = instance;
   277→      }
   278→    } catch (err) {
   279→      console.error('QR preview error — switching to SVG fallback:', err);
   280→      fallbackRef.current = true;
   281→      setShowFallback(true);
   282→      if (previewRef.current) previewRef.current.innerHTML = '';
   283→      qrInstanceRef.current = null;
   284→    }
   285→  }, [design]);
   286→
   287→  useEffect(() => {
   288→    const timer = setTimeout(updateQrPreview, 100);
   289→    return () => clearTimeout(timer);
   290→  }, [updateQrPreview]);
   291→
   292→  /* ---- template ---- */
   293→  const applyTemplate = (key: string) => {
   294→    const tpl = BATCH_TEMPLATES[key];
   295→    if (!tpl) return;
   296→    setQuantity(tpl.quantity);
   297→    setDesign((prev) => ({ ...prev, ...tpl.design }));
   298→    setActiveTemplate(key);
   299→    qrInstanceRef.current = null;
   300→    toast.success(`Template « ${tpl.label} » appliqué`);
   301→  };
   302→
   303→  const resetTemplate = () => {
   304→    setActiveTemplate(null);
   305→    setQuantity(10);
   306→    setDesign(DEFAULT_DESIGN);
   307→    qrInstanceRef.current = null;
   308→  };
   309→
   310→  /* ---- generate batch ---- */
   311→  const handleGenerate = async () => {
   312→    setIsGenerating(true);
   313→    try {
   314→      const codes = generateUniqueCodes(quantity);
   315→      const designConfig = JSON.stringify(design);
   316→
   317→      const res = await fetch('/api/admin/batches', {
   318→        method: 'POST',
   319→        headers: { 'Content-Type': 'application/json' },
   320→        body: JSON.stringify({
   321→          quantity,
   322→          designConfig,
   323→          batchName: batchName || undefined,
   324→          activationCodes: codes,
   325→        }),
   326→      });
   327→
   328→      if (!res.ok) {
   329→        const err = await res.json().catch(() => ({}));
   330→        throw new Error(err.error || 'Erreur lors de la génération');
   331→      }
   332→
   333→      const data = await res.json();
   334→      setGeneratedCodes(codes);
   335→      setBatchId(data.batch?.id || null);
   336→      toast.success(`${quantity} QR codes générés avec succès !`);
   337→    } catch (err: any) {
   338→      toast.error(err.message || 'Erreur lors de la génération');
   339→    } finally {
   340→      setIsGenerating(false);
   341→    }
   342→  };
   343→
   344→  /* ---- export PDF ---- */
   345→  const handleExportPdf = async () => {
   346→    if (generatedCodes.length === 0) return;
   347→    setIsExporting(true);
   348→    try {
   349→      const qrCodes: QrCodeForPdf[] = [];
   350→
   351→      // Try loading qr-code-styling once; if it fails, use fallback for all codes
   352→      let QRCodeStyling: any = null;
   353→      if (!fallbackRef.current) {
   354→        try {
   355→          QRCodeStyling = (await import('qr-code-styling')).default;
   356→        } catch {
   357→          fallbackRef.current = true;
   358→          setShowFallback(true);
   359→        }
   360→      }
   361→
   362→      for (const code of generatedCodes) {
   363→        const data = `https://qrdomotik.com/activate/${code}`;
   364→        let imageUrl: string;
   365→
   366→        if (QRCodeStyling) {
   367→          // Preferred path: qr-code-styling → SVG → PNG
   368→          const opts: any = {
   369→            width: 400,
   370→            height: 400,
   371→            type: 'svg',
   372→            data,
   373→            dotsOptions: { color: design.dotsColor, type: design.dotsType },
   374→            backgroundOptions: { color: design.backgroundColor },
   375→            cornersSquareOptions: { color: design.dotsColor, type: design.cornersSquareType },
   376→            cornersDotOptions: { color: design.dotsColor, type: design.cornersDotType },
   377→            qrOptions: { errorCorrectionLevel: design.errorCorrectionLevel },
   378→            imageOptions: { crossOrigin: 'anonymous', margin: 8, imageSize: 0.35 },
   379→          };
   380→          if (design.logoPreset) {
   381→            opts.image = getPresetLogoDataUrl(design.logoPreset);
   382→          }
   383→          try {
   384→            const qr = new QRCodeStyling(opts);
   385→            const svgBlob: Blob = await qr.getRawData('svg');
   386→            imageUrl = await svgBlobToPngDataUrl(svgBlob, 400);
   387→          } catch {
   388→            // qr-code-styling loaded but getRawData failed — switch to fallback
   389→            fallbackRef.current = true;
   390→            setShowFallback(true);
   391→            imageUrl = await generateFallbackQrPng(
   392→              data, 400, design.dotsColor, design.backgroundColor, design.errorCorrectionLevel,
   393→            );
   394→          }
   395→        } else {
   396→          // Fallback path: qrcode.react → SVG → PNG
   397→          imageUrl = await generateFallbackQrPng(
   398→            data, 400, design.dotsColor, design.backgroundColor, design.errorCorrectionLevel,
   399→          );
   400→        }
   401→
   402→        qrCodes.push({ code, imageUrl });
   403→      }
   404→
   405→      await downloadPdf({
   406→        qrCodes,
   407→        format: 'a4',
   408→        batchName: batchName || `Lot ${batchId?.slice(0, 8)}`,
   409→      });
   410→
   411→      toast.success('PDF téléchargé !');
   412→    } catch (err: any) {
   413→      toast.error("Erreur lors de l'export PDF");
   414→      console.error(err);
   415→    } finally {
   416→      setIsExporting(false);
   417→    }
   418→  };
   419→
   420→  /* ---- design updaters ---- */
   421→  const updateDesign = (partial: Partial<DesignConfig>) => {
   422→    setDesign((prev) => ({ ...prev, ...partial }));
   423→    qrInstanceRef.current = null;
   424→  };
   425→
   426→  const presetIcon = (icon: string) => {
   427→    switch (icon) {
   428→      case 'wifi': return <Wifi className="h-4 w-4" />;
   429→      case 'home': return <Home className="h-4 w-4" />;
   430→      case 'list': return <List className="h-4 w-4" />;
   431→      case 'bell': return <Bell className="h-4 w-4" />;
   432→      case 'shield': return <ShieldCheck className="h-4 w-4" />;
   433→      default: return <ImageIcon className="h-4 w-4" />;
   434→    }
   435→  };
   436→
   437→  /* ================================================================ */
   438→  /*  RENDER                                                            */
   439→  /* ================================================================ */
   440→  return (
   441→    <div className="space-y-6">
   442→      {/* Header */}
   443→      <div>
   444→        <h2 className="text-2xl font-bold tracking-tight">Générer un lot de QR codes</h2>
   445→        <p className="text-muted-foreground">
   446→          Créez un lot de QR codes physiques avec un design personnalisé, puis exportez-les en PDF pour l'impression.
   447→        </p>
   448→      </div>
   449→
   450→      <div className="grid grid-cols-1 lg:grid-cols-5 gap-6">
   451→        {/* ====== LEFT: FORM ====== */}
   452→        <div className="lg:col-span-3 space-y-6">
   453→          {/* Templates */}
   454→          <Card>
   455→            <CardHeader className="pb-3">
   456→              <CardTitle className="flex items-center gap-2 text-base">
   457→                <Sparkles className="h-4 w-4 text-amber-500" />
   458→                Templates prêts à l'emploi
   459→              </CardTitle>
   460→              <CardDescription>Choisissez un template pour pré-remplir la configuration</CardDescription>
   461→            </CardHeader>
   462→            <CardContent>
   463→              <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">

   465→                  <button
   466→                    key={key}

   468→                    className={`relative flex flex-col items-start gap-1 rounded-lg border-2 p-4 text-left transition-all hover:shadow-md ${
   469→                      activeTemplate === key

   476→                        <Check className="h-4 w-4 text-primary" />
   477→                      </div>

   479→                    <span className="font-semibold text-sm">{tpl.label}</span>
   480→                    <span className="text-xs text-muted-foreground">{tpl.description}</span>
